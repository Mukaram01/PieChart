' ======================================================================
' WinCC Global Script: Status % Aggregation (Simplified / Production-Safe)
' ======================================================================
Option Explicit

Const SCRIPT_VERSION = "PieChart-simplified-v4-failfast-longwindow-cooldown"

Const DEFAULT_PROVIDER = "MSOLEDBSQL"
Const FALLBACK_PROVIDER = "SQLNCLI11"
Const SQL_INSTANCE = ".\WinCC"
Const INITIAL_CATALOG = "master"
Const SQL_TIMEOUT_SECONDS_1D = 12
Const SQL_TIMEOUT_SECONDS_7D = 15
Const SQL_TIMEOUT_SECONDS_30D = 15
Const GLOBAL_MIN_SECONDS_BETWEEN_MACHINE_RUNS = 120
Const WINDOW_EXEC_WARN_THRESHOLD_MS = 15000

Const THROTTLE_1D_MINUTES = 60
Const THROTTLE_7D_MINUTES = 1440
Const THROTTLE_30D_MINUTES = 10080

Const FAILURE_COOLDOWN_7D_MINUTES = 1440
Const FAILURE_COOLDOWN_30D_MINUTES = 10080

Const COUNT_TAG_PREFIX_JOINER = "\"
Const COUNT_TAG_CORE_TEMPLATE = "{MACHINE_ID}_ProgramStatus_{STATUS_NAME}"
Const COUNT_TAG_SUFFIX = "_Count"
Const FIXED_AGG_EXPR = "AVG(RealValue)"

Const adCmdText = 1
Const adStateOpen = 1

Const LOG_LEVEL_ERROR = 1
Const LOG_LEVEL_WARN = 2
Const LOG_LEVEL_INFO = 3
Const CURRENT_LOG_LEVEL = 3

Dim g_lastRunByMachinePeriod
Dim g_lastGlobalAggregationRun
Dim g_periodFailureCooldownUntil
Dim G_MACHINES
Dim g_inited

Private Function S(ByVal v)
    On Error Resume Next
    If IsNull(v) Or IsEmpty(v) Or IsObject(v) Then
        S = ""
    Else
        S = CStr(v)
        If Err.Number <> 0 Then
            S = ""
            Err.Clear
        End If
    End If
    On Error GoTo 0
End Function

Private Sub InitGlobals()
    If g_inited Then Exit Sub

    Set g_lastRunByMachinePeriod = CreateObject("Scripting.Dictionary")
    Set g_periodFailureCooldownUntil = CreateObject("Scripting.Dictionary")
    Set G_MACHINES = CreateObject("Scripting.Dictionary")

    G_MACHINES.Add "MNA_0269", "DMG_Mori1_MNA_0269"
    G_MACHINES.Add "MNA_0270", "DMG_Mori2_MNA_0270"
    G_MACHINES.Add "MNA_0332", "DMG_Mori3_MNA_0332"
    G_MACHINES.Add "MNA_0164", "Wheelabrator_MNP_0164"
    G_MACHINES.Add "MNA_0085", "Droop_Rein_1_MNA_0085_LM33H"
    G_MACHINES.Add "MNA_0087", "Droop_Rein_4_MNA_0087_LM33F"
    G_MACHINES.Add "MNA_0235", "WFL_M120_MNA_0235"
    G_MACHINES.Add "MNA_0237", "WFL_M50_MNA_0237"
    G_MACHINES.Add "MNA_0411", "Mazak_MNA_0411"

    g_inited = True
    LogMsg LOG_LEVEL_INFO, "InitGlobals", "initialized version=" & SCRIPT_VERSION
End Sub

Private Function IsMaster()
    On Error Resume Next
    IsMaster = (CLng(HMIRuntime.Tags("@RM_MASTER").Read) = 1)
    If Err.Number <> 0 Then
        IsMaster = True
        Err.Clear
    End If
    On Error GoTo 0
End Function

Private Function OpenConnection()
    On Error Resume Next
    Dim cn, prov

    prov = DEFAULT_PROVIDER
    Set cn = CreateObject("ADODB.Connection")
    cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "OpenConnection", "provider_fallback=" & FALLBACK_PROVIDER & " reason=" & S(Err.Description)
        Err.Clear
        prov = FALLBACK_PROVIDER
        Set cn = CreateObject("ADODB.Connection")
        cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"
    End If

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_ERROR, "OpenConnection", "sql_open_failed=" & S(Err.Description)
        Set cn = Nothing
        Err.Clear
    End If

    Set OpenConnection = cn
    On Error GoTo 0
End Function

Private Sub CloseConn(ByVal cn)
    On Error Resume Next
    If IsObject(cn) Then
        If Not (cn Is Nothing) Then
            If cn.State = adStateOpen Then cn.Close
        End If
    End If
    Set cn = Nothing
    On Error GoTo 0
End Sub

Private Function GetRuntimeCatalog()
    Dim cat
    On Error Resume Next
    cat = HMIRuntime.Tags("@DatasourceNameRT").Read
    If Err.Number <> 0 Or IsNull(cat) Or IsEmpty(cat) Or Len(CStr(cat)) = 0 Then
        Err.Clear
        GetRuntimeCatalog = "WinCC"
    Else
        GetRuntimeCatalog = CStr(cat)
    End If
    On Error GoTo 0
End Function

Private Function TS3(ByVal dt)
    TS3 = Year(dt) & "-" & Right("0" & Month(dt),2) & "-" & Right("0" & Day(dt),2) & _
          " " & Right("0" & Hour(dt),2) & ":" & Right("0" & Minute(dt),2) & ":" & Right("0" & Second(dt),2) & ".000"
End Function

Private Function ElapsedMs(ByVal timerStart)
    Dim timerEnd
    timerEnd = Timer
    If timerEnd < timerStart Then timerEnd = timerEnd + 86400
    ElapsedMs = CLng((timerEnd - timerStart) * 1000)
End Function

Private Sub LogMsg(ByVal level, ByVal src, ByVal msg)
    If level > CURRENT_LOG_LEVEL Then Exit Sub
    On Error Resume Next
    HMIRuntime.Trace FormatDateTime(Now(), vbShortDate) & " " & _
                     FormatDateTime(Now(), vbLongTime) & " [PieChart/" & src & "] " & S(msg) & vbCrLf
    On Error GoTo 0
End Sub

Private Sub WriteNumericOutputTag(ByVal tagName, ByVal value)
    On Error Resume Next
    HMIRuntime.Tags(tagName).Write CDbl(value)
    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "HMITag", "write_failed tag=" & tagName & " err=" & S(Err.Description)
        Err.Clear
    End If
    On Error GoTo 0
End Sub

Private Function ShouldRun(ByVal machineId, ByVal periodKey, ByVal throttleMinutes)
    Dim k, minsSince
    k = BuildMachinePeriodKey(machineId, periodKey)

    If Not g_lastRunByMachinePeriod.Exists(k) Then
        ShouldRun = True
        Exit Function
    End If

    minsSince = DateDiff("n", CDate(g_lastRunByMachinePeriod(k)), Now())
    ShouldRun = (minsSince >= throttleMinutes)
End Function

Private Function GetCommandTimeoutSeconds(ByVal periodKey)
    Select Case UCase(CStr(periodKey))
        Case "1D"
            GetCommandTimeoutSeconds = SQL_TIMEOUT_SECONDS_1D
        Case "7D"
            GetCommandTimeoutSeconds = SQL_TIMEOUT_SECONDS_7D
        Case "30D"
            GetCommandTimeoutSeconds = SQL_TIMEOUT_SECONDS_30D
        Case Else
            GetCommandTimeoutSeconds = SQL_TIMEOUT_SECONDS_1D
    End Select
End Function

Private Function BuildMachinePeriodKey(ByVal machineId, ByVal periodKey)
    BuildMachinePeriodKey = machineId & "|" & periodKey
End Function

Private Function GetFailureCooldownMinutes(ByVal periodKey)
    Select Case UCase(CStr(periodKey))
        Case "7D"
            GetFailureCooldownMinutes = FAILURE_COOLDOWN_7D_MINUTES
        Case "30D"
            GetFailureCooldownMinutes = FAILURE_COOLDOWN_30D_MINUTES
        Case Else
            GetFailureCooldownMinutes = 0
    End Select
End Function

Private Function BuildCountTagFullName(ByVal machineId, ByVal statusName)
    Dim basePrefix, baseCore
    basePrefix = G_MACHINES(machineId)
    baseCore = COUNT_TAG_CORE_TEMPLATE
    baseCore = Replace(baseCore, "{MACHINE_ID}", machineId)
    baseCore = Replace(baseCore, "{STATUS_NAME}", statusName)
    BuildCountTagFullName = basePrefix & COUNT_TAG_PREFIX_JOINER & baseCore & COUNT_TAG_SUFFIX
End Function

Private Function ReadSingleFraction(ByVal cn, ByVal runtimeCatalog, ByVal tagFullName, ByVal tFrom, ByVal tTo, ByVal machineId, ByVal periodKey, ByVal statusName)
    Dim cmd, rs, sqlText, safeTag, t0, ms
    Dim timeoutSeconds
    Dim v
    Dim resultClass
    Dim errText

    ReadSingleFraction = Empty
    safeTag = Replace(tagFullName, "'", "''")
    sqlText = "cp_TagStatisticEx '" & runtimeCatalog & "','TAG:R,''" & safeTag & "'',''" & TS3(tFrom) & "'',''" & TS3(tTo) & "''','" & FIXED_AGG_EXPR & "'"
    timeoutSeconds = GetCommandTimeoutSeconds(periodKey)
    resultClass = "error"
    errText = ""

    t0 = Timer
    On Error Resume Next
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = cn
    cmd.CommandType = adCmdText
    cmd.CommandTimeout = timeoutSeconds
    cmd.CommandText = sqlText
    Set rs = cmd.Execute
    ms = ElapsedMs(t0)

    If Err.Number <> 0 Then
        resultClass = "error"
        errText = S(Err.Description)
        Err.Clear
    ElseIf (rs Is Nothing) Then
        resultClass = "empty"
    ElseIf rs.EOF Then
        resultClass = "empty"
    Else
        v = rs.Fields(0).Value
        If Not IsNumeric(v) Then
            resultClass = "non_numeric"
        ElseIf CDbl(v) < 0 Or CDbl(v) > 1 Then
            resultClass = "out_of_range"
        Else
            ReadSingleFraction = CDbl(v)
            resultClass = "valid"
        End If
    End If

    If Not (rs Is Nothing) Then
        rs.Close
    End If

    If resultClass = "error" Then
        LogMsg LOG_LEVEL_INFO, "cp_TagStatisticEx", "machine=" & machineId & " period=" & periodKey & " status=" & statusName & " elapsed_ms=" & ms & " timeout_s=" & timeoutSeconds & " result=" & resultClass & " err=" & errText
    Else
        LogMsg LOG_LEVEL_INFO, "cp_TagStatisticEx", "machine=" & machineId & " period=" & periodKey & " status=" & statusName & " elapsed_ms=" & ms & " timeout_s=" & timeoutSeconds & " result=" & resultClass
    End If
    On Error GoTo 0

    Set rs = Nothing
    Set cmd = Nothing
End Function

Private Sub GetCombinedFractions(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal tFrom, ByVal tTo, ByVal periodKey, ByRef fRun, ByRef fHalt, ByRef fAbort, ByRef inputValid)
    Dim a, h, r, w, i

    inputValid = False
    fRun = 0
    fHalt = 0
    fAbort = 0

    a = ReadSingleFraction(cn, runtimeCatalog, BuildCountTagFullName(machineId, "Aborted"), tFrom, tTo, machineId, periodKey, "Aborted")
    If Not IsNumeric(a) Then
        inputValid = False
        fRun = 0
        fHalt = 0
        fAbort = 0
        LogMsg LOG_LEVEL_WARN, "GetCombinedFractions", "machine=" & machineId & " period=" & periodKey & " failed_status=Aborted reason=missing_or_invalid_status_fraction"
        Exit Sub
    End If

    h = ReadSingleFraction(cn, runtimeCatalog, BuildCountTagFullName(machineId, "Halted"), tFrom, tTo, machineId, periodKey, "Halted")
    If Not IsNumeric(h) Then
        inputValid = False
        fRun = 0
        fHalt = 0
        fAbort = 0
        LogMsg LOG_LEVEL_WARN, "GetCombinedFractions", "machine=" & machineId & " period=" & periodKey & " failed_status=Halted reason=missing_or_invalid_status_fraction"
        Exit Sub
    End If

    r = ReadSingleFraction(cn, runtimeCatalog, BuildCountTagFullName(machineId, "Running"), tFrom, tTo, machineId, periodKey, "Running")
    If Not IsNumeric(r) Then
        inputValid = False
        fRun = 0
        fHalt = 0
        fAbort = 0
        LogMsg LOG_LEVEL_WARN, "GetCombinedFractions", "machine=" & machineId & " period=" & periodKey & " failed_status=Running reason=missing_or_invalid_status_fraction"
        Exit Sub
    End If

    w = ReadSingleFraction(cn, runtimeCatalog, BuildCountTagFullName(machineId, "Waiting"), tFrom, tTo, machineId, periodKey, "Waiting")
    If Not IsNumeric(w) Then
        inputValid = False
        fRun = 0
        fHalt = 0
        fAbort = 0
        LogMsg LOG_LEVEL_WARN, "GetCombinedFractions", "machine=" & machineId & " period=" & periodKey & " failed_status=Waiting reason=missing_or_invalid_status_fraction"
        Exit Sub
    End If

    i = ReadSingleFraction(cn, runtimeCatalog, BuildCountTagFullName(machineId, "Interrupted"), tFrom, tTo, machineId, periodKey, "Interrupted")
    If Not IsNumeric(i) Then
        inputValid = False
        fRun = 0
        fHalt = 0
        fAbort = 0
        LogMsg LOG_LEVEL_WARN, "GetCombinedFractions", "machine=" & machineId & " period=" & periodKey & " failed_status=Interrupted reason=missing_or_invalid_status_fraction"
        Exit Sub
    End If

    fAbort = CDbl(a) + CDbl(w)
    fHalt = CDbl(h) + CDbl(i)
    fRun = CDbl(r)
    inputValid = True
End Sub

Private Sub NormalizeFractionsToPct(ByVal fRun, ByVal fHalt, ByVal fAbort, ByRef pRun, ByRef pHalt, ByRef pAbort)
    Dim tot
    tot = CDbl(fRun) + CDbl(fHalt) + CDbl(fAbort)

    If tot <= 0 Then
        pRun = 0
        pHalt = 0
        pAbort = 0
    Else
        pRun = Round((CDbl(fRun) / tot) * 100, 2)
        pHalt = Round((CDbl(fHalt) / tot) * 100, 2)
        pAbort = Round((CDbl(fAbort) / tot) * 100, 2)
    End If
End Sub

Private Function PieVal(ByVal pct)
    If pct <= 0 Then
        PieVal = 0
    ElseIf pct < 1 Then
        PieVal = 1
    Else
        PieVal = pct
    End If
End Function

Private Sub AggregatePeriod(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal lookbackMinutes)
    Dim tFrom, tTo, t0, elapsed
    Dim fRun, fHalt, fAbort, inputValid
    Dim pRun, pHalt, pAbort
    Dim base, totalCombined
    Dim failureCooldownMinutes, failureUntil, failureKey

    t0 = Timer
    tTo = Now()
    tFrom = DateAdd("n", -lookbackMinutes, tTo)

    LogMsg LOG_LEVEL_INFO, "Aggregate", "start machine=" & machineId & " period=" & periodKey & " from=" & TS3(tFrom) & " to=" & TS3(tTo)

    GetCombinedFractions cn, runtimeCatalog, machineId, tFrom, tTo, periodKey, fRun, fHalt, fAbort, inputValid

    If inputValid Then
        NormalizeFractionsToPct fRun, fHalt, fAbort, pRun, pHalt, pAbort
        totalCombined = CDbl(fRun) + CDbl(fHalt) + CDbl(fAbort)
    Else
        pRun = 0
        pHalt = 0
        pAbort = 0
        totalCombined = 0
    End If

    base = machineId & "_" & periodKey & "_Machine_"

    WriteNumericOutputTag base & "Running_Percentage", pRun
    WriteNumericOutputTag base & "Halted_Percentage", pHalt
    WriteNumericOutputTag base & "Aborted_Percentage", pAbort

    WriteNumericOutputTag base & "Running_Percentage_PieChart", PieVal(pRun)
    WriteNumericOutputTag base & "Halted_Percentage_PieChart", PieVal(pHalt)
    WriteNumericOutputTag base & "Aborted_Percentage_PieChart", PieVal(pAbort)

    elapsed = ElapsedMs(t0)
    If inputValid Then
        If totalCombined > 0 Then
            LogMsg LOG_LEVEL_INFO, "Aggregate", "end machine=" & machineId & " period=" & periodKey & " result=valid_nonzero run=" & pRun & " halt=" & pHalt & " abort=" & pAbort & " elapsed_ms=" & elapsed
        Else
            LogMsg LOG_LEVEL_WARN, "Aggregate", "end machine=" & machineId & " period=" & periodKey & " result=valid_zero_total run=0 halt=0 abort=0 elapsed_ms=" & elapsed
        End If
    Else
        failureCooldownMinutes = GetFailureCooldownMinutes(periodKey)
        If failureCooldownMinutes > 0 Then
            failureUntil = DateAdd("n", failureCooldownMinutes, Now())
            failureKey = BuildMachinePeriodKey(machineId, periodKey)
            g_periodFailureCooldownUntil(failureKey) = failureUntil
            LogMsg LOG_LEVEL_WARN, "Aggregate", "set_failure_cooldown machine=" & machineId & " period=" & periodKey & " cooldown_until=" & TS3(failureUntil)
        End If
        LogMsg LOG_LEVEL_WARN, "Aggregate", "end machine=" & machineId & " period=" & periodKey & " result=zero_no_data run=0 halt=0 abort=0 elapsed_ms=" & elapsed
    End If

    If elapsed > WINDOW_EXEC_WARN_THRESHOLD_MS Then
        LogMsg LOG_LEVEL_WARN, "Aggregate", "slow_period machine=" & machineId & " period=" & periodKey & " elapsed_ms=" & elapsed
    End If

    g_lastRunByMachinePeriod(BuildMachinePeriodKey(machineId, periodKey)) = Now()
End Sub

Public Sub MainStatusAggregation(ByVal machineId)
    Dim cn, runtimeCatalog
    Dim targetPeriod, lookback
    Dim secondsSinceLastGlobal
    Dim periodFailureKey, periodFailureUntil

    InitGlobals

    If Not IsMaster() Then Exit Sub

    If Not G_MACHINES.Exists(machineId) Then
        LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", "unknown_machine=" & machineId
        Exit Sub
    End If

    targetPeriod = ""
    lookback = 0

    If ShouldRun(machineId, "1D", THROTTLE_1D_MINUTES) Then
        targetPeriod = "1D"
        lookback = 1440
    ElseIf ShouldRun(machineId, "7D", THROTTLE_7D_MINUTES) Then
        targetPeriod = "7D"
        lookback = 10080
    ElseIf ShouldRun(machineId, "30D", THROTTLE_30D_MINUTES) Then
        targetPeriod = "30D"
        lookback = 43200
    Else
        LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "skip machine=" & machineId & " reason=throttled"
        Exit Sub
    End If

    If targetPeriod <> "1D" Then
        periodFailureKey = BuildMachinePeriodKey(machineId, targetPeriod)
        If g_periodFailureCooldownUntil.Exists(periodFailureKey) Then
            periodFailureUntil = CDate(g_periodFailureCooldownUntil(periodFailureKey))
            If Now() < periodFailureUntil Then
                LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "skip machine=" & machineId & " period=" & targetPeriod & " reason=period_failure_cooldown cooldown_until=" & TS3(periodFailureUntil)
                Exit Sub
            Else
                g_periodFailureCooldownUntil.Remove periodFailureKey
            End If
        End If
    End If

    If Not IsEmpty(g_lastGlobalAggregationRun) Then
        secondsSinceLastGlobal = DateDiff("s", CDate(g_lastGlobalAggregationRun), Now())
        If secondsSinceLastGlobal < GLOBAL_MIN_SECONDS_BETWEEN_MACHINE_RUNS Then
            LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "skip machine=" & machineId & " reason=global_cooldown cooldown_seconds=" & GLOBAL_MIN_SECONDS_BETWEEN_MACHINE_RUNS & " seconds_since_last_run=" & secondsSinceLastGlobal
            Exit Sub
        End If
    End If

    Set cn = OpenConnection()
    If cn Is Nothing Then
        LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", "skip machine=" & machineId & " reason=no_sql_connection"
        Exit Sub
    End If

    runtimeCatalog = GetRuntimeCatalog()
    LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "process machine=" & machineId & " period=" & targetPeriod & " version=" & SCRIPT_VERSION

    g_lastGlobalAggregationRun = Now()
    AggregatePeriod cn, runtimeCatalog, machineId, targetPeriod, lookback

    CloseConn cn
End Sub
