' ======================================================================
'  WinCC Global Script: Status % Aggregation (FINAL)
'
'  - Uses ONLY *_Count tags (binary 0/1 archived every 500ms)
'  - Reads 5 base statuses:
'       Aborted, Halted, Running, Waiting, Interrupted
'  - Combines into 3 outputs (names UNCHANGED):
'       Aborted  = Aborted + Waiting
'       Halted   = Halted  + Interrupted
'       Running  = Running
'
'  OUTPUT TAGS (unchanged):
'    <Machine>_<Period>_Machine_<Status>_Percentage              (can be 0)
'    <Machine>_<Period>_Machine_<Status>_Percentage_PieChart     (min 1)
'
'  Periods: 1D / 7D / 30D
' ======================================================================

Option Explicit

' ─────────────────────────────────────────────────────────────────────
' Configuration
' ─────────────────────────────────────────────────────────────────────
Const DEFAULT_PROVIDER       = "MSOLEDBSQL"
Const FALLBACK_PROVIDER      = "SQLNCLI11"
Const SQL_INSTANCE           = ".\WinCC"
Const INITIAL_CATALOG        = "master"
Const SQL_TIMEOUT_SECONDS    = 600

Const THROTTLE_1D_MINUTES    = 10
Const THROTTLE_7D_MINUTES    = 15
Const THROTTLE_30D_MINUTES   = 57

Const adCmdText              = 1
Const adStateOpen            = 1

' ─────────────────────────────────────────────────────────────────────
' Log Levels
' ─────────────────────────────────────────────────────────────────────
Const LOG_LEVEL_ERROR        = 1
Const LOG_LEVEL_WARN         = 2
Const LOG_LEVEL_INFO         = 3
Const LOG_LEVEL_DEBUG        = 4

' IMPORTANT: must be literal in WinCC VBScript
Const CURRENT_LOG_LEVEL      = 4   ' DEBUG

' ─────────────────────────────────────────────────────────────────────
' Runtime state
' ─────────────────────────────────────────────────────────────────────
Dim g_lastRunByMachinePeriod
Dim G_MACHINES
Dim g_inited

' ─────────────────────────────────────────────────────────────────────
' Redundancy helper
' ─────────────────────────────────────────────────────────────────────
Private Function IsMaster()
    On Error Resume Next
    IsMaster = (CLng(HMIRuntime.Tags("@RM_MASTER").Read) = 1)
    If Err.Number <> 0 Then
        ' Fail-open for standalone / tag not present
        IsMaster = True
        Err.Clear
    End If
    On Error GoTo 0
End Function

' ─────────────────────────────────────────────────────────────────────
' Init dictionaries
' ─────────────────────────────────────────────────────────────────────
Private Sub InitGlobals()
    If g_inited Then Exit Sub

    Set g_lastRunByMachinePeriod = CreateObject("Scripting.Dictionary")

    Set G_MACHINES = CreateObject("Scripting.Dictionary")
    ' MachineId → Tag-logging prefix (left of backslash)
    G_MACHINES.Add "MNA_0269", "DMG_Mori1_MNA_0269"
    G_MACHINES.Add "MNA_0270", "DMG_Mori2_MNA_0270"
    G_MACHINES.Add "MNA_0332", "DMG_Mori3_MNA_0332"

    ' Wheelabrator
    G_MACHINES.Add "MNA_0164", "Wheelabrator_MNP_0164"

    ' Droop & Rein
    G_MACHINES.Add "MNA_0085", "Droop_Rein_1_MNA_0085_LM33H"
    G_MACHINES.Add "MNA_0087", "Droop_Rein_4_MNA_0087_LM33F"

    g_inited = True
End Sub

' ─────────────────────────────────────────────────────────────────────
' Logging (Null-safe)
' ─────────────────────────────────────────────────────────────────────
Private Function S(ByVal v)
    On Error Resume Next
    If IsNull(v) Or IsEmpty(v) Or IsObject(v) Then
        S = ""
    Else
        S = CStr(v)
        If Err.Number <> 0 Then
            S = ""
            Err.Clear
        End If
    End If
    On Error GoTo 0
End Function

Private Sub LogMsg(ByVal level, ByVal src, ByVal msg)
    If level > CURRENT_LOG_LEVEL Then Exit Sub
    On Error Resume Next
    HMIRuntime.Trace _
        FormatDateTime(Now(), vbShortDate) & " " & _
        FormatDateTime(Now(), vbLongTime) & " [" & src & "] " & S(msg) & vbCrLf
    On Error GoTo 0
End Sub

Private Sub SleepMs(ByVal ms)
    On Error Resume Next
    HMIRuntime.Sleep ms
    On Error GoTo 0
End Sub

' ─────────────────────────────────────────────────────────────────────
' SQL helpers
' ─────────────────────────────────────────────────────────────────────
Private Function OpenConnection()
    On Error Resume Next
    Dim cn, prov
    prov = DEFAULT_PROVIDER

    Set cn = CreateObject("ADODB.Connection")
    cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "OpenConnection", "Switching provider to " & FALLBACK_PROVIDER & " (" & S(Err.Description) & ")"
        Err.Clear
        prov = FALLBACK_PROVIDER
        cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"
    End If

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_ERROR, "OpenConnection", "ERROR: " & S(Err.Description)
        Set cn = Nothing
        Err.Clear
    End If

    Set OpenConnection = cn
    On Error GoTo 0
End Function

Private Sub CloseConn(ByVal cn)
    On Error Resume Next
    If Not cn Is Nothing Then If cn.State = adStateOpen Then cn.Close
    Set cn = Nothing
    On Error GoTo 0
End Sub

Private Function GetRuntimeCatalog()
    On Error Resume Next
    Dim cat
    cat = HMIRuntime.Tags("@DatasourceNameRT").Read
    If Err.Number <> 0 Or IsNull(cat) Or IsEmpty(cat) Then
        Err.Clear
        GetRuntimeCatalog = "WinCC"
    ElseIf Len(CStr(cat)) > 0 Then
        GetRuntimeCatalog = CStr(cat)
    Else
        GetRuntimeCatalog = "WinCC"
    End If
    On Error GoTo 0
End Function

Private Function TS3(ByVal dt)
    TS3 = Year(dt) & "-" & Right("0" & Month(dt),2) & "-" & Right("0" & Day(dt),2) & _
          " " & Right("0" & Hour(dt),2) & ":" & Right("0" & Minute(dt),2) & ":" & Right("0" & Second(dt),2) & ".000"
End Function

Private Function ElapsedMs(ByVal timerStart)
    Dim timerEnd
    timerEnd = Timer
    If timerEnd < timerStart Then timerEnd = timerEnd + 86400
    ElapsedMs = CLng((timerEnd - timerStart) * 1000)
End Function

' ─────────────────────────────────────────────────────────────────────
' cp_TagStatisticEx (3 retries, null-safe)
' ─────────────────────────────────────────────────────────────────────
Private Function ExecCpTagStatisticEx(ByVal cn, ByVal runtimeCatalog, ByVal tagFullName, ByVal tFrom, ByVal tTo, ByVal aggList, ByVal machineId, ByVal periodKey, ByVal statusName)
    Dim i, cmd, rs, sSql, safeTag, fieldValue
    Dim startTimer, startedAt, completedAt, elapsedMsValue
    Dim returnedRows, valueText, infoMsg

    startedAt = Now
    startTimer = Timer
    returnedRows = 0
    valueText = ""

    safeTag = Replace(tagFullName, "'", "''")

    For i = 1 To 3
        On Error Resume Next

        Set cmd = CreateObject("ADODB.Command")
        Set cmd.ActiveConnection = cn
        cmd.CommandTimeout = SQL_TIMEOUT_SECONDS
        cmd.CommandType = adCmdText

        sSql = "cp_TagStatisticEx '" & runtimeCatalog & "'," & _
               "'TAG:R,''" & safeTag & "'',''" & TS3(tFrom) & "'',''" & TS3(tTo) & "'''," & _
               "'" & aggList & "'"

        cmd.CommandText = sSql

        If CURRENT_LOG_LEVEL >= LOG_LEVEL_DEBUG Then
            LogMsg LOG_LEVEL_DEBUG, "cp_TagStatisticEx", "SQL=[" & sSql & "]"
        End If

        Set rs = cmd.Execute

        If Err.Number = 0 Then
            ExecCpTagStatisticEx = Empty

            If Not (rs Is Nothing) Then
                If Not rs.EOF Then
                    returnedRows = 1
                    If rs.Fields.Count >= 1 Then
                        fieldValue = rs.Fields(0).Value
                        If Not (IsNull(fieldValue) Or IsEmpty(fieldValue)) Then
                            ExecCpTagStatisticEx = fieldValue
                            valueText = CStr(fieldValue)
                        Else
                            valueText = "<null>"
                        End If
                    End If
                End If
                rs.Close
            End If

            Set rs = Nothing
            Set cmd = Nothing
            On Error GoTo 0

            completedAt = Now
            elapsedMsValue = ElapsedMs(startTimer)
            If valueText = "" Then valueText = "<empty>"
            infoMsg = "machine=" & machineId & _
                      " window=" & periodKey & _
                      " status=" & statusName & _
                      " sql_from=" & TS3(tFrom) & _
                      " sql_to=" & TS3(tTo) & _
                      " started=" & TS3(startedAt) & _
                      " ended=" & TS3(completedAt) & _
                      " elapsed_ms=" & elapsedMsValue & _
                      " success=true" & _
                      " rows=" & returnedRows & _
                      " value=" & valueText
            LogMsg LOG_LEVEL_INFO, "cp_TagStatisticEx", infoMsg
            Exit Function
        Else
            LogMsg LOG_LEVEL_WARN, "cp_TagStatisticEx", "Attempt " & i & " failed: " & S(Err.Description)
            Err.Clear
            On Error GoTo 0
            SleepMs 500
        End If
    Next

    ExecCpTagStatisticEx = Empty
    completedAt = Now
    elapsedMsValue = ElapsedMs(startTimer)
    infoMsg = "machine=" & machineId & _
              " window=" & periodKey & _
              " status=" & statusName & _
              " sql_from=" & TS3(tFrom) & _
              " sql_to=" & TS3(tTo) & _
              " started=" & TS3(startedAt) & _
              " ended=" & TS3(completedAt) & _
              " elapsed_ms=" & elapsedMsValue & _
              " success=false rows=0 value=<none>"
    LogMsg LOG_LEVEL_INFO, "cp_TagStatisticEx", infoMsg
End Function

Private Function CpAggregate(ByVal cn, ByVal runtimeCatalog, ByVal tagFullName, ByVal t1, ByVal t2, ByVal agg, ByVal machineId, ByVal periodKey, ByVal statusName)
    Dim res
    res = ExecCpTagStatisticEx(cn, runtimeCatalog, tagFullName, t1, t2, agg, machineId, periodKey, statusName)

    If IsNull(res) Or IsEmpty(res) Or (CStr(res) = "") Then
        CpAggregate = Empty
    ElseIf IsNumeric(res) Then
        CpAggregate = CDbl(res)
    Else
        CpAggregate = Empty
    End If
End Function

Private Function TryAvgFraction(ByVal cn, ByVal runtimeCatalog, ByVal tagFull, ByVal t1, ByVal t2, ByVal machineId, ByVal periodKey, ByVal statusName)
    Dim v

    v = CpAggregate(cn, runtimeCatalog, tagFull, t1, t2, "AVG(RealValue)", machineId, periodKey, statusName)
    If IsNumeric(v) Then If v >= 0 And v <= 1 Then TryAvgFraction = v : Exit Function

    v = CpAggregate(cn, runtimeCatalog, tagFull, t1, t2, "AVG(Value)", machineId, periodKey, statusName)
    If IsNumeric(v) Then If v >= 0 And v <= 1 Then TryAvgFraction = v : Exit Function

    v = CpAggregate(cn, runtimeCatalog, tagFull, t1, t2, "AVG(ValueScaled)", machineId, periodKey, statusName)
    If IsNumeric(v) Then If v >= 0 And v <= 1 Then TryAvgFraction = v : Exit Function

    TryAvgFraction = Empty
End Function

' ─────────────────────────────────────────────────────────────────────
' READ ONLY *_Count (your requirement)
' ─────────────────────────────────────────────────────────────────────
Private Function GetCountFraction(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal statusName, ByVal periodKey, ByVal t1, ByVal t2)
    Dim basePrefix, baseCore, tagFull, v

    basePrefix = G_MACHINES(machineId)
    baseCore   = machineId & "_ProgramStatus_" & statusName

    tagFull = basePrefix & "\" & baseCore & "_Count"
    v = TryAvgFraction(cn, runtimeCatalog, tagFull, t1, t2, machineId, periodKey, statusName)

    If IsNumeric(v) Then
        GetCountFraction = CDbl(v)
    Else
        ' treat missing as 0 (do NOT abort)
        GetCountFraction = 0
    End If
End Function

' ─────────────────────────────────────────────────────────────────────
' Combine 5 → 3 (your rule), normalize if overlap makes sum > 1
' ─────────────────────────────────────────────────────────────────────
Private Sub GetCombinedFractions(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal t1, ByVal t2, ByRef fRun, ByRef fHalt, ByRef fAbort)
    Dim a, h, r, w, i, total

    a = GetCountFraction(cn, runtimeCatalog, machineId, "Aborted",     periodKey, t1, t2)
    h = GetCountFraction(cn, runtimeCatalog, machineId, "Halted",      periodKey, t1, t2)
    r = GetCountFraction(cn, runtimeCatalog, machineId, "Running",     periodKey, t1, t2)
    w = GetCountFraction(cn, runtimeCatalog, machineId, "Waiting",     periodKey, t1, t2)
    i = GetCountFraction(cn, runtimeCatalog, machineId, "Interrupted", periodKey, t1, t2)

    fRun   = r
    fHalt  = h + i
    fAbort = a + w

    ' clamp just in case
    If fRun < 0 Then fRun = 0
    If fHalt < 0 Then fHalt = 0
    If fAbort < 0 Then fAbort = 0

    ' If your PLC logic overlaps states, sum can exceed 1 → normalize safely
    total = fRun + fHalt + fAbort
    If total > 1.000001 Then
        fRun   = fRun / total
        fHalt  = fHalt / total
        fAbort = fAbort / total
    End If
End Sub

' ─────────────────────────────────────────────────────────────────────
' Output helpers
' ─────────────────────────────────────────────────────────────────────
Private Function OutTagName(ByVal machineId, ByVal periodKey, ByVal statusDispName, ByVal pieChart)
    If pieChart Then
        OutTagName = machineId & "_" & periodKey & "_Machine_" & statusDispName & "_Percentage_PieChart"
    Else
        OutTagName = machineId & "_" & periodKey & "_Machine_" & statusDispName & "_Percentage"
    End If
End Function

Private Sub WriteTag(ByVal tag, ByVal val)
    On Error Resume Next
    HMIRuntime.Tags(tag).Write val
    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "HMITag", "Write failed: " & S(tag) & " -> " & S(Err.Description)
        Err.Clear
    End If
    On Error GoTo 0
End Sub

' ─────────────────────────────────────────────────────────────────────
' Aggregate one period (writes Aborted/Halted/Running only)
' ─────────────────────────────────────────────────────────────────────
Private Sub AggregatePeriod(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal startDt, ByVal endDt)
    Dim fRun, fHalt, fAbort
    Dim pctRun, pctHalt, pctAbort
    Dim tagPct, tagPie, pieVal

    LogMsg LOG_LEVEL_INFO, "Aggregate", machineId & " " & periodKey & "  " & TS3(startDt) & " -> " & TS3(endDt)

    Call GetCombinedFractions(cn, runtimeCatalog, machineId, periodKey, startDt, endDt, fRun, fHalt, fAbort)

    pctRun   = Round(CDbl(fRun)   * 100, 2)
    pctHalt  = Round(CDbl(fHalt)  * 100, 2)
    pctAbort = Round(CDbl(fAbort) * 100, 2)

    ' main % (can be 0)
    tagPct = OutTagName(machineId, periodKey, "Running", False): WriteTag tagPct, pctRun
    tagPct = OutTagName(machineId, periodKey, "Halted",  False): WriteTag tagPct, pctHalt
    tagPct = OutTagName(machineId, periodKey, "Aborted", False): WriteTag tagPct, pctAbort

    ' pie chart (min 1 for display, DOES NOT affect main %)
    pieVal = pctRun:   If pieVal < 1 Then pieVal = 1
    tagPie = OutTagName(machineId, periodKey, "Running", True): WriteTag tagPie, pieVal

    pieVal = pctHalt:  If pieVal < 1 Then pieVal = 1
    tagPie = OutTagName(machineId, periodKey, "Halted", True): WriteTag tagPie, pieVal

    pieVal = pctAbort: If pieVal < 1 Then pieVal = 1
    tagPie = OutTagName(machineId, periodKey, "Aborted", True): WriteTag tagPie, pieVal
End Sub

Private Function ShouldRun(ByVal lastRun, ByVal throttleMin)
    If IsEmpty(lastRun) Or IsNull(lastRun) Then
        ShouldRun = True
        Exit Function
    End If
    ShouldRun = (DateDiff("n", lastRun, Now()) >= throttleMin)
End Function

' ─────────────────────────────────────────────────────────────────────
' Main Entry Point
' ─────────────────────────────────────────────────────────────────────
Public Sub MainStatusAggregation(ByVal machineId)
    If Not IsMaster() Then Exit Sub

    Dim cn, rtCat, nowDT
    Dim tFrom, tTo
    Dim key, lastRun

    InitGlobals

    If Not G_MACHINES.Exists(machineId) Then
        LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", "Unknown machineId: " & S(machineId)
        Exit Sub
    End If

    Set cn = OpenConnection()
    If cn Is Nothing Or cn.State <> adStateOpen Then
        LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", "ERROR: cannot open SQL connection"
        Exit Sub
    End If

    rtCat = GetRuntimeCatalog()
    nowDT = Now()

    ' 1D
    key = machineId & "|1D"
    lastRun = Empty
    If g_lastRunByMachinePeriod.Exists(key) Then lastRun = g_lastRunByMachinePeriod(key)
    If ShouldRun(lastRun, THROTTLE_1D_MINUTES) Then
        g_lastRunByMachinePeriod(key) = nowDT
        tTo = nowDT: tFrom = DateAdd("d", -1, tTo)
        AggregatePeriod cn, rtCat, machineId, "1D", tFrom, tTo
    End If

    ' 7D
    key = machineId & "|7D"
    lastRun = Empty
    If g_lastRunByMachinePeriod.Exists(key) Then lastRun = g_lastRunByMachinePeriod(key)
    If ShouldRun(lastRun, THROTTLE_7D_MINUTES) Then
        g_lastRunByMachinePeriod(key) = nowDT
        tTo = nowDT: tFrom = DateAdd("d", -7, tTo)
        AggregatePeriod cn, rtCat, machineId, "7D", tFrom, tTo
    End If

    ' 30D
    key = machineId & "|30D"
    lastRun = Empty
    If g_lastRunByMachinePeriod.Exists(key) Then lastRun = g_lastRunByMachinePeriod(key)
    If ShouldRun(lastRun, THROTTLE_30D_MINUTES) Then
        g_lastRunByMachinePeriod(key) = nowDT
        tTo = nowDT: tFrom = DateAdd("d", -30, tTo)
        AggregatePeriod cn, rtCat, machineId, "30D", tFrom, tTo
    End If

    CloseConn cn
End Sub
