' ======================================================================
' WinCC Global Script: Status % Aggregation (Simplified / Production-Safe)
' ======================================================================
Option Explicit

Const DEFAULT_PROVIDER = "MSOLEDBSQL"
Const FALLBACK_PROVIDER = "SQLNCLI11"
Const SQL_INSTANCE = ".\WinCC"
Const INITIAL_CATALOG = "master"
Const SQL_TIMEOUT_SECONDS = 45
Const WINDOW_EXEC_WARN_THRESHOLD_MS = 15000

Const THROTTLE_1D_MINUTES = 30
Const THROTTLE_7D_MINUTES = 360
Const THROTTLE_30D_MINUTES = 1440

Const COUNT_TAG_PREFIX_JOINER = "\"
Const COUNT_TAG_CORE_TEMPLATE = "{MACHINE_ID}_ProgramStatus_{STATUS_NAME}"
Const COUNT_TAG_SUFFIX = "_Count"
Const FIXED_AGG_EXPR = "AVG(RealValue)"

Const adCmdText = 1
Const adStateOpen = 1

Const LOG_LEVEL_ERROR = 1
Const LOG_LEVEL_WARN = 2
Const LOG_LEVEL_INFO = 3
Const CURRENT_LOG_LEVEL = 3

Dim g_lastRunByMachinePeriod
Dim G_MACHINES
Dim g_inited

Private Function S(ByVal v)
    On Error Resume Next
    If IsNull(v) Or IsEmpty(v) Or IsObject(v) Then
        S = ""
    Else
        S = CStr(v)
        If Err.Number <> 0 Then
            S = ""
            Err.Clear
        End If
    End If
    On Error GoTo 0
End Function

Private Function TS3(ByVal dt)
    TS3 = Year(dt) & "-" & Right("0" & Month(dt),2) & "-" & Right("0" & Day(dt),2) & _
          " " & Right("0" & Hour(dt),2) & ":" & Right("0" & Minute(dt),2) & ":" & Right("0" & Second(dt),2) & ".000"
End Function

Private Function ElapsedMs(ByVal timerStart)
    Dim timerEnd
    timerEnd = Timer
    If timerEnd < timerStart Then timerEnd = timerEnd + 86400
    ElapsedMs = CLng((timerEnd - timerStart) * 1000)
End Function

Private Sub LogMsg(ByVal level, ByVal src, ByVal msg)
    If level > CURRENT_LOG_LEVEL Then Exit Sub
    On Error Resume Next
    HMIRuntime.Trace FormatDateTime(Now(), vbShortDate) & " " & _
                     FormatDateTime(Now(), vbLongTime) & " [PieChart/" & src & "] " & S(msg) & vbCrLf
    On Error GoTo 0
End Sub

Private Function IsMaster()
    On Error Resume Next
    IsMaster = (CLng(HMIRuntime.Tags("@RM_MASTER").Read) = 1)
    If Err.Number <> 0 Then
        IsMaster = True
        Err.Clear
    End If
    On Error GoTo 0
End Function

Private Sub InitGlobals()
    If g_inited Then Exit Sub

    Set g_lastRunByMachinePeriod = CreateObject("Scripting.Dictionary")
    Set G_MACHINES = CreateObject("Scripting.Dictionary")

    G_MACHINES.Add "MNA_0269", "DMG_Mori1_MNA_0269"
    G_MACHINES.Add "MNA_0270", "DMG_Mori2_MNA_0270"
    G_MACHINES.Add "MNA_0332", "DMG_Mori3_MNA_0332"
    G_MACHINES.Add "MNA_0164", "Wheelabrator_MNP_0164"
    G_MACHINES.Add "MNA_0085", "Droop_Rein_1_MNA_0085_LM33H"
    G_MACHINES.Add "MNA_0087", "Droop_Rein_4_MNA_0087_LM33F"
    G_MACHINES.Add "MNA_0235", "WFL_M120_MNA_0235"
    G_MACHINES.Add "MNA_0237", "WFL_M50_MNA_0237"
    G_MACHINES.Add "MNA_0411", "Mazak_MNA_0411"

    g_inited = True
    LogMsg LOG_LEVEL_INFO, "InitGlobals", "initialized"
End Sub

Private Function OpenConnection()
    On Error Resume Next
    Dim cn, prov

    prov = DEFAULT_PROVIDER
    Set cn = CreateObject("ADODB.Connection")
    cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "InitGlobals", "provider_fallback=" & FALLBACK_PROVIDER & " reason=" & S(Err.Description)
        Err.Clear
        prov = FALLBACK_PROVIDER
        Set cn = CreateObject("ADODB.Connection")
        cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"
    End If

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_ERROR, "InitGlobals", "sql_open_failed=" & S(Err.Description)
        Set cn = Nothing
        Err.Clear
    End If

    Set OpenConnection = cn
    On Error GoTo 0
End Function

Private Sub CloseConn(ByVal cn)
    On Error Resume Next
    If IsObject(cn) Then
        If Not (cn Is Nothing) Then
            If cn.State = adStateOpen Then cn.Close
        End If
    End If
    Set cn = Nothing
    On Error GoTo 0
End Sub

Private Function GetRuntimeCatalog()
    Dim cat
    On Error Resume Next
    cat = HMIRuntime.Tags("@DatasourceNameRT").Read
    If Err.Number <> 0 Or IsNull(cat) Or IsEmpty(cat) Or Len(CStr(cat)) = 0 Then
        Err.Clear
        GetRuntimeCatalog = "WinCC"
    Else
        GetRuntimeCatalog = CStr(cat)
    End If
    On Error GoTo 0
End Function

Private Sub WriteTag(ByVal tagName, ByVal value)
    On Error Resume Next
    HMIRuntime.Tags(tagName).Write CDbl(value)
    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "HMITag", "write_failed tag=" & tagName & " err=" & S(Err.Description)
        Err.Clear
    End If
    On Error GoTo 0
End Sub

Private Function ShouldRun(ByVal machineId, ByVal periodKey, ByVal throttleMinutes)
    Dim k, minsSince
    k = machineId & "|" & periodKey

    If Not g_lastRunByMachinePeriod.Exists(k) Then
        ShouldRun = True
        Exit Function
    End If

    minsSince = DateDiff("n", CDate(g_lastRunByMachinePeriod(k)), Now())
    ShouldRun = (minsSince >= throttleMinutes)
End Function

Private Function BuildCountTagFullName(ByVal machineId, ByVal statusName)
    Dim basePrefix, baseCore
    basePrefix = G_MACHINES(machineId)
    baseCore = COUNT_TAG_CORE_TEMPLATE
    baseCore = Replace(baseCore, "{MACHINE_ID}", machineId)
    baseCore = Replace(baseCore, "{STATUS_NAME}", statusName)
    BuildCountTagFullName = basePrefix & COUNT_TAG_PREFIX_JOINER & baseCore & COUNT_TAG_SUFFIX
End Function

Private Function ReadSingleFraction(ByVal cn, ByVal runtimeCatalog, ByVal tagFullName, ByVal tFrom, ByVal tTo, ByVal machineId, ByVal periodKey, ByVal statusName)
    Dim cmd, rs, sqlText, safeTag, t0, ms
    Dim v

    ReadSingleFraction = Empty
    safeTag = Replace(tagFullName, "'", "''")
    sqlText = "cp_TagStatisticEx '" & runtimeCatalog & "','TAG:R,''" & safeTag & "'',''" & TS3(tFrom) & "'',''" & TS3(tTo) & "''','" & FIXED_AGG_EXPR & "'"

    t0 = Timer
    On Error Resume Next
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = cn
    cmd.CommandType = adCmdText
    cmd.CommandTimeout = SQL_TIMEOUT_SECONDS
    cmd.CommandText = sqlText
    Set rs = cmd.Execute
    ms = ElapsedMs(t0)

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "cp_TagStatisticEx", "machine=" & machineId & " period=" & periodKey & " status=" & statusName & " elapsed_ms=" & ms & " ok=false err=" & S(Err.Description)
        Err.Clear
    Else
        If Not (rs Is Nothing) Then
            If Not rs.EOF Then
                v = rs.Fields(0).Value
                If IsNumeric(v) Then
                    If CDbl(v) >= 0 And CDbl(v) <= 1 Then
                        ReadSingleFraction = CDbl(v)
                    End If
                End If
            End If
            rs.Close
        End If
        LogMsg LOG_LEVEL_INFO, "cp_TagStatisticEx", "machine=" & machineId & " period=" & periodKey & " status=" & statusName & " elapsed_ms=" & ms & " ok=true"
    End If
    On Error GoTo 0

    Set rs = Nothing
    Set cmd = Nothing
End Function

Private Sub GetCombinedFractions(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal tFrom, ByVal tTo, ByVal periodKey, ByRef fRun, ByRef fHalt, ByRef fAbort, ByRef inputValid)
    Dim a, h, r, w, i

    inputValid = False
    fRun = 0
    fHalt = 0
    fAbort = 0

    a = ReadSingleFraction(cn, runtimeCatalog, BuildCountTagFullName(machineId, "Aborted"), tFrom, tTo, machineId, periodKey, "Aborted")
    h = ReadSingleFraction(cn, runtimeCatalog, BuildCountTagFullName(machineId, "Halted"), tFrom, tTo, machineId, periodKey, "Halted")
    r = ReadSingleFraction(cn, runtimeCatalog, BuildCountTagFullName(machineId, "Running"), tFrom, tTo, machineId, periodKey, "Running")
    w = ReadSingleFraction(cn, runtimeCatalog, BuildCountTagFullName(machineId, "Waiting"), tFrom, tTo, machineId, periodKey, "Waiting")
    i = ReadSingleFraction(cn, runtimeCatalog, BuildCountTagFullName(machineId, "Interrupted"), tFrom, tTo, machineId, periodKey, "Interrupted")

    If IsNumeric(a) And IsNumeric(h) And IsNumeric(r) And IsNumeric(w) And IsNumeric(i) Then
        fAbort = CDbl(a) + CDbl(w)
        fHalt = CDbl(h) + CDbl(i)
        fRun = CDbl(r)
        inputValid = True
    End If
End Sub

Private Sub NormalizeFractionsToPct(ByVal fRun, ByVal fHalt, ByVal fAbort, ByRef pRun, ByRef pHalt, ByRef pAbort)
    Dim tot
    tot = CDbl(fRun) + CDbl(fHalt) + CDbl(fAbort)

    If tot <= 0 Then
        pRun = 0
        pHalt = 0
        pAbort = 0
    Else
        pRun = Round((CDbl(fRun) / tot) * 100, 2)
        pHalt = Round((CDbl(fHalt) / tot) * 100, 2)
        pAbort = Round((CDbl(fAbort) / tot) * 100, 2)
    End If
End Sub

Private Function PieVal(ByVal pct)
    If pct <= 0 Then
        PieVal = 0
    ElseIf pct < 1 Then
        PieVal = 1
    Else
        PieVal = pct
    End If
End Function

Private Sub AggregatePeriod(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal lookbackMinutes)
    Dim tFrom, tTo, t0, elapsed
    Dim fRun, fHalt, fAbort, inputValid
    Dim pRun, pHalt, pAbort
    Dim base

    t0 = Timer
    tTo = Now()
    tFrom = DateAdd("n", -lookbackMinutes, tTo)

    LogMsg LOG_LEVEL_INFO, "Aggregate", "start machine=" & machineId & " period=" & periodKey & " from=" & TS3(tFrom) & " to=" & TS3(tTo)

    GetCombinedFractions cn, runtimeCatalog, machineId, tFrom, tTo, periodKey, fRun, fHalt, fAbort, inputValid

    If inputValid Then
        NormalizeFractionsToPct fRun, fHalt, fAbort, pRun, pHalt, pAbort
    Else
        pRun = 0
        pHalt = 0
        pAbort = 0
    End If

    base = machineId & "_" & periodKey & "_Machine_"

    WriteTag base & "Running_Percentage", pRun
    WriteTag base & "Halted_Percentage", pHalt
    WriteTag base & "Aborted_Percentage", pAbort

    WriteTag base & "Running_Percentage_PieChart", PieVal(pRun)
    WriteTag base & "Halted_Percentage_PieChart", PieVal(pHalt)
    WriteTag base & "Aborted_Percentage_PieChart", PieVal(pAbort)

    elapsed = ElapsedMs(t0)
    If inputValid Then
        LogMsg LOG_LEVEL_INFO, "Aggregate", "end machine=" & machineId & " period=" & periodKey & " result=valid run=" & pRun & " halt=" & pHalt & " abort=" & pAbort & " elapsed_ms=" & elapsed
    Else
        LogMsg LOG_LEVEL_WARN, "Aggregate", "end machine=" & machineId & " period=" & periodKey & " result=zero_no_data elapsed_ms=" & elapsed
    End If

    If elapsed > WINDOW_EXEC_WARN_THRESHOLD_MS Then
        LogMsg LOG_LEVEL_WARN, "Aggregate", "slow_period machine=" & machineId & " period=" & periodKey & " elapsed_ms=" & elapsed
    End If

    g_lastRunByMachinePeriod(machineId & "|" & periodKey) = Now()
End Sub

Public Sub MainStatusAggregation(ByVal machineId)
    Dim cn, runtimeCatalog
    Dim targetPeriod, lookback

    InitGlobals

    If Not IsMaster() Then Exit Sub

    If Not G_MACHINES.Exists(machineId) Then
        LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", "unknown_machine=" & machineId
        Exit Sub
    End If

    targetPeriod = ""
    lookback = 0

    If ShouldRun(machineId, "1D", THROTTLE_1D_MINUTES) Then
        targetPeriod = "1D"
        lookback = 1440
    ElseIf ShouldRun(machineId, "7D", THROTTLE_7D_MINUTES) Then
        targetPeriod = "7D"
        lookback = 10080
    ElseIf ShouldRun(machineId, "30D", THROTTLE_30D_MINUTES) Then
        targetPeriod = "30D"
        lookback = 43200
    Else
        LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "skip machine=" & machineId & " reason=throttled"
        Exit Sub
    End If

    Set cn = OpenConnection()
    If cn Is Nothing Then
        LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", "skip machine=" & machineId & " reason=no_sql_connection"
        Exit Sub
    End If

    runtimeCatalog = GetRuntimeCatalog()
    LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "process machine=" & machineId & " period=" & targetPeriod

    AggregatePeriod cn, runtimeCatalog, machineId, targetPeriod, lookback

    CloseConn cn
End Sub
