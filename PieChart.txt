Private Function S(ByVal v)
    On Error Resume Next
    If IsNull(v) Or IsEmpty(v) Or IsObject(v) Then
        S = ""
    Else
        S = CStr(v)
        If Err.Number <> 0 Then
            S = ""
            Err.Clear
        End If
    End If
    On Error GoTo 0
End Function

Private Function IsMaster()
    On Error Resume Next
    IsMaster = (CLng(HMIRuntime.Tags("@RM_MASTER").Read) = 1)
    If Err.Number <> 0 Then
        IsMaster = True
        Err.Clear
    End If
    On Error GoTo 0
End Function

Private Function OpenConnection()
    On Error Resume Next
    Dim cn, prov

    prov = DEFAULT_PROVIDER
    Set cn = CreateObject("ADODB.Connection")
    cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "OpenConnection", "provider_fallback=" & FALLBACK_PROVIDER & " reason=" & S(Err.Description)
        Err.Clear
        prov = FALLBACK_PROVIDER
        Set cn = CreateObject("ADODB.Connection")
        cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"
    End If

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_ERROR, "OpenConnection", "sql_open_failed=" & S(Err.Description)
        Set cn = Nothing
        Err.Clear
    End If

    Set OpenConnection = cn
    On Error GoTo 0
End Function

Private Sub CloseConn(ByVal cn)
    On Error Resume Next
    If IsObject(cn) Then
        If Not (cn Is Nothing) Then
            If cn.State = adStateOpen Then cn.Close
        End If
    End If
    Set cn = Nothing
    On Error GoTo 0
End Sub

Private Function GetRuntimeCatalog()
    Dim cat
    On Error Resume Next
    cat = HMIRuntime.Tags("@DatasourceNameRT").Read
    If Err.Number <> 0 Or IsNull(cat) Or IsEmpty(cat) Or Len(CStr(cat)) = 0 Then
        Err.Clear
        GetRuntimeCatalog = "WinCC"
    Else
        GetRuntimeCatalog = CStr(cat)
    End If
    On Error GoTo 0
End Function

Private Function TS3(ByVal dt)
    TS3 = Year(dt) & "-" & Right("0" & Month(dt),2) & "-" & Right("0" & Day(dt),2) & _
          " " & Right("0" & Hour(dt),2) & ":" & Right("0" & Minute(dt),2) & ":" & Right("0" & Second(dt),2) & ".000"
End Function

Private Function TSDate(ByVal dt)
    TSDate = Year(dt) & "-" & Right("0" & Month(dt),2) & "-" & Right("0" & Day(dt),2)
End Function

Private Function SqlQ(ByVal v)
    SqlQ = "'" & Replace(CStr(v), "'", "''") & "'"
End Function

Private Function BracketIdent(ByVal s)
    BracketIdent = "[" & Replace(CStr(s), "]", "]]") & "]"
End Function

Private Function ElapsedMs(ByVal timerStart)
    Dim timerEnd
    timerEnd = Timer
    If timerEnd < timerStart Then timerEnd = timerEnd + 86400
    ElapsedMs = CLng((timerEnd - timerStart) * 1000)
End Function

Private Sub LogMsg(ByVal level, ByVal src, ByVal msg)
    If level > CURRENT_LOG_LEVEL Then Exit Sub
    On Error Resume Next
    HMIRuntime.Trace FormatDateTime(Now(), vbShortDate) & " " & _
                     FormatDateTime(Now(), vbLongTime) & " [PieChart/" & src & "] " & S(msg) & vbCrLf
    On Error GoTo 0
End Sub

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "HMITag", "write_failed tag=" & tagName & " err=" & S(Err.Description)

End Sub

Private Function ShouldRun(ByVal machineId, ByVal periodKey, ByVal throttleMinutes)
    Dim k, minsSince
    k = BuildMachinePeriodKey(machineId, periodKey)

    If Not g_lastRunByMachinePeriod.Exists(k) Then
        ShouldRun = True
        Exit Function
    End If

    minsSince = DateDiff("n", CDate(g_lastRunByMachinePeriod(k)), Now())
    ShouldRun = (minsSince >= throttleMinutes)
End Function

Private Function GetCommandTimeoutSeconds(ByVal periodKey)
    Select Case UCase(CStr(periodKey))
        Case "1D"
            GetCommandTimeoutSeconds = SQL_TIMEOUT_SECONDS_1D
        Case "7D"
            GetCommandTimeoutSeconds = SQL_TIMEOUT_SECONDS_7D
        Case "30D"
            GetCommandTimeoutSeconds = SQL_TIMEOUT_SECONDS_30D
        Case Else
            GetCommandTimeoutSeconds = SQL_TIMEOUT_SECONDS_1D
    End Select
End Function

Private Function BuildMachinePeriodKey(ByVal machineId, ByVal periodKey)
    BuildMachinePeriodKey = machineId & "|" & periodKey
End Function

Private Function GetFailureCooldownMinutes(ByVal periodKey)
    Select Case UCase(CStr(periodKey))
        Case "7D"
            GetFailureCooldownMinutes = FAILURE_COOLDOWN_7D_MINUTES
        Case "30D"
            GetFailureCooldownMinutes = FAILURE_COOLDOWN_30D_MINUTES
        Case Else
            GetFailureCooldownMinutes = 0
    End Select
End Function

Private Function BuildCountTagFullName(ByVal machineId, ByVal statusName)
    Dim basePrefix, baseCore
    basePrefix = G_MACHINES(machineId)
    baseCore = COUNT_TAG_CORE_TEMPLATE
    baseCore = Replace(baseCore, "{MACHINE_ID}", machineId)
    baseCore = Replace(baseCore, "{STATUS_NAME}", statusName)
    BuildCountTagFullName = basePrefix & COUNT_TAG_PREFIX_JOINER & baseCore & COUNT_TAG_SUFFIX
End Function

Private Sub ReadSingleFraction(ByVal cn, ByVal runtimeCatalog, ByVal tagFullName, ByVal tFrom, ByVal tTo, ByVal machineId, ByVal periodKey, ByVal statusName, ByRef fractionOut, ByRef resultState, ByRef errText)
    Dim cmd, rs, sqlText, safeTag, t0, ms
    Dim timeoutSeconds
    Dim v

    fractionOut = 0
    resultState = "error"
    errText = ""
    safeTag = Replace(tagFullName, "'", "''")
    sqlText = "cp_TagStatisticEx '" & runtimeCatalog & "','TAG:R,''" & safeTag & "'',''" & TS3(tFrom) & "'',''" & TS3(tTo) & "''','" & FIXED_AGG_EXPR & "'"
    timeoutSeconds = GetCommandTimeoutSeconds(periodKey)

    t0 = Timer
    On Error Resume Next
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = cn
    cmd.CommandType = adCmdText
    cmd.CommandTimeout = timeoutSeconds
    cmd.CommandText = sqlText
    Set rs = cmd.Execute
    ms = ElapsedMs(t0)

    If Err.Number <> 0 Then
        resultState = "error"
        errText = S(Err.Description)
        Err.Clear
    ElseIf (rs Is Nothing) Then
        resultState = "empty"
    ElseIf rs.EOF Then
        resultState = "empty"
    Else
        v = rs.Fields(0).Value
        If IsNull(v) Or IsEmpty(v) Then
            resultState = "empty"
        ElseIf Len(Trim(S(v))) = 0 Then
            resultState = "empty"
        ElseIf Not IsNumeric(v) Then
            resultState = "non_numeric"
        ElseIf CDbl(v) < 0 Or CDbl(v) > 1 Then
            resultState = "out_of_range"
        Else
            fractionOut = CDbl(v)
            resultState = "valid"
        End If
    End If

    If Not (rs Is Nothing) Then
        rs.Close
    End If

    If resultState = "error" Then
        LogMsg LOG_LEVEL_INFO, "cp_TagStatisticEx", "machine=" & machineId & " period=" & periodKey & " status=" & statusName & " elapsed_ms=" & ms & " timeout_s=" & timeoutSeconds & " result=" & resultState & " err=" & errText
    Else
        LogMsg LOG_LEVEL_INFO, "cp_TagStatisticEx", "machine=" & machineId & " period=" & periodKey & " status=" & statusName & " elapsed_ms=" & ms & " timeout_s=" & timeoutSeconds & " result=" & resultState
    End If
    On Error GoTo 0

    Set rs = Nothing
    Set cmd = Nothing
End Sub

        Exit Sub
    End If

    ReadSingleFraction cn, runtimeCatalog, BuildCountTagFullName(machineId, "Running"), tFrom, tTo, machineId, periodKey, "Running", r, stateR, errR

        inputValid = False

            failureKind = "invalid_input"

        failedStatus = "Running"

        fHalt = 0
        fAbort = 0

        Exit Sub
    End If

    ReadSingleFraction cn, runtimeCatalog, BuildCountTagFullName(machineId, "Waiting"), tFrom, tTo, machineId, periodKey, "Waiting", w, stateW, errW

        inputValid = False

Private Sub EnsureDailyRollupTable(ByVal cn, ByVal runtimeCatalog)
    Dim cmd, sqlText, dbName
    dbName = BracketIdent(runtimeCatalog)
    sqlText = "IF OBJECT_ID(N'" & runtimeCatalog & ".dbo.PieChartDailyRollup', N'U') IS NULL " & _
              "BEGIN " & _
              "CREATE TABLE " & dbName & ".dbo.PieChartDailyRollup(" & _
              "machine_id NVARCHAR(32) NOT NULL," & _
              "summary_date DATE NOT NULL," & _
              "aborted_frac FLOAT NOT NULL," & _
              "halted_frac FLOAT NOT NULL," & _
              "running_frac FLOAT NOT NULL," & _
              "waiting_frac FLOAT NOT NULL," & _
              "interrupted_frac FLOAT NOT NULL," & _
              "created_at DATETIME NOT NULL," & _
              "updated_at DATETIME NOT NULL," & _
              "CONSTRAINT PK_PieChartDailyRollup PRIMARY KEY CLUSTERED(machine_id, summary_date)" & _
              "); END"

    On Error Resume Next
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = cn
    cmd.CommandType = adCmdText
    cmd.CommandTimeout = SQL_TIMEOUT_SECONDS_1D
    cmd.CommandText = sqlText
    cmd.Execute
    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "Rollup", "table_ensure_failed err=" & S(Err.Description)
        Err.Clear
    Else
        LogMsg LOG_LEVEL_INFO, "Rollup", "table_ensured catalog=" & runtimeCatalog & " table=dbo.PieChartDailyRollup"
    End If
    On Error GoTo 0
    Set cmd = Nothing
End Sub

Private Function DailyRollupRowExists(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal dayDate)
    Dim cmd, rs, sqlText, dbName
    dbName = BracketIdent(runtimeCatalog)
    sqlText = "SELECT TOP 1 1 FROM " & dbName & ".dbo.PieChartDailyRollup WHERE machine_id=" & SqlQ(machineId) & " AND summary_date=CONVERT(date," & SqlQ(TSDate(dayDate)) & ")"
    DailyRollupRowExists = False

    On Error Resume Next
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = cn
    cmd.CommandType = adCmdText
    cmd.CommandTimeout = SQL_TIMEOUT_SECONDS_1D
    cmd.CommandText = sqlText
    Set rs = cmd.Execute
    If Err.Number = 0 Then
        If Not (rs Is Nothing) Then
            If Not rs.EOF Then DailyRollupRowExists = True
        End If
    Else
        Err.Clear
    End If
    On Error GoTo 0

    If Not (rs Is Nothing) Then rs.Close
    Set rs = Nothing
    Set cmd = Nothing
End Function

Private Sub UpsertDailyRollupForDate(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal dayDate)
    Dim dFrom, dTo
    Dim a, h, r, w, i
    Dim sA, sH, sR, sW, sI
    Dim eA, eH, eR, eW, eI
    Dim cmd, sqlText, dbName

    dFrom = DateSerial(Year(dayDate), Month(dayDate), Day(dayDate))
    dTo = DateAdd("d", 1, dFrom)
    LogMsg LOG_LEVEL_INFO, "Rollup", "backfill_day_started machine=" & machineId & " day=" & TSDate(dFrom)

    ReadSingleFraction cn, runtimeCatalog, BuildCountTagFullName(machineId, "Aborted"), dFrom, dTo, machineId, "1D", "Aborted", a, sA, eA
    ReadSingleFraction cn, runtimeCatalog, BuildCountTagFullName(machineId, "Halted"), dFrom, dTo, machineId, "1D", "Halted", h, sH, eH
    ReadSingleFraction cn, runtimeCatalog, BuildCountTagFullName(machineId, "Running"), dFrom, dTo, machineId, "1D", "Running", r, sR, eR
    ReadSingleFraction cn, runtimeCatalog, BuildCountTagFullName(machineId, "Waiting"), dFrom, dTo, machineId, "1D", "Waiting", w, sW, eW
    ReadSingleFraction cn, runtimeCatalog, BuildCountTagFullName(machineId, "Interrupted"), dFrom, dTo, machineId, "1D", "Interrupted", i, sI, eI

    If sA <> "valid" Or sH <> "valid" Or sR <> "valid" Or sW <> "valid" Or sI <> "valid" Then
        LogMsg LOG_LEVEL_WARN, "Rollup", "backfill_day_failure machine=" & machineId & " day=" & TSDate(dFrom) & " states=Aborted:" & sA & ",Halted:" & sH & ",Running:" & sR & ",Waiting:" & sW & ",Interrupted:" & sI
        Exit Sub
    End If

    dbName = BracketIdent(runtimeCatalog)
    sqlText = "MERGE " & dbName & ".dbo.PieChartDailyRollup AS tgt " & _
              "USING (SELECT " & SqlQ(machineId) & " AS machine_id, CONVERT(date," & SqlQ(TSDate(dFrom)) & ") AS summary_date) AS src " & _
              "ON tgt.machine_id=src.machine_id AND tgt.summary_date=src.summary_date " & _
              "WHEN MATCHED THEN UPDATE SET aborted_frac=" & Replace(CStr(CDbl(a)), ",", ".") & ", halted_frac=" & Replace(CStr(CDbl(h)), ",", ".") & ", running_frac=" & Replace(CStr(CDbl(r)), ",", ".") & ", waiting_frac=" & Replace(CStr(CDbl(w)), ",", ".") & ", interrupted_frac=" & Replace(CStr(CDbl(i)), ",", ".") & ", updated_at=GETDATE() " & _
              "WHEN NOT MATCHED THEN INSERT(machine_id, summary_date, aborted_frac, halted_frac, running_frac, waiting_frac, interrupted_frac, created_at, updated_at) VALUES(" & SqlQ(machineId) & ", CONVERT(date," & SqlQ(TSDate(dFrom)) & ")," & Replace(CStr(CDbl(a)), ",", ".") & "," & Replace(CStr(CDbl(h)), ",", ".") & "," & Replace(CStr(CDbl(r)), ",", ".") & "," & Replace(CStr(CDbl(w)), ",", ".") & "," & Replace(CStr(CDbl(i)), ",", ".") & ",GETDATE(),GETDATE());"

    On Error Resume Next
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = cn
    cmd.CommandType = adCmdText
    cmd.CommandTimeout = SQL_TIMEOUT_SECONDS_1D
    cmd.CommandText = sqlText
    cmd.Execute
    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "Rollup", "backfill_day_failure machine=" & machineId & " day=" & TSDate(dFrom) & " err=" & S(Err.Description)
        Err.Clear
    Else
        LogMsg LOG_LEVEL_INFO, "Rollup", "backfill_day_success machine=" & machineId & " day=" & TSDate(dFrom)
    End If
    On Error GoTo 0
    Set cmd = Nothing
End Sub

Private Sub EnsureRecentDailyRollups(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal daysNeeded)
    Dim startDay, endDay, d
    Dim missingTotal, backfilled
    startDay = DateAdd("d", -CLng(daysNeeded), Date())
    endDay = DateAdd("d", -1, Date())
    missingTotal = 0
    backfilled = 0

    d = startDay
    Do While d <= endDay
        If Not DailyRollupRowExists(cn, runtimeCatalog, machineId, d) Then
            missingTotal = missingTotal + 1
            If backfilled < MAX_ROLLUP_BACKFILL_DAYS_PER_INVOCATION Then
                UpsertDailyRollupForDate cn, runtimeCatalog, machineId, d
                backfilled = backfilled + 1
            End If
        End If
        d = DateAdd("d", 1, d)
    Loop

    If missingTotal > backfilled Then
        LogMsg LOG_LEVEL_WARN, "Rollup", "partial_backfill machine=" & machineId & " days_needed=" & daysNeeded & " missing_total=" & missingTotal & " backfilled_now=" & backfilled & " cap=" & MAX_ROLLUP_BACKFILL_DAYS_PER_INVOCATION
    End If
End Sub

Private Sub RunRollupMaintenance(ByVal cn, ByVal runtimeCatalog, ByVal machineId)
    Dim startDay, endDay, d
    Dim missingTotal, backfilled

    LogMsg LOG_LEVEL_INFO, "Rollup", "rollup_backfill_started machine=" & machineId

    On Error Resume Next
    EnsureDailyRollupTable cn, runtimeCatalog

    startDay = DateAdd("d", -30, Date())
    endDay = DateAdd("d", -1, Date())
    missingTotal = 0
    backfilled = 0

    d = startDay
    Do While d <= endDay
        If Not DailyRollupRowExists(cn, runtimeCatalog, machineId, d) Then
            missingTotal = missingTotal + 1
            If backfilled < MAX_ROLLUP_BACKFILL_DAYS_PER_INVOCATION Then
                UpsertDailyRollupForDate cn, runtimeCatalog, machineId, d
                backfilled = backfilled + 1
            End If
        End If
        d = DateAdd("d", 1, d)
    Loop

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "Rollup", "rollup_backfill_failure machine=" & machineId & " err=" & S(Err.Description)
        Err.Clear
    Else
        If missingTotal > backfilled Then
            LogMsg LOG_LEVEL_WARN, "Rollup", "rollup_backfill_limited machine=" & machineId & " missing=" & missingTotal & " filled_now=" & backfilled & " cap=" & MAX_ROLLUP_BACKFILL_DAYS_PER_INVOCATION
        End If
        LogMsg LOG_LEVEL_INFO, "Rollup", "rollup_backfill_success machine=" & machineId & " missing=" & missingTotal & " filled_now=" & backfilled
    End If
    On Error GoTo 0
End Sub

Private Sub ReadWindowFractionsFromDailyRollup(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal daysCount, ByRef fRun, ByRef fHalt, ByRef fAbort, ByRef inputValid, ByRef failureKind, ByRef failureDetail)
    Dim startDay, endExclusive
    Dim cmd, rs, sqlText, dbName
    Dim rowCount, sumAbort, sumHalt, sumRun, sumWait, sumInt
    Dim minRows

    inputValid = False
    failureKind = ""
    failureDetail = ""
    fRun = 0
    fHalt = 0
    fAbort = 0
    rowCount = 0
    sumAbort = 0
    sumHalt = 0
    sumRun = 0
    sumWait = 0
    sumInt = 0
    startDay = DateAdd("d", -CLng(daysCount), Date())
    endExclusive = Date()
    dbName = BracketIdent(runtimeCatalog)
    sqlText = "SELECT summary_date, aborted_frac, halted_frac, running_frac, waiting_frac, interrupted_frac FROM " & dbName & ".dbo.PieChartDailyRollup WHERE machine_id=" & SqlQ(machineId) & " AND summary_date>=CONVERT(date," & SqlQ(TSDate(startDay)) & ") AND summary_date<CONVERT(date," & SqlQ(TSDate(endExclusive)) & ") ORDER BY summary_date ASC"

    If CLng(daysCount) = 7 Then
        minRows = 5
    ElseIf CLng(daysCount) = 30 Then
        minRows = 21
    Else
        minRows = CLng(daysCount)
    End If

    On Error Resume Next
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = cn
    cmd.CommandType = adCmdText
    cmd.CommandTimeout = SQL_TIMEOUT_SECONDS_1D
    cmd.CommandText = sqlText
    Set rs = cmd.Execute
    If Err.Number <> 0 Then
        failureKind = "rollup_missing"
        failureDetail = "insufficient_daily_rows"
        LogMsg LOG_LEVEL_WARN, "Rollup", "window_read_failure machine=" & machineId & " days=" & daysCount & " err=" & S(Err.Description)
        Err.Clear
    Else
        Do While Not rs.EOF
            rowCount = rowCount + 1
            sumAbort = sumAbort + CDbl(rs.Fields(1).Value)
            sumHalt = sumHalt + CDbl(rs.Fields(2).Value)
            sumRun = sumRun + CDbl(rs.Fields(3).Value)
            sumWait = sumWait + CDbl(rs.Fields(4).Value)
            sumInt = sumInt + CDbl(rs.Fields(5).Value)
            rs.MoveNext
        Loop
        If rowCount < CLng(minRows) Then
            failureKind = "rollup_missing"
            failureDetail = "insufficient_daily_rows"
            LogMsg LOG_LEVEL_WARN, "Rollup", "insufficient_rollup_rows machine=" & machineId & " period=" & daysCount & "D rows_found=" & rowCount & " rows_expected=" & daysCount & " rows_minimum=" & minRows
        Else
            fAbort = CDbl(sumAbort) + CDbl(sumWait)
            fHalt = CDbl(sumHalt) + CDbl(sumInt)
            fRun = CDbl(sumRun)
            inputValid = True
            If rowCount < CLng(daysCount) Then
                LogMsg LOG_LEVEL_WARN, "Rollup", "partial_rollup_window machine=" & machineId & " period=" & daysCount & "D rows_found=" & rowCount & " rows_expected=" & daysCount & " rows_minimum=" & minRows
            Else
                LogMsg LOG_LEVEL_INFO, "Rollup", "window_read_success machine=" & machineId & " days=" & daysCount & " rows=" & rowCount
            End If
        End If
    End If
    On Error GoTo 0

    If Not (rs Is Nothing) Then rs.Close
    Set rs = Nothing
    Set cmd = Nothing
End Sub

Private Function GetNextPeriodIndex(ByVal machineId)
    If Not g_nextPeriodIndexByMachine.Exists(machineId) Then
        g_nextPeriodIndexByMachine(machineId) = 0
    End If
    GetNextPeriodIndex = CLng(g_nextPeriodIndexByMachine(machineId))
End Function

Private Sub SetNextPeriodIndex(ByVal machineId, ByVal nextIndex)
    g_nextPeriodIndexByMachine(machineId) = CLng(nextIndex Mod 3)
End Sub

Private Function PeriodByIndex(ByVal idx)
    Dim safeIdx
    safeIdx = (CLng(idx) Mod 3 + 3) Mod 3
    Select Case safeIdx
        Case 0
            PeriodByIndex = "1D"
        Case 1
            PeriodByIndex = "7D"
        Case Else
            PeriodByIndex = "30D"
    End Select
End Function

Private Function LookbackMinutesForPeriod(ByVal periodKey)
    Select Case UCase(CStr(periodKey))
        Case "1D"
            LookbackMinutesForPeriod = 1440
        Case "7D"
            LookbackMinutesForPeriod = 10080
        Case "30D"
            LookbackMinutesForPeriod = 43200
        Case Else
            LookbackMinutesForPeriod = 1440
    End Select
End Function

Private Function ThrottleMinutesForPeriod(ByVal periodKey)
    Select Case UCase(CStr(periodKey))
        Case "1D"
            ThrottleMinutesForPeriod = THROTTLE_1D_MINUTES
        Case "7D"
            ThrottleMinutesForPeriod = THROTTLE_7D_MINUTES
        Case "30D"
            ThrottleMinutesForPeriod = THROTTLE_30D_MINUTES
        Case Else
            ThrottleMinutesForPeriod = THROTTLE_1D_MINUTES
    End Select
End Function

Private Sub NormalizeFractionsToPct(ByVal fRun, ByVal fHalt, ByVal fAbort, ByRef pRun, ByRef pHalt, ByRef pAbort)
    Dim tot
    tot = CDbl(fRun) + CDbl(fHalt) + CDbl(fAbort)

    If tot <= 0 Then
        pRun = 0
        pHalt = 0
        pAbort = 0
    Else
        pRun = Round((CDbl(fRun) / tot) * 100, 2)
        pHalt = Round((CDbl(fHalt) / tot) * 100, 2)
        pAbort = Round((CDbl(fAbort) / tot) * 100, 2)
    End If
End Sub

' *_Percentage_PieChart is a WinCC rendering shim (minimum 1) and must not be used as the real percentage value.
Private Function PieVal(ByVal pct)
    If pct < 1 Then
        PieVal = 1
    Else
        PieVal = pct
    End If
End Function

Private Sub AggregatePeriod(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal lookbackMinutes)
    Dim tFrom, tTo, t0, elapsed
    Dim fRun, fHalt, fAbort, inputValid
    Dim pRun, pHalt, pAbort
    Dim base, totalCombined
    Dim failureCooldownMinutes, failureUntil, failureKey
    Dim failureKind, failedStatus, failureDetail
    Dim resultLabel, stampNormalThrottle

    t0 = Timer
    tTo = Now()
    tFrom = DateAdd("n", -lookbackMinutes, tTo)

    LogMsg LOG_LEVEL_INFO, "Aggregate", "start machine=" & machineId & " period=" & periodKey & " from=" & TS3(tFrom) & " to=" & TS3(tTo)

    If UCase(CStr(periodKey)) = "1D" Then
        GetCombinedFractions cn, runtimeCatalog, machineId, tFrom, tTo, periodKey, fRun, fHalt, fAbort, inputValid, failureKind, failedStatus, failureDetail
    ElseIf UCase(CStr(periodKey)) = "7D" Then
        ReadWindowFractionsFromDailyRollup cn, runtimeCatalog, machineId, 7, fRun, fHalt, fAbort, inputValid, failureKind, failureDetail
        failedStatus = ""
    ElseIf UCase(CStr(periodKey)) = "30D" Then
        ReadWindowFractionsFromDailyRollup cn, runtimeCatalog, machineId, 30, fRun, fHalt, fAbort, inputValid, failureKind, failureDetail
        failedStatus = ""
    Else
        inputValid = False
        failureKind = "invalid_input"
        failedStatus = ""
        failureDetail = "unknown_period"
    End If

    If inputValid Then
        NormalizeFractionsToPct fRun, fHalt, fAbort, pRun, pHalt, pAbort
        totalCombined = CDbl(fRun) + CDbl(fHalt) + CDbl(fAbort)
        If totalCombined > 0 Then
            resultLabel = "valid_nonzero"
        Else
            resultLabel = "valid_zero_total"
        End If
    Else
        fRun = 0
        fHalt = 0
        fAbort = 0
        pRun = 0
        pHalt = 0
        pAbort = 0
        totalCombined = 0
        resultLabel = "zero_no_data"
    End If

    base = machineId & "_" & periodKey & "_Machine_"

    WriteNumericOutputTag base & "Running_Percentage", pRun
    WriteNumericOutputTag base & "Halted_Percentage", pHalt
    WriteNumericOutputTag base & "Aborted_Percentage", pAbort

    WriteNumericOutputTag base & "Running_Percentage_PieChart", PieVal(pRun)
    WriteNumericOutputTag base & "Halted_Percentage_PieChart", PieVal(pHalt)
    WriteNumericOutputTag base & "Aborted_Percentage_PieChart", PieVal(pAbort)

    elapsed = ElapsedMs(t0)
    If inputValid Then
        If resultLabel = "valid_nonzero" Then
            LogMsg LOG_LEVEL_INFO, "Aggregate", "end machine=" & machineId & " period=" & periodKey & " result=valid_nonzero run=" & pRun & " halt=" & pHalt & " abort=" & pAbort & " elapsed_ms=" & elapsed
        Else
            LogMsg LOG_LEVEL_WARN, "Aggregate", "end machine=" & machineId & " period=" & periodKey & " result=valid_zero_total run=0 halt=0 abort=0 elapsed_ms=" & elapsed
        End If
    Else
        If failureKind = "query_error" Then
            failureCooldownMinutes = GetFailureCooldownMinutes(periodKey)
        Else
            failureCooldownMinutes = 0
        End If
        If failureCooldownMinutes > 0 Then
            failureUntil = DateAdd("n", failureCooldownMinutes, Now())
            failureKey = BuildMachinePeriodKey(machineId, periodKey)
            g_periodFailureCooldownUntil(failureKey) = failureUntil
            LogMsg LOG_LEVEL_WARN, "Aggregate", "set_failure_cooldown machine=" & machineId & " period=" & periodKey & " cooldown_until=" & TS3(failureUntil)
        End If
        LogMsg LOG_LEVEL_WARN, "Aggregate", "end machine=" & machineId & " period=" & periodKey & " result=zero_no_data run=0 halt=0 abort=0 failure_kind=" & failureKind & " failed_status=" & failedStatus & " failure_detail=" & failureDetail & " elapsed_ms=" & elapsed
    End If

    If elapsed > WINDOW_EXEC_WARN_THRESHOLD_MS Then
        LogMsg LOG_LEVEL_WARN, "Aggregate", "slow_period machine=" & machineId & " period=" & periodKey & " elapsed_ms=" & elapsed
    End If

    stampNormalThrottle = False
    If resultLabel = "valid_nonzero" Or resultLabel = "valid_zero_total" Then
        stampNormalThrottle = True
    ElseIf resultLabel = "zero_no_data" And failureKind <> "query_error" Then
        stampNormalThrottle = True
    End If

    If stampNormalThrottle Then
        g_lastRunByMachinePeriod(BuildMachinePeriodKey(machineId, periodKey)) = Now()
    End If
End Sub

Public Sub MainStatusAggregation(ByVal machineId)
    Dim cn, runtimeCatalog
    Dim targetPeriod, lookback
    Dim secondsSinceLastGlobal
    Dim periodFailureKey, periodFailureUntil
    Dim startIdx, probeOffset, periodCandidate, periodDue, periodCooldownBlocked
    Dim selectedReason, anyDue

    InitGlobals

    If Not IsMaster() Then Exit Sub

    If Not G_MACHINES.Exists(machineId) Then
        LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", "unknown_machine=" & machineId
        Exit Sub
    End If

    targetPeriod = ""
    lookback = 0
    selectedReason = ""
    anyDue = False
    startIdx = GetNextPeriodIndex(machineId)

    For probeOffset = 0 To 2
        periodCandidate = PeriodByIndex(startIdx + probeOffset)
        periodDue = ShouldRun(machineId, periodCandidate, ThrottleMinutesForPeriod(periodCandidate))
        If Not periodDue Then
            LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "scheduler_skip machine=" & machineId & " period=" & periodCandidate & " reason=not_due"
        Else
            anyDue = True
            periodCooldownBlocked = False
            If periodCandidate <> "1D" Then
                periodFailureKey = BuildMachinePeriodKey(machineId, periodCandidate)
                If g_periodFailureCooldownUntil.Exists(periodFailureKey) Then
                    periodFailureUntil = CDate(g_periodFailureCooldownUntil(periodFailureKey))
                    If Now() < periodFailureUntil Then
                        periodCooldownBlocked = True
                        LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "scheduler_skip machine=" & machineId & " period=" & periodCandidate & " reason=cooldown"
                    Else
                        g_periodFailureCooldownUntil.Remove periodFailureKey
                    End If
                End If
            End If

            If Not periodCooldownBlocked Then
                targetPeriod = periodCandidate
                lookback = LookbackMinutesForPeriod(periodCandidate)
                selectedReason = "due_round_robin"
                Exit For
            End If
        End If
    Next

    If targetPeriod = "" Then
        If anyDue Then
            LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "skip machine=" & machineId & " reason=all_due_periods_in_cooldown"
        Else
            LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "skip machine=" & machineId & " reason=throttled"
        End If
        Exit Sub
    End If

    If Not IsEmpty(g_lastGlobalAggregationRun) Then
        secondsSinceLastGlobal = DateDiff("s", CDate(g_lastGlobalAggregationRun), Now())
        If secondsSinceLastGlobal < GLOBAL_MIN_SECONDS_BETWEEN_MACHINE_RUNS Then
            LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "skip machine=" & machineId & " reason=global_cooldown cooldown_seconds=" & GLOBAL_MIN_SECONDS_BETWEEN_MACHINE_RUNS & " seconds_since_last_run=" & secondsSinceLastGlobal
            Exit Sub
        End If
    End If

    Set cn = OpenConnection()
    If cn Is Nothing Then
        LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", "skip machine=" & machineId & " reason=no_sql_connection"
        Exit Sub
    End If

    runtimeCatalog = GetRuntimeCatalog()
    RunRollupMaintenance cn, runtimeCatalog, machineId
    LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "scheduler_selected machine=" & machineId & " period=" & targetPeriod & " reason=" & selectedReason
    LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", "process machine=" & machineId & " period=" & targetPeriod & " version=" & SCRIPT_VERSION

    g_lastGlobalAggregationRun = Now()
    AggregatePeriod cn, runtimeCatalog, machineId, targetPeriod, lookback
    SetNextPeriodIndex machineId, (startIdx + 1)

    CloseConn cn
End Sub
