' ======================================================================
'  WinCC Global Script: Status % Aggregation (FINAL)
'
'  - Uses ONLY *_Count tags (binary 0/1 archived every 500ms)
'  - Reads 5 base statuses:
'       Aborted, Halted, Running, Waiting, Interrupted
'  - Combines into 3 outputs (names UNCHANGED):
'       Aborted  = Aborted + Waiting
'       Halted   = Halted  + Interrupted
'       Running  = Running
'
'  OUTPUT TAGS (unchanged):
'    <Machine>_<Period>_Machine_<Status>_Percentage              (can be 0)
'    <Machine>_<Period>_Machine_<Status>_Percentage_PieChart     (min 1)
'
'  Periods: 1D / 7D / 30D
' ======================================================================

Option Explicit

' ─────────────────────────────────────────────────────────────────────
' Configuration
' ─────────────────────────────────────────────────────────────────────

Const FALLBACK_PROVIDER      = "SQLNCLI11"
Const SQL_INSTANCE           = ".\WinCC"
Const INITIAL_CATALOG        = "master"
Const SQL_TIMEOUT_SECONDS    = 45
Const MAX_RETRY_ATTEMPTS     = 2
Const RETRY_BACKOFF_MS       = 250
Const RETRY_BACKOFF_TIMEOUT_MS = 1250
Const WINDOW_EXEC_WARN_THRESHOLD_MS = 15000

' Project-specific batch aggregation configuration (ordered by preference).
' Use "|" to separate expressions. Keep first expression as known-good for
' your archive schema and list fallbacks after it.
Const PROJECT_BATCH_AGG_LIST_DEFAULT = "AVG(RealValue)|AVG(Value)|AVG(ValueScaled)"
Const PROJECT_BATCH_AGG_LIST_WINCC   = "AVG(Value)|AVG(RealValue)|AVG(ValueScaled)"

Const THROTTLE_1D_MINUTES    = 30    ' every 30 minutes
Const THROTTLE_7D_MINUTES    = 360   ' every 6 hours
Const THROTTLE_30D_MINUTES   = 1440  ' every 24 hours

' Source tag full-name pattern pieces (configurable)
Const COUNT_TAG_PREFIX_JOINER          = "\"
Const COUNT_TAG_CORE_TEMPLATE          = "{MACHINE_ID}_ProgramStatus_{STATUS_NAME}"
Const COUNT_TAG_SUFFIX                 = "_Count"
Const STARTUP_VALIDATE_LOOKBACK_1D_MINUTES  = 1440
Const STARTUP_VALIDATE_LOOKBACK_7D_MINUTES  = 10080
Const STARTUP_VALIDATE_LOOKBACK_30D_MINUTES = 43200

Const VALIDATE_STATE_HARD_FAIL = 0
Const VALIDATE_STATE_NO_RECENT = 1

Private Function IsMaster()
    On Error Resume Next
    IsMaster = (CLng(HMIRuntime.Tags("@RM_MASTER").Read) = 1)
    If Err.Number <> 0 Then
        ' Fail-open for standalone / tag not present
        IsMaster = True
        Err.Clear
    End If
    On Error GoTo 0
End Function

' ─────────────────────────────────────────────────────────────────────
' Logging (Null-safe)
' ─────────────────────────────────────────────────────────────────────
Private Function S(ByVal v)
    On Error Resume Next
    If IsNull(v) Or IsEmpty(v) Or IsObject(v) Then
        S = ""
    Else
        S = CStr(v)
        If Err.Number <> 0 Then
            S = ""
            Err.Clear
        End If
    End If
    On Error GoTo 0
End Function

Private Sub LogMsg(ByVal level, ByVal src, ByVal msg)
    If level > CURRENT_LOG_LEVEL Then Exit Sub
    On Error Resume Next
    HMIRuntime.Trace _
        FormatDateTime(Now(), vbShortDate) & " " & _
        FormatDateTime(Now(), vbLongTime) & " [PieChart/" & src & "] " & S(msg) & vbCrLf
    On Error GoTo 0
End Sub

Private Sub SleepMs(ByVal ms)
    On Error Resume Next
    HMIRuntime.Sleep ms
    On Error GoTo 0
End Sub

' ─────────────────────────────────────────────────────────────────────
' SQL helpers
' ─────────────────────────────────────────────────────────────────────
Private Function OpenConnection()
    On Error Resume Next
    Dim cn, prov
    prov = DEFAULT_PROVIDER

    Set cn = CreateObject("ADODB.Connection")
    cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "OpenConnection", "Switching provider to " & FALLBACK_PROVIDER & " (" & S(Err.Description) & ")"
        Err.Clear
        prov = FALLBACK_PROVIDER
        Set cn = CreateObject("ADODB.Connection")
        cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"
    End If

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_ERROR, "OpenConnection", "ERROR: " & S(Err.Description)
        Set cn = Nothing
        Err.Clear
    End If

    Set OpenConnection = cn
    On Error GoTo 0
End Function

Private Sub CloseConn(ByVal cn)
    On Error Resume Next
    If IsObject(cn) Then
        If Not (cn Is Nothing) Then
            If cn.State = adStateOpen Then cn.Close
        End If
    End If
    Set cn = Nothing
    On Error GoTo 0
End Sub

Private Function GetRuntimeCatalog()
    On Error Resume Next
    Dim cat
    cat = HMIRuntime.Tags("@DatasourceNameRT").Read
    If Err.Number <> 0 Or IsNull(cat) Or IsEmpty(cat) Then
        Err.Clear
        GetRuntimeCatalog = "WinCC"
    ElseIf Len(CStr(cat)) > 0 Then
        GetRuntimeCatalog = CStr(cat)
    Else
        GetRuntimeCatalog = "WinCC"
    End If
    On Error GoTo 0
End Function

Private Function TS3(ByVal dt)
    TS3 = Year(dt) & "-" & Right("0" & Month(dt),2) & "-" & Right("0" & Day(dt),2) & _
          " " & Right("0" & Hour(dt),2) & ":" & Right("0" & Minute(dt),2) & ":" & Right("0" & Second(dt),2) & ".000"
End Function

Private Function ElapsedMs(ByVal timerStart)
    Dim timerEnd
    timerEnd = Timer
    If timerEnd < timerStart Then timerEnd = timerEnd + 86400
    ElapsedMs = CLng((timerEnd - timerStart) * 1000)
End Function

Private Function IsTimeoutLikeError(ByVal errDesc)
    Dim lowered
    lowered = LCase(S(errDesc))
    IsTimeoutLikeError = (InStr(lowered, "timeout") > 0) Or _
                         (InStr(lowered, "time out") > 0) Or _
                         (InStr(lowered, "expired") > 0)
End Function

' ─────────────────────────────────────────────────────────────────────
' cp_TagStatisticEx (short retry, null-safe)
' ─────────────────────────────────────────────────────────────────────
Private Function ExecCpTagStatisticEx(ByVal cn, ByVal runtimeCatalog, ByVal tagFullName, ByVal tFrom, ByVal tTo, ByVal aggList, ByVal machineId, ByVal periodKey, ByVal statusName)
    Dim i, cmd, rs, sSql, safeTag, fieldValue
    Dim startTimer, startedAt, completedAt, elapsedMsValue
    Dim returnedRows, valueText, infoMsg, errDesc, backoffMs

    startedAt = Now
    startTimer = Timer
    returnedRows = 0
    valueText = ""

    safeTag = Replace(tagFullName, "'", "''")

    For i = 1 To MAX_RETRY_ATTEMPTS
        On Error Resume Next

        Set cmd = CreateObject("ADODB.Command")
        Set cmd.ActiveConnection = cn
        cmd.CommandTimeout = SQL_TIMEOUT_SECONDS
        cmd.CommandType = adCmdText

        sSql = "cp_TagStatisticEx '" & runtimeCatalog & "'," & _
               "'TAG:R,''" & safeTag & "'',''" & TS3(tFrom) & "'',''" & TS3(tTo) & "'''," & _
               "'" & aggList & "'"

        cmd.CommandText = sSql

        If CURRENT_LOG_LEVEL >= LOG_LEVEL_DEBUG Then
            LogMsg LOG_LEVEL_DEBUG, "cp_TagStatisticEx", "SQL=[" & sSql & "]"
        End If

        Set rs = cmd.Execute

        If Err.Number = 0 Then
            ExecCpTagStatisticEx = Empty

            If Not (rs Is Nothing) Then
                If Not rs.EOF Then
                    returnedRows = 1
                    If rs.Fields.Count >= 1 Then
                        fieldValue = rs.Fields(0).Value
                        If Not (IsNull(fieldValue) Or IsEmpty(fieldValue)) Then
                            ExecCpTagStatisticEx = fieldValue
                            valueText = CStr(fieldValue)
                        Else
                            valueText = "<null>"
                        End If
                    End If
                End If
                rs.Close
            End If

            Set rs = Nothing
            Set cmd = Nothing
            On Error GoTo 0

            completedAt = Now
            elapsedMsValue = ElapsedMs(startTimer)
            If valueText = "" Then valueText = "<empty>"
            infoMsg = "machine=" & machineId & _
                      " window=" & periodKey & _
                      " status=" & statusName & _
                      " sql_from=" & TS3(tFrom) & _
                      " sql_to=" & TS3(tTo) & _
                      " started=" & TS3(startedAt) & _
                      " ended=" & TS3(completedAt) & _
                      " elapsed_ms=" & elapsedMsValue & _
                      " success=true" & _
                      " rows=" & returnedRows & _
                      " value=" & valueText
            LogMsg LOG_LEVEL_INFO, "cp_TagStatisticEx", infoMsg
            Exit Function
        Else
            errDesc = S(Err.Description)
            LogMsg LOG_LEVEL_WARN, "cp_TagStatisticEx", "Attempt " & i & " failed: " & errDesc
            Err.Clear
            On Error GoTo 0
            If i < MAX_RETRY_ATTEMPTS Then
                backoffMs = RETRY_BACKOFF_MS
                If IsTimeoutLikeError(errDesc) Then backoffMs = RETRY_BACKOFF_TIMEOUT_MS
                SleepMs backoffMs
            End If
        End If
    Next

    ExecCpTagStatisticEx = Empty
    completedAt = Now
    elapsedMsValue = ElapsedMs(startTimer)
    infoMsg = "machine=" & machineId & _
              " window=" & periodKey & _
              " status=" & statusName & _
              " sql_from=" & TS3(tFrom) & _
              " sql_to=" & TS3(tTo) & _
              " started=" & TS3(startedAt) & _
              " ended=" & TS3(completedAt) & _
              " elapsed_ms=" & elapsedMsValue & _
              " success=false rows=0 value=<none>"
    LogMsg LOG_LEVEL_INFO, "cp_TagStatisticEx", infoMsg
End Function

Private Function CpAggregate(ByVal cn, ByVal runtimeCatalog, ByVal tagFullName, ByVal t1, ByVal t2, ByVal agg, ByVal machineId, ByVal periodKey, ByVal statusName)
    Dim res
    res = ExecCpTagStatisticEx(cn, runtimeCatalog, tagFullName, t1, t2, agg, machineId, periodKey, statusName)

    If IsNull(res) Or IsEmpty(res) Or (CStr(res) = "") Then
        CpAggregate = Empty
    ElseIf IsNumeric(res) Then
        CpAggregate = CDbl(res)
    Else
        CpAggregate = Empty
    End If
End Function

Private Function TryAvgFraction(ByVal cn, ByVal runtimeCatalog, ByVal tagFull, ByVal t1, ByVal t2, ByVal machineId, ByVal periodKey, ByVal statusName)
    Dim v

    v = CpAggregate(cn, runtimeCatalog, tagFull, t1, t2, "AVG(RealValue)", machineId, periodKey, statusName)
    If IsNumeric(v) Then If v >= 0 And v <= 1 Then TryAvgFraction = v : Exit Function

    v = CpAggregate(cn, runtimeCatalog, tagFull, t1, t2, "AVG(Value)", machineId, periodKey, statusName)
    If IsNumeric(v) Then If v >= 0 And v <= 1 Then TryAvgFraction = v : Exit Function

    v = CpAggregate(cn, runtimeCatalog, tagFull, t1, t2, "AVG(ValueScaled)", machineId, periodKey, statusName)
    If IsNumeric(v) Then If v >= 0 And v <= 1 Then TryAvgFraction = v : Exit Function

    TryAvgFraction = Empty
End Function

' ─────────────────────────────────────────────────────────────────────
' READ ONLY *_Count (single SQL roundtrip for all 5 statuses)
' ─────────────────────────────────────────────────────────────────────
Private Function BuildCountTagFullName(ByVal machineId, ByVal statusName)
    Dim basePrefix, baseCore
    basePrefix = G_MACHINES(machineId)
    baseCore = COUNT_TAG_CORE_TEMPLATE
    baseCore = Replace(baseCore, "{MACHINE_ID}", machineId)
    baseCore = Replace(baseCore, "{STATUS_NAME}", statusName)
    BuildCountTagFullName = basePrefix & COUNT_TAG_PREFIX_JOINER & baseCore & COUNT_TAG_SUFFIX
End Function

Private Function ProbeCountTagArchive(ByVal cn, ByVal runtimeCatalog, ByVal tagFullName, ByVal tFrom, ByVal tTo, ByVal aggExpr, ByRef insertedRows, ByRef hasNumeric, ByRef elapsedMsValue, ByRef errDesc)
    Dim cmd, rs, sqlText, cpCall, startTimer, tmp

    insertedRows = 0
    hasNumeric = 0
    elapsedMsValue = 0
    errDesc = ""
    ProbeCountTagArchive = False

    cpCall = BuildCpTagStatisticExLiteral(runtimeCatalog, tagFullName, tFrom, tTo, aggExpr)
    sqlText = "DECLARE @Probe TABLE (frac SQL_VARIANT);" & vbCrLf & _
              "INSERT INTO @Probe (frac) EXEC " & cpCall & ";" & vbCrLf & _
              "SELECT COUNT(1) AS inserted_rows, " & _
              "MAX(CASE WHEN TRY_CONVERT(FLOAT, frac) IS NOT NULL THEN 1 ELSE 0 END) AS has_numeric " & _
              "FROM @Probe;"

    startTimer = Timer

    On Error Resume Next
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = cn
    cmd.CommandTimeout = SQL_TIMEOUT_SECONDS
    cmd.CommandType = adCmdText
    cmd.CommandText = sqlText

    Set rs = cmd.Execute
    elapsedMsValue = ElapsedMs(startTimer)
    If Err.Number <> 0 Then
        errDesc = S(Err.Description)
        Err.Clear
    ElseIf Not (rs Is Nothing) Then
        If Not rs.EOF Then
            tmp = rs.Fields("inserted_rows").Value
            If IsNumeric(tmp) Then insertedRows = CLng(tmp)
            tmp = rs.Fields("has_numeric").Value
            If IsNumeric(tmp) Then hasNumeric = CLng(tmp)
            ProbeCountTagArchive = True
        End If
        rs.Close
    End If
    On Error GoTo 0

    Set rs = Nothing
    Set cmd = Nothing
End Function

Private Function ResolveStartupValidationLookbackMinutes(ByVal due1D, ByVal due7D, ByVal due30D)
    ResolveStartupValidationLookbackMinutes = STARTUP_VALIDATE_LOOKBACK_1D_MINUTES
    If due7D Then
        If STARTUP_VALIDATE_LOOKBACK_7D_MINUTES > ResolveStartupValidationLookbackMinutes Then
            ResolveStartupValidationLookbackMinutes = STARTUP_VALIDATE_LOOKBACK_7D_MINUTES
        End If
    End If
    If due30D Then
        If STARTUP_VALIDATE_LOOKBACK_30D_MINUTES > ResolveStartupValidationLookbackMinutes Then
            ResolveStartupValidationLookbackMinutes = STARTUP_VALIDATE_LOOKBACK_30D_MINUTES
        End If
    End If
End Function

Private Function ValidateMachineCountTags(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal lookbackMinutes, ByRef validationState, ByRef validationReason)
    Dim statuses, idx, statusName, tagFull, aggExpr, aggs
    Dim tFrom, tTo, insertedRows, hasNumeric, elapsedMsValue, errDesc
    Dim probeOk, hasArchiveData, validStatusCount, hardFailCount
    Dim aggIdx, aggCount, statusProbeSucceeded, statusHadRows, lastErrDesc

    If g_machineValidationDone.Exists(machineId) Then
        validationState = CLng(g_machineValidationState(machineId))
        validationReason = CStr(g_machineValidationReason(machineId))
        ValidateMachineCountTags = CBool(g_machineValidationOk(machineId))
        Exit Function
    End If

    statuses = Array("Aborted", "Halted", "Running", "Waiting", "Interrupted")
    aggs = GetAggExprList(runtimeCatalog)
    aggCount = UBound(aggs) + 1
    tTo = Now()
    tFrom = DateAdd("n", -lookbackMinutes, tTo)
    validStatusCount = 0
    hardFailCount = 0

    For idx = 0 To UBound(statuses)
        statusName = CStr(statuses(idx))
        tagFull = BuildCountTagFullName(machineId, statusName)
        statusProbeSucceeded = False
        statusHadRows = False
        lastErrDesc = ""

        For aggIdx = 0 To UBound(aggs)
            aggExpr = CStr(aggs(aggIdx))
            probeOk = ProbeCountTagArchive(cn, runtimeCatalog, tagFull, tFrom, tTo, aggExpr, insertedRows, hasNumeric, elapsedMsValue, errDesc)
            If probeOk Then
                statusProbeSucceeded = True
                hasArchiveData = (insertedRows > 0)
                If hasArchiveData Then statusHadRows = True
                LogMsg LOG_LEVEL_INFO, "StartupValidate", _
                       "machine=" & machineId & _
                       " status=" & statusName & _
                       " tag_full=" & tagFull & _
                       " agg=" & aggExpr & _
                       " data_exists=" & CStr(hasArchiveData) & _
                       " inserted_rows=" & insertedRows & _
                       " has_numeric=" & hasNumeric & _
                       " lookback_min=" & lookbackMinutes & _
                       " elapsed_ms=" & elapsedMsValue
                Exit For
            Else
                lastErrDesc = errDesc
                LogMsg LOG_LEVEL_WARN, "StartupValidate", _
                       "machine=" & machineId & _
                       " status=" & statusName & _
                       " tag_full=" & tagFull & _
                       " agg=" & aggExpr & _
                       " data_exists=False" & _
                       " probe_failed=1" & _
                       " err=" & errDesc
            End If
        Next

        hasArchiveData = statusHadRows
        If hasArchiveData Then validStatusCount = validStatusCount + 1

        If Not statusProbeSucceeded Then
            hardFailCount = hardFailCount + 1
            LogMsg LOG_LEVEL_WARN, "StartupValidate", _
                   "machine=" & machineId & _
                   " status=" & statusName & _
                   " tag_full=" & tagFull & _
                   " data_exists=False" & _
                   " probe_failed=1" & _
                   " err=" & lastErrDesc & _
                   " fallback_aggs_attempted=" & aggCount
        End If
    Next

    g_machineValidationDone(machineId) = True

    If hardFailCount > 0 Then
        validationState = VALIDATE_STATE_HARD_FAIL
        validationReason = "hard_fail_probe_or_schema_error"
        g_machineValidationOk(machineId) = False
        ValidateMachineCountTags = False
        LogMsg LOG_LEVEL_WARN, "StartupValidate", _
               "HIGH_PRIORITY machine=" & machineId & _
               " message=startup_validation_hard_failure; output_writes_suppressed" & _
               " lookback_min=" & lookbackMinutes & _
               " fallback_aggs_attempted=" & aggCount
    ElseIf validStatusCount > 0 Then
        validationState = VALIDATE_STATE_OK
        validationReason = "valid_recent_rows_detected"
        g_machineValidationOk(machineId) = True
        ValidateMachineCountTags = True
    Else
        validationState = VALIDATE_STATE_NO_RECENT
        validationReason = "no_recent_rows_in_lookback"
        g_machineValidationOk(machineId) = True
        ValidateMachineCountTags = True
        LogMsg LOG_LEVEL_WARN, "StartupValidate", _
               "HIGH_PRIORITY machine=" & machineId & _
               " message=no_recent_rows_detected; output_writes_allowed_with_warning" & _
               " lookback_min=" & lookbackMinutes & _
               " fallback_aggs_attempted=" & aggCount
    End If

    g_machineValidationState(machineId) = validationState
    g_machineValidationReason(machineId) = validationReason
End Function

Private Function BuildCpTagStatisticExLiteral(ByVal runtimeCatalog, ByVal tagFullName, ByVal tFrom, ByVal tTo, ByVal aggExpr)
    Dim safeCatalog, safeTag
    safeCatalog = Replace(runtimeCatalog, "'", "''")
    safeTag = Replace(tagFullName, "'", "''")
    BuildCpTagStatisticExLiteral = "cp_TagStatisticEx '" & safeCatalog & "'," & _
                                   "'TAG:R,''" & safeTag & "'',''" & TS3(tFrom) & "'',''" & TS3(tTo) & "'''," & _
                                   "'" & aggExpr & "'"
End Function

Private Function BuildBatchCacheKey(ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal t1, ByVal t2, ByVal aggExpr)
    BuildBatchCacheKey = runtimeCatalog & "|" & machineId & "|" & periodKey & "|" & TS3(t1) & "|" & TS3(t2) & "|" & aggExpr
End Function

Private Function BuildStatusTagMap(ByVal machineId, ByVal periodKey, ByVal t1, ByVal t2)
    Dim cacheKey, d
    cacheKey = machineId & "|" & periodKey & "|" & TS3(t1) & "|" & TS3(t2)

    If g_statusTagFullNameCache.Exists(cacheKey) Then
        Set BuildStatusTagMap = g_statusTagFullNameCache(cacheKey)
        Exit Function
    End If

    Set d = CreateObject("Scripting.Dictionary")
    d.Add "Aborted", BuildCountTagFullName(machineId, "Aborted")
    d.Add "Halted", BuildCountTagFullName(machineId, "Halted")
    d.Add "Running", BuildCountTagFullName(machineId, "Running")
    d.Add "Waiting", BuildCountTagFullName(machineId, "Waiting")
    d.Add "Interrupted", BuildCountTagFullName(machineId, "Interrupted")
    Set g_statusTagFullNameCache(cacheKey) = d
    Set BuildStatusTagMap = d
End Function

Private Function CopyStatusFractions(ByVal srcDict)
    Dim dst
    Set dst = CreateObject("Scripting.Dictionary")
    dst.Add "Aborted", srcDict("Aborted")
    dst.Add "Halted", srcDict("Halted")
    dst.Add "Running", srcDict("Running")
    dst.Add "Waiting", srcDict("Waiting")
    dst.Add "Interrupted", srcDict("Interrupted")
    Set CopyStatusFractions = dst
End Function

Private Function SumStatusFractions(ByVal d)
    SumStatusFractions = CDbl(d("Aborted")) + CDbl(d("Halted")) + CDbl(d("Running")) + CDbl(d("Waiting")) + CDbl(d("Interrupted"))
End Function

Private Function GetAggExprList(ByVal runtimeCatalog)
    Dim cfg
    cfg = PROJECT_BATCH_AGG_LIST_DEFAULT
    If UCase(S(runtimeCatalog)) = "WINCC" Then cfg = PROJECT_BATCH_AGG_LIST_WINCC
    GetAggExprList = Split(cfg, "|")
End Function

Private Function ExecuteStatusFractionsSingleFallback(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal t1, ByVal t2, ByVal tagMap, ByRef numericCount)
    Dim outDict, statuses, idx, statusName, tagFull, v, infoMsg
    Set outDict = CreateObject("Scripting.Dictionary")
    outDict.Add "Aborted", 0
    outDict.Add "Halted", 0
    outDict.Add "Running", 0
    outDict.Add "Waiting", 0
    outDict.Add "Interrupted", 0
    numericCount = 0

    statuses = Array("Aborted", "Halted", "Running", "Waiting", "Interrupted")
    For idx = 0 To UBound(statuses)
        statusName = CStr(statuses(idx))
        tagFull = CStr(tagMap(statusName))
        v = TryAvgFraction(cn, runtimeCatalog, tagFull, t1, t2, machineId, periodKey, statusName)
        If IsNumeric(v) Then
            outDict(statusName) = CDbl(v)
            numericCount = numericCount + 1
        End If
    Next

    infoMsg = "single_fallback machine=" & machineId & _
              " window=" & periodKey & _
              " catalog=" & runtimeCatalog & _
              " sql_from=" & TS3(t1) & _
              " sql_to=" & TS3(t2) & _
              " numeric=" & numericCount
    LogMsg LOG_LEVEL_INFO, "GetStatusFractionsBatch", infoMsg

    If numericCount > 0 Then
        Set ExecuteStatusFractionsSingleFallback = outDict
    Else
        Set ExecuteStatusFractionsSingleFallback = Nothing
    End If
End Function

Private Function ExecuteStatusFractionsBatch(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal t1, ByVal t2, ByVal aggExpr, ByVal tagMap, ByRef hasNumericCount, ByRef insertedStatusCount, ByRef allNullNoNumeric, ByRef batchOk)
    Dim cmd, rs, sqlText, startedAt, completedAt, startTimer, elapsedMsValue, infoMsg
    Dim statuses, aliases, idx, statusName, tagFull, cpCall
    Dim outDict, fieldValue, diagRows, diagNumeric, diagElapsedMs
    Dim aliasKey, statusLabel, insertedRowsValue, isNumericValue, insertedFlag, numericFlag, statusElapsedMs
    Dim valA, valH, valR, valW, valI
    Dim cacheKey, cacheEntry

    startedAt = Now
    startTimer = Timer
    hasNumericCount = 0
    insertedStatusCount = 0
    allNullNoNumeric = 0
    batchOk = 0

    Set outDict = CreateObject("Scripting.Dictionary")
    outDict.Add "Aborted", 0
    outDict.Add "Halted", 0
    outDict.Add "Running", 0
    outDict.Add "Waiting", 0
    outDict.Add "Interrupted", 0

    statuses = Array("Aborted", "Halted", "Running", "Waiting", "Interrupted")
    aliases  = Array("a", "h", "r", "w", "i")

    Set diagRows = CreateObject("Scripting.Dictionary")
    Set diagNumeric = CreateObject("Scripting.Dictionary")
    Set diagElapsedMs = CreateObject("Scripting.Dictionary")

    cacheKey = BuildBatchCacheKey(runtimeCatalog, machineId, periodKey, t1, t2, aggExpr)
    If g_batchResultCache.Exists(cacheKey) Then
        Set cacheEntry = g_batchResultCache(cacheKey)
        hasNumericCount = CLng(cacheEntry("hasNumericCount"))
        insertedStatusCount = CLng(cacheEntry("insertedStatusCount"))
        allNullNoNumeric = CLng(cacheEntry("allNullNoNumeric"))
        batchOk = CLng(cacheEntry("batchOk"))
        If cacheEntry.Exists("hasOutDict") And CLng(cacheEntry("hasOutDict")) = 1 Then
            Set outDict = CreateObject("Scripting.Dictionary")
            outDict.Add "Aborted", CDbl(cacheEntry("Aborted"))
            outDict.Add "Halted", CDbl(cacheEntry("Halted"))
            outDict.Add "Running", CDbl(cacheEntry("Running"))
            outDict.Add "Waiting", CDbl(cacheEntry("Waiting"))
            outDict.Add "Interrupted", CDbl(cacheEntry("Interrupted"))
            Set ExecuteStatusFractionsBatch = outDict
        Else
            Set ExecuteStatusFractionsBatch = Nothing
        End If
        LogMsg LOG_LEVEL_DEBUG, "GetStatusFractionsBatch", _
               "batch_cache_hit machine=" & machineId & _
               " window=" & periodKey & _
               " agg=" & aggExpr & _
               " batch_ok=" & batchOk
        Exit Function
    End If

    sqlText = "DECLARE @Status TABLE (status_alias NVARCHAR(1), frac FLOAT);" & vbCrLf
    sqlText = sqlText & "DECLARE @StatusDiag TABLE (status_alias NVARCHAR(1), inserted_rows INT, is_numeric INT, elapsed_ms INT);" & vbCrLf
    sqlText = sqlText & "DECLARE @cpStart DATETIME2(3);" & vbCrLf
    For idx = 0 To UBound(statuses)
        statusName = CStr(statuses(idx))
        tagFull = CStr(tagMap(statusName))
        cpCall = BuildCpTagStatisticExLiteral(runtimeCatalog, tagFull, t1, t2, aggExpr)
        sqlText = sqlText & "SET @cpStart = SYSDATETIME();" & vbCrLf
        sqlText = sqlText & "INSERT INTO @Status (frac) EXEC " & cpCall & ";" & vbCrLf
        sqlText = sqlText & "INSERT INTO @StatusDiag (status_alias, inserted_rows, is_numeric, elapsed_ms) VALUES ('" & CStr(aliases(idx)) & "', @@ROWCOUNT, 0, DATEDIFF(MILLISECOND, @cpStart, SYSDATETIME()));" & vbCrLf
        sqlText = sqlText & "UPDATE @Status SET status_alias='" & CStr(aliases(idx)) & "' WHERE status_alias IS NULL;" & vbCrLf
        sqlText = sqlText & "UPDATE d SET is_numeric = CASE " & _
                 "WHEN d.inserted_rows > 0 AND EXISTS (SELECT 1 FROM @Status s WHERE s.status_alias=d.status_alias AND TRY_CONVERT(FLOAT, s.frac) IS NOT NULL) " & _
                 "THEN 1 ELSE 0 END " & _
                 "FROM @StatusDiag d WHERE d.status_alias='" & CStr(aliases(idx)) & "';" & vbCrLf
    Next
    sqlText = sqlText & "SELECT " & _
              "MAX(CASE WHEN status_alias='a' THEN frac END) AS a," & _
              "MAX(CASE WHEN status_alias='h' THEN frac END) AS h," & _
              "MAX(CASE WHEN status_alias='r' THEN frac END) AS r," & _
              "MAX(CASE WHEN status_alias='w' THEN frac END) AS w," & _
              "MAX(CASE WHEN status_alias='i' THEN frac END) AS i," & _
              "MAX(CASE WHEN status_alias='a' THEN inserted_rows END) AS a_rows," & _
              "MAX(CASE WHEN status_alias='h' THEN inserted_rows END) AS h_rows," & _
              "MAX(CASE WHEN status_alias='r' THEN inserted_rows END) AS r_rows," & _
              "MAX(CASE WHEN status_alias='w' THEN inserted_rows END) AS w_rows," & _
              "MAX(CASE WHEN status_alias='i' THEN inserted_rows END) AS i_rows," & _
              "MAX(CASE WHEN status_alias='a' THEN is_numeric END) AS a_num," & _
              "MAX(CASE WHEN status_alias='h' THEN is_numeric END) AS h_num," & _
              "MAX(CASE WHEN status_alias='r' THEN is_numeric END) AS r_num," & _
              "MAX(CASE WHEN status_alias='w' THEN is_numeric END) AS w_num," & _
              "MAX(CASE WHEN status_alias='i' THEN is_numeric END) AS i_num," & _
              "MAX(CASE WHEN status_alias='a' THEN elapsed_ms END) AS a_ms," & _
              "MAX(CASE WHEN status_alias='h' THEN elapsed_ms END) AS h_ms," & _
              "MAX(CASE WHEN status_alias='r' THEN elapsed_ms END) AS r_ms," & _
              "MAX(CASE WHEN status_alias='w' THEN elapsed_ms END) AS w_ms," & _
              "MAX(CASE WHEN status_alias='i' THEN elapsed_ms END) AS i_ms " & _
              "FROM @StatusDiag d LEFT JOIN @Status s ON s.status_alias=d.status_alias;"

    On Error Resume Next
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = cn
    cmd.CommandTimeout = SQL_TIMEOUT_SECONDS
    cmd.CommandType = adCmdText
    cmd.CommandText = sqlText

    If CURRENT_LOG_LEVEL >= LOG_LEVEL_DEBUG Then
        LogMsg LOG_LEVEL_DEBUG, "GetStatusFractionsBatch", "SQL=[" & sqlText & "]"
    End If

    Set rs = cmd.Execute
    If Err.Number = 0 Then
        If Not (rs Is Nothing) Then
            If Not rs.EOF Then
                If rs.Fields.Count >= 5 Then
                    fieldValue = rs.Fields("a").Value: If IsNumeric(fieldValue) Then outDict("Aborted") = CDbl(fieldValue): hasNumericCount = hasNumericCount + 1
                    fieldValue = rs.Fields("h").Value: If IsNumeric(fieldValue) Then outDict("Halted") = CDbl(fieldValue): hasNumericCount = hasNumericCount + 1
                    fieldValue = rs.Fields("r").Value: If IsNumeric(fieldValue) Then outDict("Running") = CDbl(fieldValue): hasNumericCount = hasNumericCount + 1
                    fieldValue = rs.Fields("w").Value: If IsNumeric(fieldValue) Then outDict("Waiting") = CDbl(fieldValue): hasNumericCount = hasNumericCount + 1
                    fieldValue = rs.Fields("i").Value: If IsNumeric(fieldValue) Then outDict("Interrupted") = CDbl(fieldValue): hasNumericCount = hasNumericCount + 1

                    diagRows("a") = 0: diagRows("h") = 0: diagRows("r") = 0: diagRows("w") = 0: diagRows("i") = 0
                    diagNumeric("a") = 0: diagNumeric("h") = 0: diagNumeric("r") = 0: diagNumeric("w") = 0: diagNumeric("i") = 0
                    diagElapsedMs("a") = 0: diagElapsedMs("h") = 0: diagElapsedMs("r") = 0: diagElapsedMs("w") = 0: diagElapsedMs("i") = 0

                    If rs.Fields.Count >= 20 Then
                        insertedRowsValue = rs.Fields("a_rows").Value: If IsNumeric(insertedRowsValue) Then diagRows("a") = CLng(insertedRowsValue)
                        insertedRowsValue = rs.Fields("h_rows").Value: If IsNumeric(insertedRowsValue) Then diagRows("h") = CLng(insertedRowsValue)
                        insertedRowsValue = rs.Fields("r_rows").Value: If IsNumeric(insertedRowsValue) Then diagRows("r") = CLng(insertedRowsValue)
                        insertedRowsValue = rs.Fields("w_rows").Value: If IsNumeric(insertedRowsValue) Then diagRows("w") = CLng(insertedRowsValue)
                        insertedRowsValue = rs.Fields("i_rows").Value: If IsNumeric(insertedRowsValue) Then diagRows("i") = CLng(insertedRowsValue)

                        isNumericValue = rs.Fields("a_num").Value: If IsNumeric(isNumericValue) Then diagNumeric("a") = CLng(isNumericValue)
                        isNumericValue = rs.Fields("h_num").Value: If IsNumeric(isNumericValue) Then diagNumeric("h") = CLng(isNumericValue)
                        isNumericValue = rs.Fields("r_num").Value: If IsNumeric(isNumericValue) Then diagNumeric("r") = CLng(isNumericValue)
                        isNumericValue = rs.Fields("w_num").Value: If IsNumeric(isNumericValue) Then diagNumeric("w") = CLng(isNumericValue)
                        isNumericValue = rs.Fields("i_num").Value: If IsNumeric(isNumericValue) Then diagNumeric("i") = CLng(isNumericValue)

                        statusElapsedMs = rs.Fields("a_ms").Value: If IsNumeric(statusElapsedMs) Then diagElapsedMs("a") = CLng(statusElapsedMs)
                        statusElapsedMs = rs.Fields("h_ms").Value: If IsNumeric(statusElapsedMs) Then diagElapsedMs("h") = CLng(statusElapsedMs)
                        statusElapsedMs = rs.Fields("r_ms").Value: If IsNumeric(statusElapsedMs) Then diagElapsedMs("r") = CLng(statusElapsedMs)
                        statusElapsedMs = rs.Fields("w_ms").Value: If IsNumeric(statusElapsedMs) Then diagElapsedMs("w") = CLng(statusElapsedMs)
                        statusElapsedMs = rs.Fields("i_ms").Value: If IsNumeric(statusElapsedMs) Then diagElapsedMs("i") = CLng(statusElapsedMs)
                    End If
                    batchOk = 1
                End If
            End If
            rs.Close
        End If
    Else
        LogMsg LOG_LEVEL_WARN, "GetStatusFractionsBatch", "Batch query failed; no fractions returned. machine=" & machineId & " window=" & periodKey & " agg=" & aggExpr & " err=" & S(Err.Description)
        Err.Clear
    End If
    On Error GoTo 0

    Set rs = Nothing
    Set cmd = Nothing

    For idx = 0 To UBound(statuses)
        aliasKey = CStr(aliases(idx))
        statusLabel = CStr(statuses(idx))
        insertedRowsValue = 0
        isNumericValue = 0
        insertedFlag = 0
        numericFlag = 0
        statusElapsedMs = 0

        If diagRows.Exists(aliasKey) Then insertedRowsValue = CLng(diagRows(aliasKey))
        If diagNumeric.Exists(aliasKey) Then isNumericValue = CLng(diagNumeric(aliasKey))
        If diagElapsedMs.Exists(aliasKey) Then statusElapsedMs = CLng(diagElapsedMs(aliasKey))
        If insertedRowsValue > 0 Then insertedFlag = 1
        If isNumericValue <> 0 Then numericFlag = 1
        If insertedFlag = 1 Then insertedStatusCount = insertedStatusCount + 1

        tagFull = CStr(tagMap(statusLabel))
        LogMsg LOG_LEVEL_INFO, "GetStatusFractionsBatch", _
               "status_diag machine=" & machineId & _
               " window=" & periodKey & _
               " agg=" & aggExpr & _
               " status=" & statusLabel & _
               " tag_full=" & tagFull & _
               " inserted=" & insertedFlag & _
               " inserted_rows=" & insertedRowsValue & _
               " is_numeric=" & numericFlag & _
               " elapsed_ms=" & statusElapsedMs
    Next
    If hasNumericCount = 0 And insertedStatusCount = 5 Then allNullNoNumeric = 1

    completedAt = Now
    elapsedMsValue = ElapsedMs(startTimer)
    If batchOk = 1 Then
        valA = outDict("Aborted")
        valH = outDict("Halted")
        valR = outDict("Running")
        valW = outDict("Waiting")
        valI = outDict("Interrupted")
        infoMsg = "machine=" & machineId & _
                  " window=" & periodKey & _
                  " agg=" & aggExpr & _
                  " sql_from=" & TS3(t1) & _
                  " sql_to=" & TS3(t2) & _
                  " started=" & TS3(startedAt) & _
                  " ended=" & TS3(completedAt) & _
                  " elapsed_ms=" & elapsedMsValue & _
                  " numeric=" & hasNumericCount & _
                  " a=" & valA & _
                  " h=" & valH & _
                  " r=" & valR & _
                  " w=" & valW & _
                  " i=" & valI
        LogMsg LOG_LEVEL_INFO, "GetStatusFractionsBatch", infoMsg
        Set ExecuteStatusFractionsBatch = outDict
    Else
        LogMsg LOG_LEVEL_WARN, "GetStatusFractionsBatch", _
               "batch_result_invalid machine=" & machineId & _
               " window=" & periodKey & _
               " agg=" & aggExpr & _
               " sql_from=" & TS3(t1) & _
               " sql_to=" & TS3(t2) & _
               " started=" & TS3(startedAt) & _
               " ended=" & TS3(completedAt) & _
               " elapsed_ms=" & elapsedMsValue & _
               " reason=no_usable_result_row"
        Set ExecuteStatusFractionsBatch = Nothing
    End If

    Set cacheEntry = CreateObject("Scripting.Dictionary")
    cacheEntry.Add "hasNumericCount", hasNumericCount
    cacheEntry.Add "insertedStatusCount", insertedStatusCount
    cacheEntry.Add "allNullNoNumeric", allNullNoNumeric
    cacheEntry.Add "batchOk", batchOk
    If ExecuteStatusFractionsBatch Is Nothing Then
        cacheEntry.Add "hasOutDict", 0
    Else
        cacheEntry.Add "hasOutDict", 1
        cacheEntry.Add "Aborted", CDbl(ExecuteStatusFractionsBatch("Aborted"))
        cacheEntry.Add "Halted", CDbl(ExecuteStatusFractionsBatch("Halted"))
        cacheEntry.Add "Running", CDbl(ExecuteStatusFractionsBatch("Running"))
        cacheEntry.Add "Waiting", CDbl(ExecuteStatusFractionsBatch("Waiting"))
        cacheEntry.Add "Interrupted", CDbl(ExecuteStatusFractionsBatch("Interrupted"))
    End If
    Set g_batchResultCache(cacheKey) = cacheEntry
End Function

Private Function GetStatusFractionsBatch(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal t1, ByVal t2)
    Dim aggs, idx, aggExpr
    Dim currentDict, bestDict
    Dim numericCount, bestNumericCount, insertedStatusCount, allNullNoNumeric, batchOk
    Dim currentSum, bestSum
    Dim selectedAgg, selectedIdx, aggTrialSummary, currentTrial, fastFailApplied, fallbackUsed, zeroInsertedWarnLogged
    Dim queryFailureCount
    Dim tagMap, allowFallbackAggs, selectedPath

    aggs = GetAggExprList(runtimeCatalog)
    Set tagMap = BuildStatusTagMap(machineId, periodKey, t1, t2)
    bestNumericCount = -1
    bestSum = -1
    selectedAgg = ""
    selectedIdx = -1
    aggTrialSummary = ""
    fastFailApplied = 0
    fallbackUsed = 0
    zeroInsertedWarnLogged = 0
    queryFailureCount = 0
    allowFallbackAggs = 0
    selectedPath = "no_data"

    For idx = 0 To UBound(aggs)
        If idx > 0 And allowFallbackAggs = 0 And queryFailureCount = 0 Then Exit For

        aggExpr = CStr(aggs(idx))
        Set currentDict = ExecuteStatusFractionsBatch(cn, runtimeCatalog, machineId, periodKey, t1, t2, aggExpr, tagMap, numericCount, insertedStatusCount, allNullNoNumeric, batchOk)
        If batchOk = 1 And Not (currentDict Is Nothing) Then
            currentSum = SumStatusFractions(currentDict)
            currentTrial = aggExpr & "{ok=1,numeric=" & numericCount & ",inserted=" & insertedStatusCount & ",sum=" & currentSum & "}"
            If idx = 0 Then
                If numericCount > 0 Then
                    Set bestDict = CopyStatusFractions(currentDict)
                    bestNumericCount = numericCount
                    bestSum = currentSum
                    selectedAgg = aggExpr
                    selectedIdx = idx
                    selectedPath = "batch"
                    Exit For
                Else
                    allowFallbackAggs = 1
                End If
            End If
        Else
            currentSum = 0
            currentTrial = aggExpr & "{ok=0,query_failure=1}"
            queryFailureCount = queryFailureCount + 1
            LogMsg LOG_LEVEL_WARN, "GetStatusFractionsBatch", _
                   "agg_query_failed machine=" & machineId & _
                   " window=" & periodKey & _
                   " catalog=" & runtimeCatalog & _
               " agg=" & aggExpr & _
               " sql_from=" & TS3(t1) & _
               " sql_to=" & TS3(t2)
        End If
        If Len(aggTrialSummary) > 0 Then
            aggTrialSummary = aggTrialSummary & "; "
        End If
        aggTrialSummary = aggTrialSummary & currentTrial

        If batchOk = 1 And zeroInsertedWarnLogged = 0 Then
            If insertedStatusCount > 0 And currentSum = 0 Then
                LogMsg LOG_LEVEL_WARN, "GetStatusFractionsBatch", _
                       "zero_sum_with_inserted_rows machine=" & machineId & _
                       " window=" & periodKey & _
                       " catalog=" & runtimeCatalog & _
                       " agg=" & aggExpr & _
                       " inserted_statuses=" & insertedStatusCount & _
                       " numeric=" & numericCount & _
                       " sum=" & currentSum
                zeroInsertedWarnLogged = 1
            End If
        End If

        If batchOk = 1 And (bestDict Is Nothing) Then
            Set bestDict = CopyStatusFractions(currentDict)
            bestNumericCount = numericCount
            bestSum = currentSum
            selectedAgg = aggExpr
            selectedIdx = idx
            selectedPath = "batch"
        ElseIf batchOk = 1 And ((numericCount > bestNumericCount) Or _
               (numericCount = bestNumericCount And currentSum > bestSum And bestSum <= 0)) Then
            Set bestDict = CopyStatusFractions(currentDict)
            bestNumericCount = numericCount
            bestSum = currentSum
            selectedAgg = aggExpr
            selectedIdx = idx
            selectedPath = "batch"
        End If

        If batchOk = 1 And idx = 0 And allNullNoNumeric = 1 Then
            fastFailApplied = 1
            LogMsg LOG_LEVEL_WARN, "GetStatusFractionsBatch", _
                   "fast_fail_unsupported_first_agg machine=" & machineId & _
                   " window=" & periodKey & _
                   " catalog=" & runtimeCatalog & _
                   " agg=" & aggExpr & _
                   " reason=all_null_no_numeric; continuing_fallback_scan=1"
            allowFallbackAggs = 1
        End If

        If batchOk = 1 And numericCount = 5 And currentSum > 0 Then Exit For
    Next

    If selectedIdx > 0 Then fallbackUsed = 1

    LogMsg LOG_LEVEL_INFO, "GetStatusFractionsBatch", _
           "agg_trials machine=" & machineId & _
           " window=" & periodKey & _
           " catalog=" & runtimeCatalog & _
           " sql_from=" & TS3(t1) & _
           " sql_to=" & TS3(t2) & _
           " trials=[" & aggTrialSummary & "]" & _
           " fast_fail=" & fastFailApplied & _
           " data_path=" & selectedPath

    If Not (bestDict Is Nothing) And bestNumericCount = 0 Then
        LogMsg LOG_LEVEL_WARN, "GetStatusFractionsBatch", _
               "all_aggs_non_numeric machine=" & machineId & _
               " catalog=" & runtimeCatalog & _
               " window=" & periodKey & _
               " sql_from=" & TS3(t1) & _
               " sql_to=" & TS3(t2)
    End If

    If bestDict Is Nothing And queryFailureCount = (UBound(aggs) + 1) Then
        Set bestDict = ExecuteStatusFractionsSingleFallback(cn, runtimeCatalog, machineId, periodKey, t1, t2, tagMap, numericCount)
        If Not (bestDict Is Nothing) Then
            bestNumericCount = numericCount
            bestSum = SumStatusFractions(bestDict)
            selectedAgg = "single_status_path"
            selectedIdx = -1
            fallbackUsed = 1
            selectedPath = "fallback_single"
        End If
    End If

    If bestDict Is Nothing Then
        LogMsg LOG_LEVEL_WARN, "GetStatusFractionsBatch", _
               "output_suppressed_due_query_failure machine=" & machineId & _
               " window=" & periodKey & _
               " catalog=" & runtimeCatalog & _
               " sql_from=" & TS3(t1) & _
               " sql_to=" & TS3(t2) & _
               " failed_trials=" & queryFailureCount & _
               " data_path=no_data"
        Set GetStatusFractionsBatch = Nothing
        Exit Function
    End If

    LogMsg LOG_LEVEL_INFO, "GetStatusFractionsBatch", _
           "selected_agg machine=" & machineId & _
           " window=" & periodKey & _
           " catalog=" & runtimeCatalog & _
           " agg=" & selectedAgg & _
           " idx=" & selectedIdx & _
           " fallback_used=" & fallbackUsed & _
           " numeric=" & bestNumericCount & _
           " sum=" & bestSum & _
           " data_path=" & selectedPath

    Set GetStatusFractionsBatch = bestDict
End Function

Private Function GetCountFraction(ByVal statusFractions, ByVal statusName)
    If (statusFractions Is Nothing) Then
        GetCountFraction = 0
    ElseIf statusFractions.Exists(statusName) Then
        If IsNumeric(statusFractions(statusName)) Then
            GetCountFraction = CDbl(statusFractions(statusName))
        Else
            GetCountFraction = 0
        End If
    Else
        GetCountFraction = 0
    End If
End Function

' ─────────────────────────────────────────────────────────────────────
' Combine 5 → 3 (your rule), normalize if overlap makes sum > 1
' ─────────────────────────────────────────────────────────────────────
Private Sub GetCombinedFractions(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal t1, ByVal t2, ByRef fRun, ByRef fHalt, ByRef fAbort, ByRef inputValid)
    Dim a, h, r, w, i, total, statusFractions

    ' Initialise outputs so callers always receive well-defined values
    fRun = 0 : fHalt = 0 : fAbort = 0
    inputValid = False

    Set statusFractions = GetStatusFractionsBatch(cn, runtimeCatalog, machineId, periodKey, t1, t2)
    If statusFractions Is Nothing Then Exit Sub
    inputValid = True

    a = GetCountFraction(statusFractions, "Aborted")
    h = GetCountFraction(statusFractions, "Halted")
    r = GetCountFraction(statusFractions, "Running")
    w = GetCountFraction(statusFractions, "Waiting")
    i = GetCountFraction(statusFractions, "Interrupted")

    fRun   = r
    fHalt  = h + i
    fAbort = a + w

    ' clamp just in case
    If fRun < 0 Then fRun = 0
    If fHalt < 0 Then fHalt = 0
    If fAbort < 0 Then fAbort = 0

    ' If your PLC logic overlaps states, sum can exceed 1 → normalize safely
    total = fRun + fHalt + fAbort
    If total > 1.000001 Then
        fRun   = fRun / total
        fHalt  = fHalt / total
        fAbort = fAbort / total
    End If
End Sub

Private Function OutTagName(ByVal machineId, ByVal periodKey, ByVal statusDispName, ByVal pieChart)
    If pieChart Then
        OutTagName = machineId & "_" & periodKey & "_Machine_" & statusDispName & "_Percentage_PieChart"
    Else

    End If

Private Function IsPieChartOutputTag(ByVal tag)
    IsPieChartOutputTag = (InStr(1, S(tag), "_Percentage_PieChart", vbTextCompare) > 0)
End Function

Private Sub WriteTag(ByVal tag, ByVal val)
    Dim readBack, requestedVal, delta, isPieChartTag

    isPieChartTag = IsPieChartOutputTag(tag)

    On Error Resume Next
    HMIRuntime.Tags(tag).Write val
    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "HMITag", "Write failed: " & S(tag) & " -> " & S(Err.Description)
        Err.Clear
        On Error GoTo 0
        Exit Sub
    End If

    If CURRENT_LOG_LEVEL >= LOG_LEVEL_DEBUG And isPieChartTag Then
        LogMsg LOG_LEVEL_DEBUG, "HMITag", "write_ok tag=" & S(tag) & " value=" & S(val)
    End If

    If CURRENT_LOG_LEVEL >= LOG_LEVEL_DEBUG And isPieChartTag Then
        readBack = HMIRuntime.Tags(tag).Read
        If Err.Number <> 0 Then
            LogMsg LOG_LEVEL_WARN, "HMITag", "Read-back failed: " & S(tag) & " -> " & S(Err.Description)
            Err.Clear
        ElseIf IsNumeric(readBack) And IsNumeric(val) Then
            requestedVal = CDbl(val)
            delta = Abs(CDbl(readBack) - requestedVal)
            If delta > WRITE_VERIFY_TOLERANCE Then
                LogMsg LOG_LEVEL_WARN, "HMITag", "write_mismatch tag=" & S(tag) & " wrote=" & requestedVal & " read=" & CDbl(readBack) & " delta=" & delta
            End If
        Else
            LogMsg LOG_LEVEL_DEBUG, "HMITag", "write_verify_non_numeric tag=" & S(tag) & " wrote=" & S(val) & " read=" & S(readBack)
        End If
    End If
    On Error GoTo 0
End Sub

' ─────────────────────────────────────────────────────────────────────
' Aggregate one period (writes Aborted/Halted/Running only)
' ─────────────────────────────────────────────────────────────────────
Private Sub AggregatePeriod(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal startDt, ByVal endDt)
    Dim fRun, fHalt, fAbort
    Dim pctRun, pctHalt, pctAbort
    Dim tagPct, tagPie, pieVal
    Dim totalPct
    Dim inputValid
    Dim usedFallbackPie

    LogMsg LOG_LEVEL_INFO, "Aggregate", machineId & " " & periodKey & "  " & TS3(startDt) & " -> " & TS3(endDt)

    usedFallbackPie = False

    Call GetCombinedFractions(cn, runtimeCatalog, machineId, periodKey, startDt, endDt, fRun, fHalt, fAbort, inputValid)
    If Not inputValid Then
        LogMsg LOG_LEVEL_WARN, "Aggregate", _
               "aggregate_input_invalid machineId=" & machineId & _
               " periodKey=" & periodKey & _
               " runtimeCatalog=" & runtimeCatalog & _
               " aggExpr=none_selected"

        ' Invalid/no-data path: keep main percentage tags at 0 but force a
        ' deterministic pie split so UI can distinguish this from true idle.
        pctRun = 0: pctHalt = 0: pctAbort = 0
        totalPct = 0
        usedFallbackPie = True
    Else
        pctRun   = Round(CDbl(fRun)   * 100, 2)
        pctHalt  = Round(CDbl(fHalt)  * 100, 2)
        pctAbort = Round(CDbl(fAbort) * 100, 2)
        totalPct = Round(pctRun + pctHalt + pctAbort, 2)
    End If

    LogMsg LOG_LEVEL_INFO, "Aggregate", _
           "computed machine=" & machineId & _
           " period=" & periodKey & _
           " input_valid=" & CStr(inputValid) & _
           " pctRun=" & pctRun & _
           " pctHalt=" & pctHalt & _
           " pctAbort=" & pctAbort & _
           " totalPct=" & totalPct & _
           " fallback_render_path=" & CStr(usedFallbackPie)

    ' main % (can be 0)
    tagPct = OutTagName(machineId, periodKey, "Running", False): WriteTag tagPct, pctRun
    tagPct = OutTagName(machineId, periodKey, "Halted",  False): WriteTag tagPct, pctHalt
    tagPct = OutTagName(machineId, periodKey, "Aborted", False): WriteTag tagPct, pctAbort

    ' pie chart output strategy:
    ' 1) invalid/no-data input -> deterministic fallback split for rendering.
    ' 2) valid input -> preserve current true-idle behavior: all pie outputs
    '    remain 0 when totalPct=0 (operationally required for idle windows).
    If usedFallbackPie Then
        tagPie = OutTagName(machineId, periodKey, "Running", True): WriteTag tagPie, 34
        tagPie = OutTagName(machineId, periodKey, "Halted",  True): WriteTag tagPie, 33
        tagPie = OutTagName(machineId, periodKey, "Aborted", True): WriteTag tagPie, 33
        LogMsg LOG_LEVEL_WARN, "Aggregate", _
               "pie_fallback_applied machine=" & machineId & _
               " period=" & periodKey & _
               " input_valid=" & CStr(inputValid) & _
               " totalPct=" & totalPct & _
               " fallback_render_path=True"
    Else
        pieVal = pctRun:   If pieVal < 1 And totalPct > 0 Then pieVal = 1
        tagPie = OutTagName(machineId, periodKey, "Running", True): WriteTag tagPie, pieVal

        pieVal = pctHalt:  If pieVal < 1 And totalPct > 0 Then pieVal = 1
        tagPie = OutTagName(machineId, periodKey, "Halted", True): WriteTag tagPie, pieVal

        pieVal = pctAbort: If pieVal < 1 And totalPct > 0 Then pieVal = 1
        tagPie = OutTagName(machineId, periodKey, "Aborted", True): WriteTag tagPie, pieVal

        If totalPct = 0 Then
            LogMsg LOG_LEVEL_INFO, "Aggregate", _
                   "pie_idle_window_preserved machine=" & machineId & _
                   " period=" & periodKey & _
                   " input_valid=" & CStr(inputValid) & _
                   " totalPct=" & totalPct & _
                   " fallback_render_path=False"
        End If
    End If

    LogMsg LOG_LEVEL_INFO, "Aggregate", "writes_completed machine=" & machineId & " period=" & periodKey
End Sub

Private Function ShouldRun(ByVal lastRun, ByVal throttleMin)
    If IsEmpty(lastRun) Or IsNull(lastRun) Then
        ShouldRun = True
        Exit Function
    End If
    ShouldRun = (DateDiff("n", lastRun, Now()) >= throttleMin)
End Function

Private Function RunWindowWithGuard(ByVal cn, ByVal rtCat, ByVal machineId, ByVal periodKey, ByVal tFrom, ByVal tTo, ByRef windowElapsedMs)
    Dim windowTimer
    windowTimer = Timer
    AggregatePeriod cn, rtCat, machineId, periodKey, tFrom, tTo
    windowElapsedMs = ElapsedMs(windowTimer)
    RunWindowWithGuard = (windowElapsedMs > WINDOW_EXEC_WARN_THRESHOLD_MS)
End Function

' ─────────────────────────────────────────────────────────────────────
' Main Entry Point
' ─────────────────────────────────────────────────────────────────────
Public Sub MainStatusAggregation(ByVal machineId)
    Dim cn, rtCat, nowDT
    Dim tFrom, tTo
    Dim key, lastRun
    Dim due1D, due7D, due30D
    Dim startedAt, completedAt, startTimer, totalElapsedMs
    Dim guardReason, lockAcquired, canRun
    Dim skipLowerPriority, windowElapsedMs
    Dim validationState, validationReason, validationLookbackMinutes

    ' WinCC VBScript does not support label-based handlers (On Error GoTo FAIL).
    On Error Resume Next

    startedAt = Now
    startTimer = Timer
    guardReason = "completed"
    lockAcquired = False
    canRun = True
    skipLowerPriority = False
    Set cn = Nothing

    InitGlobals

    If Not TryAcquireMachineLock(machineId) Then
        LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", "Lock already active; skipping machineId: " & S(machineId)
        guardReason = "skip_lock_held"
        canRun = False
    End If
    If canRun Then lockAcquired = True

    If canRun Then
        If Not IsMaster() Then
            guardReason = "skip_not_master"
            canRun = False
        End If
    End If

    If canRun Then
        If Not G_MACHINES.Exists(machineId) Then
            LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", "Unknown machineId: " & S(machineId)
            guardReason = "skip_unknown_machine"
            canRun = False
        End If
    End If

    If canRun Then
        nowDT = Now()
        due1D = False : due7D = False : due30D = False

        key = machineId & "|1D"
        lastRun = Empty
        If g_lastRunByMachinePeriod.Exists(key) Then lastRun = g_lastRunByMachinePeriod(key)
        due1D = ShouldRun(lastRun, THROTTLE_1D_MINUTES)

        key = machineId & "|7D"
        lastRun = Empty
        If g_lastRunByMachinePeriod.Exists(key) Then lastRun = g_lastRunByMachinePeriod(key)
        due7D = ShouldRun(lastRun, THROTTLE_7D_MINUTES)

        key = machineId & "|30D"
        lastRun = Empty
        If g_lastRunByMachinePeriod.Exists(key) Then lastRun = g_lastRunByMachinePeriod(key)
        due30D = ShouldRun(lastRun, THROTTLE_30D_MINUTES)

        ' Important: avoid SQL connection work when nothing is due.
        If Not (due1D Or due7D Or due30D) Then
            guardReason = "skip_throttle"
            canRun = False
        End If
    End If

    If canRun Then
        Set cn = OpenConnection()
        If cn Is Nothing Then
            LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", "ERROR: cannot open SQL connection"
            guardReason = "fail_open_connection"
            canRun = False
        ElseIf cn.State <> adStateOpen Then
            LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", "ERROR: cannot open SQL connection"
            guardReason = "fail_connection_state"
            canRun = False
        End If
    End If

    If canRun Then rtCat = GetRuntimeCatalog()
    If canRun Then
        validationLookbackMinutes = ResolveStartupValidationLookbackMinutes(due1D, due7D, due30D)
        validationState = VALIDATE_STATE_HARD_FAIL
        validationReason = ""
        If Not ValidateMachineCountTags(cn, rtCat, machineId, validationLookbackMinutes, validationState, validationReason) Then
            guardReason = "skip_startup_validation_failed"
            canRun = False
        ElseIf validationState = VALIDATE_STATE_NO_RECENT Then
            LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", _
                   "startup_validation_warning machine=" & machineId & _
                   " reason=" & validationReason & _
                   " lookback_min=" & validationLookbackMinutes & _
                   " action=proceed_with_aggregation"
        End If
    End If
    If canRun Then
        LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", _
               "aggregation_start machine=" & machineId & _
               " cn_state=" & CStr(cn.State) & _
               " runtime_catalog=" & rtCat
    End If

    ' Execute due periods by priority:
    ' 1D -> 7D -> 30D
    ' If any period exceeds threshold, skip lower-priority windows for responsiveness.
    If canRun Then
        If due1D Then
            tTo = nowDT: tFrom = DateAdd("d", -1, tTo)
            Err.Clear
            skipLowerPriority = RunWindowWithGuard(cn, rtCat, machineId, "1D", tFrom, tTo, windowElapsedMs)
            If Err.Number <> 0 Then
                LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", _
                       "window_error machine=" & machineId & _
                       " period=1D err=" & S(Err.Description)
                Err.Clear
            Else
                If skipLowerPriority Then
                    LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", _
                           "window_slow machine=" & machineId & _
                           " period=1D elapsed_ms=" & windowElapsedMs & _
                           " threshold_ms=" & WINDOW_EXEC_WARN_THRESHOLD_MS
                End If
                g_lastRunByMachinePeriod(machineId & "|1D") = nowDT
                guardReason = "ran_1D"
            End If
        End If

        If due7D And (Not skipLowerPriority) Then
            tTo = nowDT: tFrom = DateAdd("d", -7, tTo)
            Err.Clear
            skipLowerPriority = RunWindowWithGuard(cn, rtCat, machineId, "7D", tFrom, tTo, windowElapsedMs)
            If Err.Number <> 0 Then
                LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", _
                       "window_error machine=" & machineId & _
                       " period=7D err=" & S(Err.Description)
                Err.Clear
            Else
                If skipLowerPriority Then
                    LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", _
                           "window_slow machine=" & machineId & _
                           " period=7D elapsed_ms=" & windowElapsedMs & _
                           " threshold_ms=" & WINDOW_EXEC_WARN_THRESHOLD_MS
                End If
                g_lastRunByMachinePeriod(machineId & "|7D") = nowDT
                guardReason = "ran_7D"
            End If
        End If

        If due30D And (Not skipLowerPriority) Then
            tTo = nowDT: tFrom = DateAdd("d", -30, tTo)
            Err.Clear
            skipLowerPriority = RunWindowWithGuard(cn, rtCat, machineId, "30D", tFrom, tTo, windowElapsedMs)
            If Err.Number <> 0 Then
                LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", _
                       "window_error machine=" & machineId & _
                       " period=30D err=" & S(Err.Description)
                Err.Clear
            Else
                If skipLowerPriority Then
                    LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", _
                           "window_slow machine=" & machineId & _
                           " period=30D elapsed_ms=" & windowElapsedMs & _
                           " threshold_ms=" & WINDOW_EXEC_WARN_THRESHOLD_MS
                End If
                g_lastRunByMachinePeriod(machineId & "|30D") = nowDT
                guardReason = "ran_30D"
            End If
        End If
    End If

CLEANUP:
    If IsObject(cn) Then CloseConn cn
    If lockAcquired Then ReleaseMachineLock machineId

    If Err.Number <> 0 Then
        guardReason = "fail_runtime_error"
        LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", _
               "Runtime error number=" & CStr(Err.Number) & _
               " description=" & S(Err.Description)
        Err.Clear
    End If

    On Error GoTo 0

    completedAt = Now
    totalElapsedMs = ElapsedMs(startTimer)
    LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", _
           "runtime_guard machine=" & machineId & _
           " started=" & TS3(startedAt) & _
           " ended=" & TS3(completedAt) & _
           " elapsed_ms=" & totalElapsedMs & _
           " reason=" & guardReason
    Exit Sub
End Sub

Private Function TryAcquireMachineLock(ByVal machineId)
    If g_machineLocks.Exists(machineId) Then
        TryAcquireMachineLock = False
    Else
        g_machineLocks.Add machineId, True
        TryAcquireMachineLock = True
    End If
End Function

Private Sub ReleaseMachineLock(ByVal machineId)
    If g_machineLocks.Exists(machineId) Then
        g_machineLocks.Remove machineId
    End If
End Sub
