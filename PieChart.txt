' ======================================================================
' WinCC Global Script: Status % Aggregation (simplified + robust logs)
'
' Output tags (unchanged):
'   <Machine>_<Period>_Machine_<Status>_Percentage
'   <Machine>_<Period>_Machine_<Status>_Percentage_PieChart
'
' Periods: 1D / 7D / 30D
' ======================================================================

Option Explicit

' ---- SQL / runtime ----
Const DEFAULT_PROVIDER = "MSOLEDBSQL"
Const FALLBACK_PROVIDER = "SQLNCLI11"
Const SQL_INSTANCE = ".\WinCC"
Const INITIAL_CATALOG = "master"
Const SQL_TIMEOUT_SECONDS = 45
Const MAX_RETRIES = 2
Const RETRY_BACKOFF_MS = 250

Const adCmdText = 1
Const adStateOpen = 1

' ---- logging ----
Const LOG_LEVEL_ERROR = 1
Const LOG_LEVEL_WARN  = 2
Const LOG_LEVEL_INFO  = 3
Const LOG_LEVEL_DEBUG = 4
Const CURRENT_LOG_LEVEL = 3

' ---- behavior ----
Const RENDER_ZERO_TOTAL_AS_FALLBACK_PIE = True
Const THROTTLE_1D_MINUTES  = 30
Const THROTTLE_7D_MINUTES  = 360
Const THROTTLE_30D_MINUTES = 1440
Const WRITE_VERIFY_TOLERANCE = 0.01

' ---- count-tag naming ----
Const COUNT_TAG_PREFIX_JOINER = "\"
Const COUNT_TAG_CORE_TEMPLATE = "{MACHINE_ID}_ProgramStatus_{STATUS_NAME}"
Const COUNT_TAG_SUFFIX = "_Count"

' ---- project aggregation preference ----
Const PROJECT_BATCH_AGG_LIST_DEFAULT = "AVG(RealValue)|AVG(Value)|AVG(ValueScaled)"
Const PROJECT_BATCH_AGG_LIST_WINCC   = "AVG(Value)|AVG(RealValue)|AVG(ValueScaled)"

Dim g_inited
Dim g_lastRunByMachinePeriod
Dim g_machineLocks
Dim G_MACHINES

Private Sub InitGlobals()
    If g_inited Then Exit Sub

    Set g_lastRunByMachinePeriod = CreateObject("Scripting.Dictionary")
    Set g_machineLocks = CreateObject("Scripting.Dictionary")
    Set G_MACHINES = CreateObject("Scripting.Dictionary")

    G_MACHINES.Add "MNA_0269", "DMG_Mori1_MNA_0269"
    G_MACHINES.Add "MNA_0270", "DMG_Mori2_MNA_0270"
    G_MACHINES.Add "MNA_0332", "DMG_Mori3_MNA_0332"
    G_MACHINES.Add "MNA_0164", "Wheelabrator_MNP_0164"
    G_MACHINES.Add "MNA_0085", "Droop_Rein_1_MNA_0085_LM33H"
    G_MACHINES.Add "MNA_0087", "Droop_Rein_4_MNA_0087_LM33F"
    G_MACHINES.Add "MNA_0235", "WFL_M120_MNA_0235"
    G_MACHINES.Add "MNA_0237", "WFL_M50_MNA_0237"
    G_MACHINES.Add "MNA_0411", "Mazak_MNA_0411"

    g_inited = True
End Sub

Private Function S(ByVal v)
    On Error Resume Next
    If IsNull(v) Or IsEmpty(v) Or IsObject(v) Then
        S = ""
    Else
        S = CStr(v)
        If Err.Number <> 0 Then S = "": Err.Clear
    End If
    On Error GoTo 0
End Function

Private Sub LogMsg(ByVal level, ByVal src, ByVal msg)
    If level > CURRENT_LOG_LEVEL Then Exit Sub
    On Error Resume Next
    HMIRuntime.Trace FormatDateTime(Now(), vbShortDate) & " " & _
                     FormatDateTime(Now(), vbLongTime) & " [PieChart/" & src & "] " & S(msg) & vbCrLf
    On Error GoTo 0
End Sub

Private Sub SleepMs(ByVal ms)
    On Error Resume Next
    HMIRuntime.Sleep CLng(ms)
    On Error GoTo 0
End Sub

Private Function TS3(ByVal dt)
    TS3 = Year(dt) & "-" & Right("0" & Month(dt),2) & "-" & Right("0" & Day(dt),2) & _
          " " & Right("0" & Hour(dt),2) & ":" & Right("0" & Minute(dt),2) & ":" & Right("0" & Second(dt),2) & ".000"
End Function

Private Function ElapsedMs(ByVal timerStart)
    Dim t
    t = Timer
    If t < timerStart Then t = t + 86400
    ElapsedMs = CLng((t - timerStart) * 1000)
End Function

Private Function IsMaster()
    On Error Resume Next
    IsMaster = (CLng(HMIRuntime.Tags("@RM_MASTER").Read) = 1)
    If Err.Number <> 0 Then IsMaster = True: Err.Clear
    On Error GoTo 0
End Function

Private Function OpenConnection()
    On Error Resume Next
    Dim cn

    Set cn = CreateObject("ADODB.Connection")
    cn.Open "Provider=" & DEFAULT_PROVIDER & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "OpenConnection", "provider=" & DEFAULT_PROVIDER & " failed err=" & S(Err.Description)
        Err.Clear
        Set cn = CreateObject("ADODB.Connection")
        cn.Open "Provider=" & FALLBACK_PROVIDER & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"
    End If

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_ERROR, "OpenConnection", "failed err=" & S(Err.Description)
        Err.Clear
        Set cn = Nothing
    End If

    Set OpenConnection = cn
    On Error GoTo 0
End Function

Private Sub CloseConn(ByVal cn)
    On Error Resume Next
    If IsObject(cn) Then
        If Not (cn Is Nothing) Then
            If cn.State = adStateOpen Then cn.Close
        End If
    End If
    Set cn = Nothing
    On Error GoTo 0
End Sub

Private Function GetRuntimeCatalog()
    On Error Resume Next
    Dim cat
    cat = HMIRuntime.Tags("@DatasourceNameRT").Read
    If Err.Number <> 0 Or IsNull(cat) Or IsEmpty(cat) Or Len(CStr(cat)) = 0 Then
        Err.Clear
        GetRuntimeCatalog = "WinCC"
    Else
        GetRuntimeCatalog = CStr(cat)
    End If
    On Error GoTo 0
End Function

Private Function GetAggExprList(ByVal runtimeCatalog)
    If UCase(CStr(runtimeCatalog)) = "WINCC" Then
        GetAggExprList = Split(PROJECT_BATCH_AGG_LIST_WINCC, "|")
    Else
        GetAggExprList = Split(PROJECT_BATCH_AGG_LIST_DEFAULT, "|")
    End If
End Function

Private Function BuildCountTagFullName(ByVal machineId, ByVal statusName)
    Dim core
    core = Replace(COUNT_TAG_CORE_TEMPLATE, "{MACHINE_ID}", machineId)
    core = Replace(core, "{STATUS_NAME}", statusName)
    BuildCountTagFullName = G_MACHINES(machineId) & COUNT_TAG_PREFIX_JOINER & core & COUNT_TAG_SUFFIX
End Function

Private Function BuildBaseTagFullName(ByVal machineId, ByVal statusName)
    Dim core
    core = Replace(COUNT_TAG_CORE_TEMPLATE, "{MACHINE_ID}", machineId)
    core = Replace(core, "{STATUS_NAME}", statusName)
    BuildBaseTagFullName = G_MACHINES(machineId) & COUNT_TAG_PREFIX_JOINER & core
End Function

Private Function BuildCpTagStatisticExLiteral(ByVal runtimeCatalog, ByVal tagFullName, ByVal tFrom, ByVal tTo, ByVal aggExpr)
    Dim safeCatalog, safeTag, safeAgg
    safeCatalog = Replace(runtimeCatalog, "'", "''")
    safeTag = Replace(tagFullName, "'", "''")
    safeAgg = Replace(aggExpr, "'", "''")

    BuildCpTagStatisticExLiteral = "cp_TagStatisticEx '" & safeCatalog & "'," & _
                                   "'TAG:R,''" & safeTag & "'',''" & TS3(tFrom) & "'',''" & TS3(tTo) & "'''," & _
                                   "'" & safeAgg & "'"
End Function

Private Function ExecAgg(ByVal cn, ByVal runtimeCatalog, ByVal tagFull, ByVal tFrom, ByVal tTo, ByVal aggExpr, ByRef fractionOut, ByRef errOut)
    On Error Resume Next
    Dim i, cmd, rs, sqlText, gotNumeric

    fractionOut = 0
    errOut = ""
    ExecAgg = False

    For i = 1 To MAX_RETRIES
        Err.Clear
        sqlText = BuildCpTagStatisticExLiteral(runtimeCatalog, tagFull, tFrom, tTo, aggExpr)

        Set cmd = CreateObject("ADODB.Command")
        cmd.ActiveConnection = cn
        cmd.CommandType = adCmdText
        cmd.CommandTimeout = SQL_TIMEOUT_SECONDS
        cmd.CommandText = sqlText

        Set rs = cmd.Execute
        If Err.Number = 0 Then
            gotNumeric = False
            If Not (rs Is Nothing) Then
                If Not rs.EOF Then
                    If rs.Fields.Count > 0 And Not IsNull(rs.Fields(0).Value) Then
                        fractionOut = CDbl(rs.Fields(0).Value)
                        gotNumeric = True
                    End If
                End If
                rs.Close
            End If

            Set rs = Nothing
            Set cmd = Nothing
            If gotNumeric Then
                ExecAgg = True
            Else
                ExecAgg = True
                fractionOut = 0
            End If
            Exit Function
        End If

        errOut = S(Err.Description)
        Set rs = Nothing
        Set cmd = Nothing

        If i < MAX_RETRIES Then SleepMs RETRY_BACKOFF_MS * i
    Next

    On Error GoTo 0
End Function

Private Function ReadCountFraction(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal statusName, ByVal tFrom, ByVal tTo)
    Dim countTag, baseTag, aggs, idx, frac, errText

    ReadCountFraction = 0
    countTag = BuildCountTagFullName(machineId, statusName)
    baseTag = BuildBaseTagFullName(machineId, statusName)
    aggs = GetAggExprList(runtimeCatalog)

    For idx = 0 To UBound(aggs)
        If ExecAgg(cn, runtimeCatalog, countTag, tFrom, tTo, CStr(aggs(idx)), frac, errText) Then
            ReadCountFraction = frac
            If CURRENT_LOG_LEVEL >= LOG_LEVEL_DEBUG Then
                LogMsg LOG_LEVEL_DEBUG, "ReadCountFraction", _
                       "machine=" & machineId & " status=" & statusName & " tag=" & countTag & " agg=" & CStr(aggs(idx)) & " frac=" & frac
            End If
            Exit Function
        End If
    Next

    For idx = 0 To UBound(aggs)
        If ExecAgg(cn, runtimeCatalog, baseTag, tFrom, tTo, CStr(aggs(idx)), frac, errText) Then
            ReadCountFraction = frac
            LogMsg LOG_LEVEL_INFO, "ReadCountFraction", _
                   "fallback_to_plain_tag machine=" & machineId & " status=" & statusName & " tag=" & baseTag & " agg=" & CStr(aggs(idx)) & " frac=" & frac
            Exit Function
        End If
    Next

    LogMsg LOG_LEVEL_WARN, "ReadCountFraction", _
           "machine=" & machineId & " status=" & statusName & " count_tag=" & countTag & " plain_tag=" & baseTag & " err=" & errText
End Function

Private Function OutTagName(ByVal machineId, ByVal periodKey, ByVal statusDispName, ByVal pieChart)
    OutTagName = machineId & "_" & periodKey & "_Machine_" & statusDispName & "_Percentage"
    If pieChart Then OutTagName = OutTagName & "_PieChart"
End Function

Private Sub WriteTag(ByVal tag, ByVal val)
    On Error Resume Next
    Dim t, beforeVal, afterVal

    Set t = HMIRuntime.Tags(tag)
    beforeVal = t.Read
    t.Write CDbl(val)
    afterVal = t.Read

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_ERROR, "WriteTag", "tag=" & tag & " val=" & CDbl(val) & " err=" & S(Err.Description)
        Err.Clear
    ElseIf Abs(CDbl(afterVal) - CDbl(val)) > WRITE_VERIFY_TOLERANCE Then
        LogMsg LOG_LEVEL_WARN, "WriteTag", "tag=" & tag & " write_verify_failed wanted=" & CDbl(val) & " got=" & CDbl(afterVal)
    ElseIf CURRENT_LOG_LEVEL >= LOG_LEVEL_DEBUG Then
        LogMsg LOG_LEVEL_DEBUG, "WriteTag", "tag=" & tag & " before=" & CDbl(beforeVal) & " after=" & CDbl(afterVal)
    End If

    Set t = Nothing
    On Error GoTo 0
End Sub

Private Sub NormalizeFractionsToPct(ByVal runFrac, ByVal haltFrac, ByVal abortFrac, ByRef pctRun, ByRef pctHalt, ByRef pctAbort)
    Dim safeRun, safeHalt, safeAbort, sumFrac
    Dim rawRun, rawHalt, rawAbort
    Dim floorRun, floorHalt, floorAbort
    Dim remRun, remHalt, remAbort
    Dim remaining

    safeRun = CDbl(runFrac)
    safeHalt = CDbl(haltFrac)
    safeAbort = CDbl(abortFrac)

    If safeRun < 0 Then safeRun = 0
    If safeHalt < 0 Then safeHalt = 0
    If safeAbort < 0 Then safeAbort = 0

    sumFrac = safeRun + safeHalt + safeAbort
    If sumFrac <= 0 Then
        pctRun = 0
        pctHalt = 0
        pctAbort = 0
        Exit Sub
    End If

    rawRun = (safeRun / sumFrac) * 100
    rawHalt = (safeHalt / sumFrac) * 100
    rawAbort = (safeAbort / sumFrac) * 100

    floorRun = Int(rawRun)
    floorHalt = Int(rawHalt)
    floorAbort = Int(rawAbort)

    remRun = rawRun - floorRun
    remHalt = rawHalt - floorHalt
    remAbort = rawAbort - floorAbort

    remaining = 100 - (floorRun + floorHalt + floorAbort)

    pctRun = floorRun
    pctHalt = floorHalt
    pctAbort = floorAbort

    Do While remaining > 0
        If remRun >= remHalt And remRun >= remAbort Then
            pctRun = pctRun + 1
            remRun = -1
        ElseIf remHalt >= remRun And remHalt >= remAbort Then
            pctHalt = pctHalt + 1
            remHalt = -1
        Else
            pctAbort = pctAbort + 1
            remAbort = -1
        End If
        remaining = remaining - 1
    Loop
End Sub

Private Sub AggregatePeriod(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal tFrom, ByVal tTo)
    Dim runFrac, haltFrac, abortFrac, waitFrac, intFrac
    Dim pctRun, pctHalt, pctAbort, totalPct

    runFrac = ReadCountFraction(cn, runtimeCatalog, machineId, "Running", tFrom, tTo)
    haltFrac = ReadCountFraction(cn, runtimeCatalog, machineId, "Halted", tFrom, tTo)
    abortFrac = ReadCountFraction(cn, runtimeCatalog, machineId, "Aborted", tFrom, tTo)
    waitFrac = ReadCountFraction(cn, runtimeCatalog, machineId, "Waiting", tFrom, tTo)
    intFrac = ReadCountFraction(cn, runtimeCatalog, machineId, "Interrupted", tFrom, tTo)

    abortFrac = abortFrac + waitFrac
    haltFrac = haltFrac + intFrac

    NormalizeFractionsToPct runFrac, haltFrac, abortFrac, pctRun, pctHalt, pctAbort

    WriteTag OutTagName(machineId, periodKey, "Running", False), pctRun
    WriteTag OutTagName(machineId, periodKey, "Halted", False),  pctHalt
    WriteTag OutTagName(machineId, periodKey, "Aborted", False), pctAbort

    totalPct = pctRun + pctHalt + pctAbort
    If totalPct = 0 And RENDER_ZERO_TOTAL_AS_FALLBACK_PIE Then
        WriteTag OutTagName(machineId, periodKey, "Running", True), 34
        WriteTag OutTagName(machineId, periodKey, "Halted", True), 33
        WriteTag OutTagName(machineId, periodKey, "Aborted", True), 33
    Else
        WriteTag OutTagName(machineId, periodKey, "Running", True), pctRun
        WriteTag OutTagName(machineId, periodKey, "Halted", True),  pctHalt
        WriteTag OutTagName(machineId, periodKey, "Aborted", True), pctAbort
    End If

    LogMsg LOG_LEVEL_INFO, "Aggregate", _
           "machine=" & machineId & " period=" & periodKey & _
           " run=" & pctRun & " halt=" & pctHalt & " abort=" & pctAbort & " total=" & totalPct
End Sub

Private Function ShouldRun(ByVal machineId, ByVal periodKey, ByVal throttleMin)
    Dim key, lastRun
    key = machineId & "|" & periodKey

    If Not g_lastRunByMachinePeriod.Exists(key) Then
        ShouldRun = True
        Exit Function
    End If

    lastRun = g_lastRunByMachinePeriod(key)
    ShouldRun = (DateDiff("n", lastRun, Now()) >= throttleMin)
End Function

Private Sub MarkRun(ByVal machineId, ByVal periodKey)
    g_lastRunByMachinePeriod(machineId & "|" & periodKey) = Now()
End Sub

Private Function TryAcquireMachineLock(ByVal machineId)
    If g_machineLocks.Exists(machineId) Then
        TryAcquireMachineLock = False
    Else
        g_machineLocks.Add machineId, True
        TryAcquireMachineLock = True
    End If
End Function

Private Sub ReleaseMachineLock(ByVal machineId)
    If g_machineLocks.Exists(machineId) Then g_machineLocks.Remove machineId
End Sub

Public Sub MainStatusAggregation(ByVal machineId)
    On Error Resume Next

    Dim cn, rtCat, nowDT
    Dim startTimer, reason
    Dim due1D, due7D, due30D
    Dim lockAcquired, canRun

    InitGlobals
    startTimer = Timer
    reason = "completed"
    lockAcquired = False
    canRun = True

    LogMsg LOG_LEVEL_INFO, "Main", "start machine=" & machineId

    If Not TryAcquireMachineLock(machineId) Then
        LogMsg LOG_LEVEL_WARN, "Main", "skip reason=lock machine=" & machineId
        Exit Sub
    End If
    lockAcquired = True

    If canRun And (Not IsMaster()) Then
        reason = "skip_not_master"
        LogMsg LOG_LEVEL_INFO, "Main", "skip reason=" & reason & " machine=" & machineId
        canRun = False
    End If

    If canRun And (Not G_MACHINES.Exists(machineId)) Then
        reason = "skip_unknown_machine"
        LogMsg LOG_LEVEL_WARN, "Main", "skip reason=" & reason & " machine=" & machineId
        canRun = False
    End If

    If canRun Then
        due1D = ShouldRun(machineId, "1D", THROTTLE_1D_MINUTES)
        due7D = ShouldRun(machineId, "7D", THROTTLE_7D_MINUTES)
        due30D = ShouldRun(machineId, "30D", THROTTLE_30D_MINUTES)
    End If

    If canRun And (Not (due1D Or due7D Or due30D)) Then
        reason = "skip_throttle"
        LogMsg LOG_LEVEL_INFO, "Main", "skip reason=" & reason & " machine=" & machineId
        canRun = False
    End If

    If canRun Then Set cn = OpenConnection()
    If canRun And (cn Is Nothing) Then
        reason = "fail_open_connection"
        LogMsg LOG_LEVEL_ERROR, "Main", "fail reason=" & reason & " machine=" & machineId
        canRun = False
    End If

    If canRun Then
        rtCat = GetRuntimeCatalog()
        nowDT = Now()
    End If

    If canRun And due1D Then
        AggregatePeriod cn, rtCat, machineId, "1D", DateAdd("d", -1, nowDT), nowDT
        MarkRun machineId, "1D"
    End If

    If canRun And due7D Then
        AggregatePeriod cn, rtCat, machineId, "7D", DateAdd("d", -7, nowDT), nowDT
        MarkRun machineId, "7D"
    End If

    If canRun And due30D Then
        AggregatePeriod cn, rtCat, machineId, "30D", DateAdd("d", -30, nowDT), nowDT
        MarkRun machineId, "30D"
    End If

CLEANUP:
    If Err.Number <> 0 Then
        reason = "runtime_error"
        LogMsg LOG_LEVEL_ERROR, "Main", "machine=" & machineId & " err=" & S(Err.Description)
        Err.Clear
    End If

    If IsObject(cn) Then CloseConn cn
    If lockAcquired Then ReleaseMachineLock machineId

    LogMsg LOG_LEVEL_WARN, "Main", "end machine=" & machineId & " reason=" & reason & " elapsed_ms=" & ElapsedMs(startTimer)
    On Error GoTo 0
End Sub
