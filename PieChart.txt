' ======================================================================
'  WinCC Global Script: Status % Aggregation (FINAL)
'
'  - Uses ONLY *_Count tags (binary 0/1 archived every 500ms)
'  - Reads 5 base statuses:
'       Aborted, Halted, Running, Waiting, Interrupted
'  - Combines into 3 outputs (names UNCHANGED):
'       Aborted  = Aborted + Waiting
'       Halted   = Halted  + Interrupted
'       Running  = Running
'
'  OUTPUT TAGS (unchanged):
'    <Machine>_<Period>_Machine_<Status>_Percentage              (can be 0)
'    <Machine>_<Period>_Machine_<Status>_Percentage_PieChart     (min 1)
'
'  Periods: 1D / 7D / 30D
' ======================================================================

Option Explicit

' ─────────────────────────────────────────────────────────────────────
' Configuration
' ─────────────────────────────────────────────────────────────────────
Const DEFAULT_PROVIDER       = "MSOLEDBSQL"
Const FALLBACK_PROVIDER      = "SQLNCLI11"
Const SQL_INSTANCE           = ".\WinCC"
Const INITIAL_CATALOG        = "master"
Const SQL_TIMEOUT_SECONDS    = 45
Const MAX_RETRY_ATTEMPTS     = 2
Const RETRY_BACKOFF_MS       = 250
Const RETRY_BACKOFF_TIMEOUT_MS = 1250

Const THROTTLE_1D_MINUTES    = 30    ' every 30 minutes
Const THROTTLE_7D_MINUTES    = 360   ' every 6 hours
Const THROTTLE_30D_MINUTES   = 1440  ' every 24 hours

Const adCmdText              = 1
Const adStateOpen            = 1

' ─────────────────────────────────────────────────────────────────────
' Log Levels
' ─────────────────────────────────────────────────────────────────────
Const LOG_LEVEL_ERROR        = 1
Const LOG_LEVEL_WARN         = 2
Const LOG_LEVEL_INFO         = 3
Const LOG_LEVEL_DEBUG        = 4

' IMPORTANT: must be literal in WinCC VBScript
Const CURRENT_LOG_LEVEL      = 4   ' DEBUG
Const WRITE_VERIFY_TOLERANCE = 0.01

' ─────────────────────────────────────────────────────────────────────
' Runtime state
' ─────────────────────────────────────────────────────────────────────
Dim g_lastRunByMachinePeriod
Dim g_machineLocks
Dim G_MACHINES
Dim g_inited

' ─────────────────────────────────────────────────────────────────────
' Redundancy helper
' ─────────────────────────────────────────────────────────────────────
Private Function IsMaster()
    On Error Resume Next
    IsMaster = (CLng(HMIRuntime.Tags("@RM_MASTER").Read) = 1)
    If Err.Number <> 0 Then
        ' Fail-open for standalone / tag not present
        IsMaster = True
        Err.Clear
    End If
    On Error GoTo 0
End Function

' ─────────────────────────────────────────────────────────────────────
' Init dictionaries
' ─────────────────────────────────────────────────────────────────────
Private Sub InitGlobals()
    If g_inited Then Exit Sub

    Set g_lastRunByMachinePeriod = CreateObject("Scripting.Dictionary")
    Set g_machineLocks = CreateObject("Scripting.Dictionary")

    Set G_MACHINES = CreateObject("Scripting.Dictionary")
    ' MachineId → Tag-logging prefix (left of backslash)
    G_MACHINES.Add "MNA_0269", "DMG_Mori1_MNA_0269"
    G_MACHINES.Add "MNA_0270", "DMG_Mori2_MNA_0270"
    G_MACHINES.Add "MNA_0332", "DMG_Mori3_MNA_0332"

    ' Wheelabrator
    G_MACHINES.Add "MNA_0164", "Wheelabrator_MNP_0164"

    ' Droop & Rein
    G_MACHINES.Add "MNA_0085", "Droop_Rein_1_MNA_0085_LM33H"
    G_MACHINES.Add "MNA_0087", "Droop_Rein_4_MNA_0087_LM33F"

    g_inited = True
End Sub

' ─────────────────────────────────────────────────────────────────────
' Logging (Null-safe)
' ─────────────────────────────────────────────────────────────────────
Private Function S(ByVal v)
    On Error Resume Next
    If IsNull(v) Or IsEmpty(v) Or IsObject(v) Then
        S = ""
    Else
        S = CStr(v)
        If Err.Number <> 0 Then
            S = ""
            Err.Clear
        End If
    End If
    On Error GoTo 0
End Function

Private Sub LogMsg(ByVal level, ByVal src, ByVal msg)
    If level > CURRENT_LOG_LEVEL Then Exit Sub
    On Error Resume Next
    HMIRuntime.Trace _
        FormatDateTime(Now(), vbShortDate) & " " & _
        FormatDateTime(Now(), vbLongTime) & " [" & src & "] " & S(msg) & vbCrLf
    On Error GoTo 0
End Sub

Private Sub SleepMs(ByVal ms)
    On Error Resume Next
    HMIRuntime.Sleep ms
    On Error GoTo 0
End Sub

' ─────────────────────────────────────────────────────────────────────
' SQL helpers
' ─────────────────────────────────────────────────────────────────────
Private Function OpenConnection()
    On Error Resume Next
    Dim cn, prov
    prov = DEFAULT_PROVIDER

    Set cn = CreateObject("ADODB.Connection")
    cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "OpenConnection", "Switching provider to " & FALLBACK_PROVIDER & " (" & S(Err.Description) & ")"
        Err.Clear
        prov = FALLBACK_PROVIDER
        Set cn = CreateObject("ADODB.Connection")
        cn.Open "Provider=" & prov & ";Data Source=" & SQL_INSTANCE & ";Initial Catalog=" & INITIAL_CATALOG & ";Integrated Security=SSPI;"
    End If

    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_ERROR, "OpenConnection", "ERROR: " & S(Err.Description)
        Set cn = Nothing
        Err.Clear
    End If

    Set OpenConnection = cn
    On Error GoTo 0
End Function

Private Sub CloseConn(ByVal cn)
    On Error Resume Next
    If IsObject(cn) Then
        If Not (cn Is Nothing) Then
            If cn.State = adStateOpen Then cn.Close
        End If
    End If
    Set cn = Nothing
    On Error GoTo 0
End Sub

Private Function GetRuntimeCatalog()
    On Error Resume Next
    Dim cat
    cat = HMIRuntime.Tags("@DatasourceNameRT").Read
    If Err.Number <> 0 Or IsNull(cat) Or IsEmpty(cat) Then
        Err.Clear
        GetRuntimeCatalog = "WinCC"
    ElseIf Len(CStr(cat)) > 0 Then
        GetRuntimeCatalog = CStr(cat)
    Else
        GetRuntimeCatalog = "WinCC"
    End If
    On Error GoTo 0
End Function

Private Function TS3(ByVal dt)
    TS3 = Year(dt) & "-" & Right("0" & Month(dt),2) & "-" & Right("0" & Day(dt),2) & _
          " " & Right("0" & Hour(dt),2) & ":" & Right("0" & Minute(dt),2) & ":" & Right("0" & Second(dt),2) & ".000"
End Function

Private Function ElapsedMs(ByVal timerStart)
    Dim timerEnd
    timerEnd = Timer
    If timerEnd < timerStart Then timerEnd = timerEnd + 86400
    ElapsedMs = CLng((timerEnd - timerStart) * 1000)
End Function

Private Function IsTimeoutLikeError(ByVal errDesc)
    Dim lowered
    lowered = LCase(S(errDesc))
    IsTimeoutLikeError = (InStr(lowered, "timeout") > 0) Or _
                         (InStr(lowered, "time out") > 0) Or _
                         (InStr(lowered, "expired") > 0)
End Function

' ─────────────────────────────────────────────────────────────────────
' cp_TagStatisticEx (short retry, null-safe)
' ─────────────────────────────────────────────────────────────────────
Private Function ExecCpTagStatisticEx(ByVal cn, ByVal runtimeCatalog, ByVal tagFullName, ByVal tFrom, ByVal tTo, ByVal aggList, ByVal machineId, ByVal periodKey, ByVal statusName)
    Dim i, cmd, rs, sSql, safeTag, fieldValue
    Dim startTimer, startedAt, completedAt, elapsedMsValue
    Dim returnedRows, valueText, infoMsg, errDesc, backoffMs

    startedAt = Now
    startTimer = Timer
    returnedRows = 0
    valueText = ""

    safeTag = Replace(tagFullName, "'", "''")

    For i = 1 To MAX_RETRY_ATTEMPTS
        On Error Resume Next

        Set cmd = CreateObject("ADODB.Command")
        Set cmd.ActiveConnection = cn
        cmd.CommandTimeout = SQL_TIMEOUT_SECONDS
        cmd.CommandType = adCmdText

        sSql = "cp_TagStatisticEx '" & runtimeCatalog & "'," & _
               "'TAG:R,''" & safeTag & "'',''" & TS3(tFrom) & "'',''" & TS3(tTo) & "'''," & _
               "'" & aggList & "'"

        cmd.CommandText = sSql

        If CURRENT_LOG_LEVEL >= LOG_LEVEL_DEBUG Then
            LogMsg LOG_LEVEL_DEBUG, "cp_TagStatisticEx", "SQL=[" & sSql & "]"
        End If

        Set rs = cmd.Execute

        If Err.Number = 0 Then
            ExecCpTagStatisticEx = Empty

            If Not (rs Is Nothing) Then
                If Not rs.EOF Then
                    returnedRows = 1
                    If rs.Fields.Count >= 1 Then
                        fieldValue = rs.Fields(0).Value
                        If Not (IsNull(fieldValue) Or IsEmpty(fieldValue)) Then
                            ExecCpTagStatisticEx = fieldValue
                            valueText = CStr(fieldValue)
                        Else
                            valueText = "<null>"
                        End If
                    End If
                End If
                rs.Close
            End If

            Set rs = Nothing
            Set cmd = Nothing
            On Error GoTo 0

            completedAt = Now
            elapsedMsValue = ElapsedMs(startTimer)
            If valueText = "" Then valueText = "<empty>"
            infoMsg = "machine=" & machineId & _
                      " window=" & periodKey & _
                      " status=" & statusName & _
                      " sql_from=" & TS3(tFrom) & _
                      " sql_to=" & TS3(tTo) & _
                      " started=" & TS3(startedAt) & _
                      " ended=" & TS3(completedAt) & _
                      " elapsed_ms=" & elapsedMsValue & _
                      " success=true" & _
                      " rows=" & returnedRows & _
                      " value=" & valueText
            LogMsg LOG_LEVEL_INFO, "cp_TagStatisticEx", infoMsg
            Exit Function
        Else
            errDesc = S(Err.Description)
            LogMsg LOG_LEVEL_WARN, "cp_TagStatisticEx", "Attempt " & i & " failed: " & errDesc
            Err.Clear
            On Error GoTo 0
            If i < MAX_RETRY_ATTEMPTS Then
                backoffMs = RETRY_BACKOFF_MS
                If IsTimeoutLikeError(errDesc) Then backoffMs = RETRY_BACKOFF_TIMEOUT_MS
                SleepMs backoffMs
            End If
        End If
    Next

    ExecCpTagStatisticEx = Empty
    completedAt = Now
    elapsedMsValue = ElapsedMs(startTimer)
    infoMsg = "machine=" & machineId & _
              " window=" & periodKey & _
              " status=" & statusName & _
              " sql_from=" & TS3(tFrom) & _
              " sql_to=" & TS3(tTo) & _
              " started=" & TS3(startedAt) & _
              " ended=" & TS3(completedAt) & _
              " elapsed_ms=" & elapsedMsValue & _
              " success=false rows=0 value=<none>"
    LogMsg LOG_LEVEL_INFO, "cp_TagStatisticEx", infoMsg
End Function

Private Function CpAggregate(ByVal cn, ByVal runtimeCatalog, ByVal tagFullName, ByVal t1, ByVal t2, ByVal agg, ByVal machineId, ByVal periodKey, ByVal statusName)
    Dim res
    res = ExecCpTagStatisticEx(cn, runtimeCatalog, tagFullName, t1, t2, agg, machineId, periodKey, statusName)

    If IsNull(res) Or IsEmpty(res) Or (CStr(res) = "") Then
        CpAggregate = Empty
    ElseIf IsNumeric(res) Then
        CpAggregate = CDbl(res)
    Else
        CpAggregate = Empty
    End If
End Function

Private Function TryAvgFraction(ByVal cn, ByVal runtimeCatalog, ByVal tagFull, ByVal t1, ByVal t2, ByVal machineId, ByVal periodKey, ByVal statusName)
    Dim v

    v = CpAggregate(cn, runtimeCatalog, tagFull, t1, t2, "AVG(RealValue)", machineId, periodKey, statusName)
    If IsNumeric(v) Then If v >= 0 And v <= 1 Then TryAvgFraction = v : Exit Function

    v = CpAggregate(cn, runtimeCatalog, tagFull, t1, t2, "AVG(Value)", machineId, periodKey, statusName)
    If IsNumeric(v) Then If v >= 0 And v <= 1 Then TryAvgFraction = v : Exit Function

    v = CpAggregate(cn, runtimeCatalog, tagFull, t1, t2, "AVG(ValueScaled)", machineId, periodKey, statusName)
    If IsNumeric(v) Then If v >= 0 And v <= 1 Then TryAvgFraction = v : Exit Function

    TryAvgFraction = Empty
End Function

' ─────────────────────────────────────────────────────────────────────
' READ ONLY *_Count (single SQL roundtrip for all 5 statuses)
' ─────────────────────────────────────────────────────────────────────
Private Function BuildCountTagFullName(ByVal machineId, ByVal statusName)
    Dim basePrefix, baseCore
    basePrefix = G_MACHINES(machineId)
    baseCore   = machineId & "_ProgramStatus_" & statusName
    BuildCountTagFullName = basePrefix & "\" & baseCore & "_Count"
End Function

Private Function BuildCpTagStatisticExLiteral(ByVal runtimeCatalog, ByVal tagFullName, ByVal tFrom, ByVal tTo, ByVal aggExpr)
    Dim safeCatalog, safeTag
    safeCatalog = Replace(runtimeCatalog, "'", "''")
    safeTag = Replace(tagFullName, "'", "''")
    BuildCpTagStatisticExLiteral = "cp_TagStatisticEx '" & safeCatalog & "'," & _
                                   "'TAG:R,''" & safeTag & "'',''" & TS3(tFrom) & "'',''" & TS3(tTo) & "'''," & _
                                   "'" & aggExpr & "'"
End Function

Private Function CopyStatusFractions(ByVal srcDict)
    Dim dst
    Set dst = CreateObject("Scripting.Dictionary")
    dst.Add "Aborted", srcDict("Aborted")
    dst.Add "Halted", srcDict("Halted")
    dst.Add "Running", srcDict("Running")
    dst.Add "Waiting", srcDict("Waiting")
    dst.Add "Interrupted", srcDict("Interrupted")
    Set CopyStatusFractions = dst
End Function

Private Function SumStatusFractions(ByVal d)
    SumStatusFractions = CDbl(d("Aborted")) + CDbl(d("Halted")) + CDbl(d("Running")) + CDbl(d("Waiting")) + CDbl(d("Interrupted"))
End Function

Private Function ExecuteStatusFractionsBatch(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal t1, ByVal t2, ByVal aggExpr, ByRef hasNumericCount)
    Dim cmd, rs, sqlText, startedAt, completedAt, startTimer, elapsedMsValue, infoMsg
    Dim statuses, aliases, idx, statusName, tagFull, cpCall
    Dim outDict, fieldValue
    Dim valA, valH, valR, valW, valI

    startedAt = Now
    startTimer = Timer
    hasNumericCount = 0

    Set outDict = CreateObject("Scripting.Dictionary")
    outDict.Add "Aborted", 0
    outDict.Add "Halted", 0
    outDict.Add "Running", 0
    outDict.Add "Waiting", 0
    outDict.Add "Interrupted", 0

    statuses = Array("Aborted", "Halted", "Running", "Waiting", "Interrupted")
    aliases  = Array("a", "h", "r", "w", "i")

    sqlText = "DECLARE @Status TABLE (status_alias NVARCHAR(1), frac FLOAT);" & vbCrLf
    For idx = 0 To UBound(statuses)
        statusName = CStr(statuses(idx))
        tagFull = BuildCountTagFullName(machineId, statusName)
        cpCall = BuildCpTagStatisticExLiteral(runtimeCatalog, tagFull, t1, t2, aggExpr)
        sqlText = sqlText & "INSERT INTO @Status (frac) EXEC " & cpCall & ";" & vbCrLf
        sqlText = sqlText & "UPDATE @Status SET status_alias='" & CStr(aliases(idx)) & "' WHERE status_alias IS NULL;" & vbCrLf
    Next
    sqlText = sqlText & "SELECT " & _
              "MAX(CASE WHEN status_alias='a' THEN frac END) AS a," & _
              "MAX(CASE WHEN status_alias='h' THEN frac END) AS h," & _
              "MAX(CASE WHEN status_alias='r' THEN frac END) AS r," & _
              "MAX(CASE WHEN status_alias='w' THEN frac END) AS w," & _
              "MAX(CASE WHEN status_alias='i' THEN frac END) AS i " & _
              "FROM @Status;"

    On Error Resume Next
    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = cn
    cmd.CommandTimeout = SQL_TIMEOUT_SECONDS
    cmd.CommandType = adCmdText
    cmd.CommandText = sqlText

    If CURRENT_LOG_LEVEL >= LOG_LEVEL_DEBUG Then
        LogMsg LOG_LEVEL_DEBUG, "GetStatusFractionsBatch", "SQL=[" & sqlText & "]"
    End If

    Set rs = cmd.Execute
    If Err.Number = 0 Then
        If Not (rs Is Nothing) Then
            If Not rs.EOF Then
                If rs.Fields.Count >= 5 Then
                    fieldValue = rs.Fields("a").Value: If IsNumeric(fieldValue) Then outDict("Aborted") = CDbl(fieldValue): hasNumericCount = hasNumericCount + 1
                    fieldValue = rs.Fields("h").Value: If IsNumeric(fieldValue) Then outDict("Halted") = CDbl(fieldValue): hasNumericCount = hasNumericCount + 1
                    fieldValue = rs.Fields("r").Value: If IsNumeric(fieldValue) Then outDict("Running") = CDbl(fieldValue): hasNumericCount = hasNumericCount + 1
                    fieldValue = rs.Fields("w").Value: If IsNumeric(fieldValue) Then outDict("Waiting") = CDbl(fieldValue): hasNumericCount = hasNumericCount + 1
                    fieldValue = rs.Fields("i").Value: If IsNumeric(fieldValue) Then outDict("Interrupted") = CDbl(fieldValue): hasNumericCount = hasNumericCount + 1
                End If
            End If
            rs.Close
        End If
    Else
        LogMsg LOG_LEVEL_WARN, "GetStatusFractionsBatch", "Batch query failed; status fractions defaulted to 0. err=" & S(Err.Description)
        Err.Clear
    End If
    On Error GoTo 0

    Set rs = Nothing
    Set cmd = Nothing

    completedAt = Now
    elapsedMsValue = ElapsedMs(startTimer)
    valA = outDict("Aborted")
    valH = outDict("Halted")
    valR = outDict("Running")
    valW = outDict("Waiting")
    valI = outDict("Interrupted")
    infoMsg = "machine=" & machineId & _
              " window=" & periodKey & _
              " agg=" & aggExpr & _
              " sql_from=" & TS3(t1) & _
              " sql_to=" & TS3(t2) & _
              " started=" & TS3(startedAt) & _
              " ended=" & TS3(completedAt) & _
              " elapsed_ms=" & elapsedMsValue & _
              " numeric=" & hasNumericCount & _
              " a=" & valA & _
              " h=" & valH & _
              " r=" & valR & _
              " w=" & valW & _
              " i=" & valI
    LogMsg LOG_LEVEL_INFO, "GetStatusFractionsBatch", infoMsg

    Set ExecuteStatusFractionsBatch = outDict
End Function

Private Function GetStatusFractionsBatch(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal t1, ByVal t2)
    Dim aggs, idx, aggExpr
    Dim currentDict, bestDict
    Dim numericCount, bestNumericCount
    Dim currentSum, bestSum
    Dim selectedAgg

    aggs = Array("AVG(RealValue)", "AVG(Value)", "AVG(ValueScaled)")
    bestNumericCount = -1
    bestSum = -1
    selectedAgg = ""

    For idx = 0 To UBound(aggs)
        aggExpr = CStr(aggs(idx))
        Set currentDict = ExecuteStatusFractionsBatch(cn, runtimeCatalog, machineId, periodKey, t1, t2, aggExpr, numericCount)
        currentSum = SumStatusFractions(currentDict)

        If (bestDict Is Nothing) Then
            Set bestDict = CopyStatusFractions(currentDict)
            bestNumericCount = numericCount
            bestSum = currentSum
            selectedAgg = aggExpr
        ElseIf (numericCount > bestNumericCount) Or _
               (numericCount = bestNumericCount And currentSum > bestSum And bestSum <= 0) Then
            Set bestDict = CopyStatusFractions(currentDict)
            bestNumericCount = numericCount
            bestSum = currentSum
            selectedAgg = aggExpr
        End If

        If numericCount = 5 And currentSum > 0 Then Exit For
    Next

    LogMsg LOG_LEVEL_INFO, "GetStatusFractionsBatch", _
           "selected_agg machine=" & machineId & _
           " window=" & periodKey & _
           " agg=" & selectedAgg & _
           " numeric=" & bestNumericCount & _
           " sum=" & bestSum

    Set GetStatusFractionsBatch = bestDict
End Function

Private Function GetCountFraction(ByVal statusFractions, ByVal statusName)
    If (statusFractions Is Nothing) Then
        GetCountFraction = 0
    ElseIf statusFractions.Exists(statusName) Then
        If IsNumeric(statusFractions(statusName)) Then
            GetCountFraction = CDbl(statusFractions(statusName))
        Else
            GetCountFraction = 0
        End If
    Else
        GetCountFraction = 0
    End If
End Function

' ─────────────────────────────────────────────────────────────────────
' Combine 5 → 3 (your rule), normalize if overlap makes sum > 1
' ─────────────────────────────────────────────────────────────────────
Private Sub GetCombinedFractions(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal t1, ByVal t2, ByRef fRun, ByRef fHalt, ByRef fAbort)
    Dim a, h, r, w, i, total, statusFractions

    ' Initialise outputs so callers always receive well-defined values
    fRun = 0 : fHalt = 0 : fAbort = 0

    Set statusFractions = GetStatusFractionsBatch(cn, runtimeCatalog, machineId, periodKey, t1, t2)

    a = GetCountFraction(statusFractions, "Aborted")
    h = GetCountFraction(statusFractions, "Halted")
    r = GetCountFraction(statusFractions, "Running")
    w = GetCountFraction(statusFractions, "Waiting")
    i = GetCountFraction(statusFractions, "Interrupted")

    fRun   = r
    fHalt  = h + i
    fAbort = a + w

    ' clamp just in case
    If fRun < 0 Then fRun = 0
    If fHalt < 0 Then fHalt = 0
    If fAbort < 0 Then fAbort = 0

    ' If your PLC logic overlaps states, sum can exceed 1 → normalize safely
    total = fRun + fHalt + fAbort
    If total > 1.000001 Then
        fRun   = fRun / total
        fHalt  = fHalt / total
        fAbort = fAbort / total
    End If
End Sub

' ─────────────────────────────────────────────────────────────────────
' Output helpers
' ─────────────────────────────────────────────────────────────────────
Private Function OutTagName(ByVal machineId, ByVal periodKey, ByVal statusDispName, ByVal pieChart)
    If pieChart Then
        OutTagName = machineId & "_" & periodKey & "_Machine_" & statusDispName & "_Percentage_PieChart"
    Else
        OutTagName = machineId & "_" & periodKey & "_Machine_" & statusDispName & "_Percentage"
    End If
End Function

Private Sub WriteTag(ByVal tag, ByVal val)
    Dim readBack, requestedVal, delta

    On Error Resume Next
    HMIRuntime.Tags(tag).Write val
    If Err.Number <> 0 Then
        LogMsg LOG_LEVEL_WARN, "HMITag", "Write failed: " & S(tag) & " -> " & S(Err.Description)
        Err.Clear
        On Error GoTo 0
        Exit Sub
    End If

    LogMsg LOG_LEVEL_DEBUG, "HMITag", "write_ok tag=" & S(tag) & " value=" & S(val)

    If CURRENT_LOG_LEVEL >= LOG_LEVEL_DEBUG Then
        readBack = HMIRuntime.Tags(tag).Read
        If Err.Number <> 0 Then
            LogMsg LOG_LEVEL_WARN, "HMITag", "Read-back failed: " & S(tag) & " -> " & S(Err.Description)
            Err.Clear
        ElseIf IsNumeric(readBack) And IsNumeric(val) Then
            requestedVal = CDbl(val)
            delta = Abs(CDbl(readBack) - requestedVal)
            If delta > WRITE_VERIFY_TOLERANCE Then
                LogMsg LOG_LEVEL_WARN, "HMITag", "write_mismatch tag=" & S(tag) & " wrote=" & requestedVal & " read=" & CDbl(readBack) & " delta=" & delta
            End If
        Else
            LogMsg LOG_LEVEL_DEBUG, "HMITag", "write_verify_non_numeric tag=" & S(tag) & " wrote=" & S(val) & " read=" & S(readBack)
        End If
    End If
    On Error GoTo 0
End Sub

' ─────────────────────────────────────────────────────────────────────
' Aggregate one period (writes Aborted/Halted/Running only)
' ─────────────────────────────────────────────────────────────────────
Private Sub AggregatePeriod(ByVal cn, ByVal runtimeCatalog, ByVal machineId, ByVal periodKey, ByVal startDt, ByVal endDt)
    Dim fRun, fHalt, fAbort
    Dim pctRun, pctHalt, pctAbort
    Dim tagPct, tagPie, pieVal
    Dim totalPct

    LogMsg LOG_LEVEL_INFO, "Aggregate", machineId & " " & periodKey & "  " & TS3(startDt) & " -> " & TS3(endDt)

    Call GetCombinedFractions(cn, runtimeCatalog, machineId, periodKey, startDt, endDt, fRun, fHalt, fAbort)

    pctRun   = Round(CDbl(fRun)   * 100, 2)
    pctHalt  = Round(CDbl(fHalt)  * 100, 2)
    pctAbort = Round(CDbl(fAbort) * 100, 2)
    totalPct = Round(pctRun + pctHalt + pctAbort, 2)

    LogMsg LOG_LEVEL_INFO, "Aggregate", _
           "computed machine=" & machineId & _
           " period=" & periodKey & _
           " pctRun=" & pctRun & _
           " pctHalt=" & pctHalt & _
           " pctAbort=" & pctAbort & _
           " totalPct=" & totalPct

    ' main % (can be 0)
    tagPct = OutTagName(machineId, periodKey, "Running", False): WriteTag tagPct, pctRun
    tagPct = OutTagName(machineId, periodKey, "Halted",  False): WriteTag tagPct, pctHalt
    tagPct = OutTagName(machineId, periodKey, "Aborted", False): WriteTag tagPct, pctAbort

    ' pie chart (min 1 for display, DOES NOT affect main %)
    ' Only apply the minimum-1 floor when there is actual activity; if the
    ' machine was completely idle all three raw percentages are 0 and forcing
    ' each to 1 would create a false 33/33/33 split on the chart.

    pieVal = pctRun:   If pieVal < 1 And totalPct > 0 Then pieVal = 1
    tagPie = OutTagName(machineId, periodKey, "Running", True): WriteTag tagPie, pieVal

    pieVal = pctHalt:  If pieVal < 1 And totalPct > 0 Then pieVal = 1
    tagPie = OutTagName(machineId, periodKey, "Halted", True): WriteTag tagPie, pieVal

    pieVal = pctAbort: If pieVal < 1 And totalPct > 0 Then pieVal = 1
    tagPie = OutTagName(machineId, periodKey, "Aborted", True): WriteTag tagPie, pieVal

    LogMsg LOG_LEVEL_INFO, "Aggregate", "writes_completed machine=" & machineId & " period=" & periodKey
End Sub

Private Function ShouldRun(ByVal lastRun, ByVal throttleMin)
    If IsEmpty(lastRun) Or IsNull(lastRun) Then
        ShouldRun = True
        Exit Function
    End If
    ShouldRun = (DateDiff("n", lastRun, Now()) >= throttleMin)
End Function

' ─────────────────────────────────────────────────────────────────────
' Main Entry Point
' ─────────────────────────────────────────────────────────────────────
Public Sub MainStatusAggregation(ByVal machineId)
    Dim cn, rtCat, nowDT
    Dim tFrom, tTo
    Dim key, lastRun
    Dim due1D, due7D, due30D
    Dim startedAt, completedAt, startTimer, totalElapsedMs
    Dim guardReason, lockAcquired, canRun

    ' WinCC VBScript does not support label-based handlers (On Error GoTo FAIL).
    On Error Resume Next

    startedAt = Now
    startTimer = Timer
    guardReason = "completed"
    lockAcquired = False
    canRun = True
    Set cn = Nothing

    InitGlobals

    If Not TryAcquireMachineLock(machineId) Then
        LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", "Lock already active; skipping machineId: " & S(machineId)
        guardReason = "skip_lock_held"
        canRun = False
    End If
    If canRun Then lockAcquired = True

    If canRun Then
        If Not IsMaster() Then
            guardReason = "skip_not_master"
            canRun = False
        End If
    End If

    If canRun Then
        If Not G_MACHINES.Exists(machineId) Then
            LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", "Unknown machineId: " & S(machineId)
            guardReason = "skip_unknown_machine"
            canRun = False
        End If
    End If

    If canRun Then
        nowDT = Now()
        due1D = False : due7D = False : due30D = False

        key = machineId & "|1D"
        lastRun = Empty
        If g_lastRunByMachinePeriod.Exists(key) Then lastRun = g_lastRunByMachinePeriod(key)
        due1D = ShouldRun(lastRun, THROTTLE_1D_MINUTES)

        key = machineId & "|7D"
        lastRun = Empty
        If g_lastRunByMachinePeriod.Exists(key) Then lastRun = g_lastRunByMachinePeriod(key)
        due7D = ShouldRun(lastRun, THROTTLE_7D_MINUTES)

        key = machineId & "|30D"
        lastRun = Empty
        If g_lastRunByMachinePeriod.Exists(key) Then lastRun = g_lastRunByMachinePeriod(key)
        due30D = ShouldRun(lastRun, THROTTLE_30D_MINUTES)

        ' Important: avoid SQL connection work when nothing is due.
        If Not (due1D Or due7D Or due30D) Then
            guardReason = "skip_throttle"
            canRun = False
        End If
    End If

    If canRun Then
        Set cn = OpenConnection()
        If cn Is Nothing Then
            LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", "ERROR: cannot open SQL connection"
            guardReason = "fail_open_connection"
            canRun = False
        ElseIf cn.State <> adStateOpen Then
            LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", "ERROR: cannot open SQL connection"
            guardReason = "fail_connection_state"
            canRun = False
        End If
    End If

    If canRun Then rtCat = GetRuntimeCatalog()
    If canRun Then
        LogMsg LOG_LEVEL_INFO, "MainStatusAggregation", _
               "aggregation_start machine=" & machineId & _
               " cn_state=" & CStr(cn.State) & _
               " runtime_catalog=" & rtCat
    End If

    ' Execute at most one period per invocation, with fixed priority:
    ' 1D -> 7D -> 30D
    If canRun Then
        If due1D Then
            tTo = nowDT: tFrom = DateAdd("d", -1, tTo)
            AggregatePeriod cn, rtCat, machineId, "1D", tFrom, tTo
            g_lastRunByMachinePeriod(machineId & "|1D") = nowDT
            guardReason = "ran_1D"
        ElseIf due7D Then
            tTo = nowDT: tFrom = DateAdd("d", -7, tTo)
            AggregatePeriod cn, rtCat, machineId, "7D", tFrom, tTo
            g_lastRunByMachinePeriod(machineId & "|7D") = nowDT
            guardReason = "ran_7D"
        ElseIf due30D Then
            tTo = nowDT: tFrom = DateAdd("d", -30, tTo)
            AggregatePeriod cn, rtCat, machineId, "30D", tFrom, tTo
            g_lastRunByMachinePeriod(machineId & "|30D") = nowDT
            guardReason = "ran_30D"
        End If
    End If

CLEANUP:
    If IsObject(cn) Then CloseConn cn
    If lockAcquired Then ReleaseMachineLock machineId

    If Err.Number <> 0 Then
        guardReason = "fail_runtime_error"
        LogMsg LOG_LEVEL_ERROR, "MainStatusAggregation", _
               "Runtime error number=" & CStr(Err.Number) & _
               " description=" & S(Err.Description)
        Err.Clear
    End If

    On Error GoTo 0

    completedAt = Now
    totalElapsedMs = ElapsedMs(startTimer)
    LogMsg LOG_LEVEL_WARN, "MainStatusAggregation", _
           "runtime_guard machine=" & machineId & _
           " started=" & TS3(startedAt) & _
           " ended=" & TS3(completedAt) & _
           " elapsed_ms=" & totalElapsedMs & _
           " reason=" & guardReason
    Exit Sub
End Sub

Private Function TryAcquireMachineLock(ByVal machineId)
    If g_machineLocks.Exists(machineId) Then
        TryAcquireMachineLock = False
    Else
        g_machineLocks.Add machineId, True
        TryAcquireMachineLock = True
    End If
End Function

Private Sub ReleaseMachineLock(ByVal machineId)
    If g_machineLocks.Exists(machineId) Then
        g_machineLocks.Remove machineId
    End If
End Sub
